namespace = AIO_Aforgomon_events

# Victim callback: only direct full strikes respawn, never final-situation fleets.
country_event = {
    id = AIO_Aforgomon_events.202
    hide_window = yes
    is_triggered_only = yes
    trigger = { fromfrom? = { has_fleet_flag = AIO_DAIT_strike_fleet } }
    immediate = {
        fromfrom.solar_system = {
            if = {
                limit = {
                    has_star_flag = AIO_DAIT_strike_active
                    has_star_flag = AIO_DAIT_strike_complete
                }
                if = {
                    limit = { check_variable = { which = AIO_Aforgomon_Wrath value < 10 } }
                    change_variable = { which = AIO_Aforgomon_Wrath value = 1 }
                }
                AIO_create_Aforgomon_fleet = {
                    DESIGN = NAME_AIO_Aforgomon_Wrath
                    FLAG = AIO_DAIT_strike_fleet
                }
            }
        }
    }
}

system_event = {
    id = AIO_Aforgomon_events.203
    hide_window = yes
    is_triggered_only = yes
    trigger = { has_star_flag = AIO_DAIT_strike_active }
    immediate = {
        remove_star_flag = AIO_DAIT_strike_active
        every_fleet_in_system = {
            limit = { has_fleet_flag = AIO_DAIT_strike_fleet }
            delete_fleet = this
        }
        if = {
            limit = { has_star_flag = AIO_DAIT_strike_complete }
            AIO_DAIT_erase_system_effect = yes
        }
        remove_star_flag = AIO_DAIT_strike_complete
        remove_star_flag = AIO_Aforgomon_existed
        clear_variable = AIO_Aforgomon_Wrath
    }
}
