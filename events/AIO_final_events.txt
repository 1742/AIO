    id = AIO_final_events.301
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        has_country_flag = AIO_final_start
        NOT = { any_situation = { is_situation_type = AIO_final_location_situation } }
    }
    immediate = { AIO_start_location = yes }
}

# 只在前兆阶段修复目标；显现中的星系本来就有占用标志。
situation_event = {
    id = AIO_final_events.302
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        current_stage = AIO_final_omen_stage
        OR = {
            NOT = { exists = target }
            target? = { AIO_final_valid_target = no }
        }
    }
    immediate = {
        AIO_final_repair_target = yes
        if = {
            limit = {
                exists = target
                owner = { has_country_flag = AIO_final_start }
            }
            owner = {
                create_message = {
                    type = MESSAGE_DYSON_SHOT
                    localization = AIO_final_location_started
                    days = 30
                    target = root.target
                }
            }
        }
    }
}

    id = AIO_final_events.502
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        exists = fromfrom
        has_country_flag = AIO_Aforgomon_country
    }
    immediate = {
        every_playable_country = {
            limit = { has_country_flag = AIO_final_start }
            random_situation = {
                limit = {
                    is_situation_type = AIO_final_location_situation

                    has_situation_flag = AIO_final_constrainted
                    NOT = { has_situation_flag = AIO_final_settled }
                    fromfrom = { has_fleet_flag = AIO_final_fleet_@prev }
                }
                set_situation_flag = AIO_final_settled
                owner = { AIO_final_win = yes }
            }
        }
    }
