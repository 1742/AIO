# AIO 亚弗戈蒙相关事件
namespace = AIO_Aforgomom_events


# 销毁亚弗戈蒙之怒事件
fleet_event = {
    id = AIO_Aforgomom_events.201
    hide_window =yes

    is_triggered_only = yes
    trigger = { has_fleet_flag = AIO_Aforgomom_fleet }

    immediate = {
        # 时间到了没打完，且数值没有溢出
        if = {
            limit = {
                solar_system = { any_fleet_in_system = { NOT = { has_fleet_flag = AIO_Aforgomom_fleet } } }
                owner = { check_variable = { which = AIO_Aforgomom_Wrath value < 10 } }
            }
            owner = { change_variable = { which = AIO_Aforgomom_Wrath value = 1 } }
            add_modifier = { modifier = AIO_Aforgomom_Wrath_modifier mult = value:AIO_Aforgomom_Wrath_factor }
            fleet_event = { id = AIO_Aforgomom_events.201 days = 100 }
        }
        # 时间到了没打完，且数值溢出，直接删船
        else_if = {
            limit = {
                solar_system = { any_fleet_in_system = { NOT = { has_fleet_flag = AIO_Aforgomom_fleet } } }
                owner = { check_variable = { which = AIO_Aforgomom_Wrath value >= 10 } }
            }
            owner = { set_variable = { which = AIO_Aforgomom_Wrath value = 0 } }
            solar_system = {
                every_fleet_in_system = {
                    if = {
                        limit = { NOT = { has_fleet_flag = AIO_Aforgomom_fleet } }
                        destroy_fleet = this
                    }
                    else = { delete_fleet = this }
                }
                remove_star_flag = AIO_Aformom_exist
            }
        }
        else = {
            owner = { set_variable = { which = AIO_Aforgomom_Wrath value = 0 } }
            delete_fleet = this
            solar_system = { remove_star_flag = AIO_Aformom_exist }
        }
    }
}


# 亚弗戈蒙之怒被摧毁
# This = owner of fleet 1 (combatant)
# From = owner of fleet 2 (destroyed)
# FromFrom = fleet 1
# FromFromFrom = fleet 2
country_event = {
    id = AIO_Aforgomom_events.202
    hide_window = yes

    is_triggered_only = yes
    trigger = { fromfromfrom = { has_fleet_flag = AIO_Aforgomom_fleet } }

    immediate = {
        # 打不过，加buff
        from = {
            if = {
                limit = { check_variable = { which = AIO_Aforgomom_Wrath value < 10 } }
                change_variable = { which = AIO_Aforgomom_Wrath value = 1 }
            }
        }
        fromfrom.solar_system = { AIO_create_Aformom_fleet = yes }
    }
}
