namespace = AIO_Aforgomom_events

# 正式化身每 100 天检查本星系；锚点不会进入这条事件链。
fleet_event = {
	id = AIO_Aforgomom_events.201
	hide_window = yes
	is_triggered_only = yes
	trigger = { has_fleet_flag = AIO_Aforgomom_fleet }
	immediate = {
		solar_system = {
			if = {
				limit = {
					AIO_Aforgomom_has_targets = yes
					check_variable = { which = AIO_Aforgomom_Wrath value < 10 }
				}
				change_variable = { which = AIO_Aforgomom_Wrath value = 1 }
				root = {
					add_modifier = { modifier = AIO_Aforgomom_Wrath_modifier mult = value:AIO_Aforgomom_Wrath_factor }
					fleet_event = { id = AIO_Aforgomom_events.201 days = 100 }
				}
			}
			else = {
				# 无目标则离去；怒气到阈值时清场。两者都先结束本星系状态。
				remove_star_flag = AIO_Aformom_exist
				clear_variable = AIO_Aforgomom_Wrath
				every_fleet_in_system = {
					limit = { NOT = { is_owned_by = event_target:AIO_Aforgomom_country } }
					destroy_fleet = this
				}
				root = { delete_fleet = this }
			}
		}
	}
}

# ROOT = 被摧毁舰队的国家；FROMFROM = 被摧毁舰队。
country_event = {
	id = AIO_Aforgomom_events.202
	hide_window = yes
	is_triggered_only = yes
	trigger = { fromfrom = { has_fleet_flag = AIO_Aforgomom_fleet } }
	immediate = {
		fromfrom.solar_system = {
			if = {
				limit = { has_star_flag = AIO_Aformom_exist }
				if = {
					limit = { check_variable = { which = AIO_Aforgomom_Wrath value < 10 } }
					change_variable = { which = AIO_Aforgomom_Wrath value = 1 }
				}
				AIO_create_Aformom_fleet = yes
			}
		}
	}
}
