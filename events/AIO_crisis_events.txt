#region 初始化事件

# 起源
country_event = {
    id = AIO_crisis_events.201
    hide_window = yes

    is_triggered_only = yes
    trigger = {
        has_origin = origin_AIO_YogSothoth
        no_scope = { NOT = { any_country = { is_country_type = AIO_empire } } }
    }
    fire_only_once = yes

    # Galaxy generation still has pending origin/colony/ship initialization.
    # Finish ascension on the first live day, after those references are ready.
    immediate = { country_event = { id = AIO_crisis_events.202 days = 1 } }
}

country_event = {
    id = AIO_crisis_events.202
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        has_origin = origin_AIO_YogSothoth
        NOT = { is_country_type = AIO_empire }
        no_scope = { NOT = { any_country = { is_country_type = AIO_empire } } }
    }

    immediate = {
        set_country_flag = AIO_built_AIO_horizon_engine
        set_country_flag = AIO_built_AIO_mega_01
        set_country_flag = AIO_built_AIO_mega_02
        set_country_flag = AIO_built_AIO_mega_03
        set_country_flag = AIO_built_AIO_mega_04
        set_country_flag = AIO_empire_country
        save_global_event_target_as = AIO_empire_country
        # 允许进入牢塞星系
        set_country_flag = protected_from_queen_storm

        # 激活灾飞路径
        add_ascension_perk = ap_AIO_allinone
        add_resource = { AIO_advanced_logic = 45000 }

        # 空想要3点国民理念点，不加够会转变政体失败
        give_technology = { tech = tech_planetary_unification }
        give_technology = { tech = tech_colonial_centralization }
        give_technology = { tech = tech_galactic_administration }
        # 创建穷极之门
        capital_scope.solar_system = {
            spawn_megastructure = {
                type = AIO_mega_01_fact
                planet = star
                owner = root
                graphical_culture = root
            }
            # 先保留首都，转为无需殖民地的国家类型后再清理星系。
            set_star_flag = AIO_gate_system
            save_global_event_target_as = AIO_gate_system
        }
        # 创建无名之雾
        AIO_create_mist_system_effect = yes
        # 创建犹格索托斯
        event_target:AIO_mist_system = {
            random_system_megastructure = {
                limit = { is_megastructure_type = AIO_mega_01 }
                remove_megastructure = this
            }
            spawn_megastructure = {
                type = AIO_mega_04
                planet = star
                owner = root
            }
        }

        AIO_ascension_effect = yes
        event_target:AIO_gate_system = { AIO_create_gate_system_effect = yes }
    }
}

# Creates a purpose-built, non-virtual machine species for the Matryoshka Brain.
# It is a separate species identity so cybernetics.5049 cannot digitize it again.
country_event = {
    id = AIO_crisis_events.902
    title = AIO_crisis_events.902.title
    desc = AIO_crisis_events.902.desc
    picture = GFX_evt_robot_assembly_plant
    show_sound = event_radio_chatter

    is_triggered_only = yes

    trigger = {
        is_country_type = default
        is_virtual_empire = yes
        has_technology = tech_AIO_exponential_learning_theory
        NOT = { has_country_flag = AIO_recursive_chassis_created }
    }

    immediate = {
        set_country_flag = AIO_recursive_chassis_created
        capital_scope = { save_event_target_as = AIO_recursive_chassis_colony }
        random_owned_colony = {
            limit = { has_district = district_AIO_lathe }
            save_event_target_as = AIO_recursive_chassis_colony
        }
    }

    option = {
        name = AIO_crisis_events.902.option
        default_hide_option = yes
        owner_main_species = {
            create_species = {
                name = "AIO_RECURSIVE_CHASSIS"
                plural = "AIO_RECURSIVE_CHASSIS_PLURAL"
                class = this
                portrait = this
                traits = this
                can_be_modified = yes
                homeworld = this
                namelist = this
                gender = this

                effect = {
                    change_species_characteristics = {
                        remove_trait = trait_robot_digital_1
                        remove_trait = trait_robot_digital_2
                        add_trait = trait_AIO_recursive_chassis
                    }
                    if = {
                        limit = { root = { is_machine_empire = yes } }
                        set_citizenship_type = {
                            country = root
                            type = citizenship_full_machine
                            cooldown = no
                        }
                    }
                    else = {
                        set_citizenship_type = {
                            country = root
                            type = citizenship_full
                            cooldown = no
                        }
                    }
                    set_population_controls = {
                        country = root
                        type = population_control_no
                        cooldown = no
                    }
                    set_species_flag = AIO_recursive_chassis_species
                    save_event_target_as = AIO_recursive_chassis_species
                }
            }
        }
        event_target:AIO_recursive_chassis_colony = {
            create_pop_group = {
                species = event_target:AIO_recursive_chassis_species
                size = 1000
            }
        }
    }
}
