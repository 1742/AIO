namespace = AIO_DAIT_events

# scripted_action 的星系作用域桥接事件。
# THIS = 目标星系
# FROM = 发起动作的巨构
system_event = {
	id = AIO_DAIT_events.201
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		# 后续菜单和效果统一使用这两个 event target。
		save_event_target_as = AIO_DAIT_target_system
		from = {
			save_event_target_as = AIO_DAIT_source_megastructure
			owner = {
				country_event = { id = AIO_DAIT_events.202 }
			}
		}
	}
}

country_event = {
	id = AIO_DAIT_events.202
	title = AIO_DAIT_events.202.title
	desc = AIO_DAIT_events.202.desc

	is_triggered_only = yes

	option = {
		name = AIO_DAIT_events.202.creation
		country_event = { id = AIO_DAIT_events.203 }
	}

	option = {
		name = AIO_DAIT_events.202.redefinition
		country_event = { id = AIO_DAIT_events.204 }
	}

	option = {
		name = AIO_DAIT_events.202.erasure
		country_event = { id = AIO_DAIT_events.205 }
	}

	option = {
		name = AIO_DAIT_events.202.cancel
		default_hide_option = yes
	}
}

country_event = {
	id = AIO_DAIT_events.203
	title = AIO_DAIT_events.203.title
	desc = AIO_DAIT_events.203.desc

	is_triggered_only = yes

	option = {
		name = AIO_DAIT_events.203.create_habitable_world
		trigger = {
			event_target:AIO_DAIT_target_system = {
				any_system_planet = {
					is_star = no
					is_artificial = no
					is_colony = no
					is_under_colonization = no
					habitable_planet = no
					planet_size > 5
					AIO_safe_remove_planet_exclude_API = no
				}
			}
		}
		event_target:AIO_DAIT_target_system = { AIO_DAIT_create_habitable_world_effect = yes }
		country_event = { id = AIO_DAIT_events.202 }
	}

	option = {
		name = AIO_DAIT_events.203.create_life
		trigger = {
			event_target:AIO_DAIT_target_system = {
				any_system_planet = {
					habitable_planet = yes
					NOT = { any_owned_pop_group = { is_sapient = no } }
				}
			}
		}
		event_target:AIO_DAIT_target_system = {
			AIO_DAIT_create_life_effect = yes
		}
		country_event = { id = AIO_DAIT_events.202 }
	}

	option = {
		name = AIO_DAIT_events.back
		country_event = { id = AIO_DAIT_events.202 }
		default_hide_option = yes
	}
}

			event_target:AIO_DAIT_target_system = {

		name = AIO_DAIT_events.back

	}
}
