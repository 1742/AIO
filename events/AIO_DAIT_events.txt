	id = AIO_DAIT_events.201
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		solar_system = { save_event_target_as = AIO_DAIT_target_system }

			save_event_target_as = AIO_DAIT_source_megastructure
			owner = { country_event = { id = AIO_DAIT_events.202 } }
		}
	}
}

country_event = {
	id = AIO_DAIT_events.203
	title = AIO_DAIT_events.203.title
	desc = AIO_DAIT_events.203.desc

	is_triggered_only = yes

	option = {
		name = AIO_DAIT_events.203.add_deposits
		allow = {
			event_target:AIO_DAIT_target_system = {
				any_system_planet = { NOT = { is_planet_class = pc_AIO_empty_space } }
			}
		}
		event_target:AIO_DAIT_target_system = { AIO_DAIT_add_deposits_effect = { TIMES = 10 } }
		country_event = { id = AIO_DAIT_events.203 }
	}

	option = {
		name = AIO_DAIT_events.203.reshape_habitable_planet
		allow = {
			event_target:AIO_DAIT_target_system = {
				any_system_planet = {
					is_star = no
					is_artificial = no
					is_colony = no
					habitable_planet = no
					AIO_safe_remove_planet_exclude_API = no
				}
			}
		}
		event_target:AIO_DAIT_target_system = { AIO_DAIT_reshape_habitable_planet_effect = yes }
		country_event = { id = AIO_DAIT_events.203 }
	}

	option = {
		name = AIO_DAIT_events.203.create_life
		allow = {
			event_target:AIO_DAIT_target_system = {
				any_system_planet = {
					habitable_planet = yes
					pop_amount = 0
				}
			}
		}
		event_target:AIO_DAIT_target_system = { AIO_DAIT_create_life_effect = yes }
		country_event = { id = AIO_DAIT_events.203 }
	}

	option = {
		name = AIO_DAIT_events.203.build_bypass
		allow = {
			custom_tooltip = {
				fail_text = AIO_DAIT_locak_target.fail.build_bypass
				event_target:AIO_DAIT_target_system = {
					NOT = { any_bypass = { is_bypass_type = bypass_AIO_ultimate_gate } }
				}
			}
		}
		event_target:AIO_DAIT_target_system = { AIO_DAIT_build_bypass_effect = yes }
		country_event = { id = AIO_DAIT_events.203 }
	}

	option = {
		name = AIO_DAIT_events.203.create_storm
		allow = { event_target:AIO_DAIT_target_system = { is_inside_storm = no } }
		event_target:AIO_DAIT_target_system = { AIO_DAIT_create_storm_effect = yes }
		country_event = { id = AIO_DAIT_events.203 }
	}

	option = {
		name = AIO_DAIT_events.back
		country_event = { id = AIO_DAIT_events.202 }
		default_hide_option = yes
	}
}

country_event = {
	id = AIO_DAIT_events.204
	title = AIO_DAIT_events.204.title
	desc = AIO_DAIT_events.204.desc

	is_triggered_only = yes

	option = {
		name = AIO_DAIT_events.204.reshape_system
		allow = {
			custom_tooltip = {
				fail_text = AIO_DAIT_locak_target.fail.colony
				event_target:AIO_DAIT_target_system = {
					any_system_colony = {
						NOT = { owner? = { is_same_empire = root } }
					}
				}
			}
			custom_tooltip = {
				fail_text = AIO_DAIT_locak_target.fail.crisis_path
				event_target:AIO_DAIT_target_system = { NOT = { has_star_flag = AIO_TUG_system } }
			}
			custom_tooltip = {
				fail_text = AIO_DAIT_locak_target.fail.synth_queen
				event_target:AIO_DAIT_target_system = { NOT = { has_star_flag = synth_queen_capital } }
			}
		}
		event_target:AIO_DAIT_target_system = { AIO_DAIT_make_special_system_effect = yes }
		country_event = { id = AIO_DAIT_events.204 }
	}

	option = {
		name = AIO_DAIT_events.204.degenerate
		allow = {
			custom_tooltip = {
				fail_text = AIO_DAIT_locak_target.fail.colony
				event_target:AIO_DAIT_target_system = {
					any_system_colony = {
						NOT = { owner? = { is_same_empire = root } }
					}
				}
			}
			custom_tooltip = {
				fail_text = AIO_DAIT_locak_target.fail.degenerate
				event_target:AIO_DAIT_target_system = {
					any_system_colony = {
						owner? = {
							OR = {
								is_country_type = default
								is_country_type = fallen_empire
								is_country_type = awakened_fallen_empire
							}
						}
					}
				}
			}
		}
		event_target:AIO_DAIT_target_system = { AIO_DAIT_degenerate_effect = yes }
		country_event = { id = AIO_DAIT_events.204 }
	}

	option = {
		name = AIO_DAIT_events.204.enlighten
		allow = {
			custom_tooltip = {
				fail_text = AIO_DAIT_locak_target.fail.synth_queen
				event_target:AIO_DAIT_target_system = {
					any_system_colony = {
						NOT = { has_carrier_flag = synth_queen_bastille }
					}
				}
			}
			event_target:AIO_DAIT_target_system = {
				any_system_planet = {
					pop_amount > 1
					NOT = { is_planet_class = pc_ai }
				}
			}
		}
		event_target:AIO_DAIT_target_system = { AIO_DAIT_enlighted_effect = yes }
		country_event = { id = AIO_DAIT_events.204 }
	}

	option = {
		name = AIO_DAIT_events.204.change_ethic
		allow = { event_target:AIO_DAIT_target_system = { exists = owner } }
		event_target:AIO_DAIT_target_system = { owner = { AIO_DAIT_change_ethic_effect = yes } }
		country_event = { id = AIO_DAIT_events.204 }
	}

	option = {
		name = AIO_DAIT_events.back
		country_event = { id = AIO_DAIT_events.202 }
		default_hide_option = yes
	}
}
