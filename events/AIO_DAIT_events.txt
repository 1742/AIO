namespace = AIO_DAIT_events

#region 定向应用入口

# scripted_action
# THIS = 目标
# FROM = 发起动作的巨构
system_event = {
	id = AIO_DAIT_events.201
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		save_global_event_target_as = AIO_DAIT_target_system
		from = {
			set_megastructure_flag = AIO_DAIT_active
			owner = {
				set_country_flag = AIO_DAIT_active
				save_global_event_target_as = AIO_DAIT_maker
				country_event = { id = AIO_DAIT_events.202 }
			}
		}
	}
}

country_event = {
	id = AIO_DAIT_events.202
	title = AIO_DAIT_text
	desc = AIO_DAIT_remote_projection_placeholder_desc
	picture = GFX_evt_shroud_psionic_aura

	is_triggered_only = yes

	option = {
		name = EXIT
		default_hide_option = yes
	}

	after = {
		remove_country_flag = AIO_DAIT_active
		clear_global_event_target = AIO_DAIT_maker
		fromfrom = { remove_megastructure_flag = AIO_DAIT_active }
		clear_global_event_target = AIO_DAIT_target_system
	}
}

#endregion
