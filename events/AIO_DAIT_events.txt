namespace = AIO_DAIT_events

# scripted_action 的行星作用域桥接事件。
# THIS = 目标行星（planet/carrier）
# FROM = 发起动作的巨构
planet_event = {
	id = AIO_DAIT_events.201
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		# 行星操作使用 target_planet；星系操作使用 target_system。
		save_event_target_as = AIO_DAIT_target_planet
		solar_system = { save_event_target_as = AIO_DAIT_target_system }
		from = {
			save_event_target_as = AIO_DAIT_source_megastructure
			owner = { country_event = { id = AIO_DAIT_events.202 } }
		}
	}
}

country_event = {
	id = AIO_DAIT_events.203
	title = AIO_DAIT_events.203.title
	desc = AIO_DAIT_events.203.desc

	is_triggered_only = yes

	option = {
		name = AIO_DAIT_events.203.add_deposits
		trigger = {
			event_target:AIO_DAIT_target_system = {
				any_system_planet = { NOT = { is_planet_class = pc_AIO_empty_space } }
			}
		}
		event_target:AIO_DAIT_target_system = { AIO_DAIT_create_habitable_world_effect = yes }
		country_event = { id = AIO_DAIT_events.202 }
	}

	option = {
		name = AIO_DAIT_events.203.create_habitable_world
		trigger = {
			event_target:AIO_DAIT_target_planet = {
				is_star = no
				is_artificial = no
				is_colony = no
				habitable_planet = no
				planet_size > 5
				AIO_safe_remove_planet_exclude_API = no
			}
		}
		event_target:AIO_DAIT_target_planet = { AIO_DAIT_create_habitable_world_effect = yes }
		country_event = { id = AIO_DAIT_events.202 }
	}

	option = {
		name = AIO_DAIT_events.203.create_life
		trigger = {
			event_target:AIO_DAIT_target_planet = {
				habitable_planet = yes
				pop_amount = 0
			}
		}
		event_target:AIO_DAIT_target_planet = { AIO_DAIT_create_life_effect = yes }
		country_event = { id = AIO_DAIT_events.202 }
	}

	option = {
		name = AIO_DAIT_events.203.create_storm
		trigger = {
			event_target:AIO_DAIT_target_system = {
				NOR = {
					is_inside_storm = yes
					has_star_flag = AIO_HS_system
				}
			}
		}
		event_target:AIO_DAIT_target_system = { AIO_DAIT_create_storm_effect = yes }
		country_event = { id = AIO_DAIT_events.202 }
	}

	option = {
		name = AIO_DAIT_events.back
		country_event = { id = AIO_DAIT_events.202 }
		default_hide_option = yes
	}
}

country_event = {
	id = AIO_DAIT_events.204
	title = AIO_DAIT_events.204.title
	desc = AIO_DAIT_events.204.desc

	is_triggered_only = yes

	option = {
		name = AIO_DAIT_events.204.reshape_system
		trigger = {
			event_target:AIO_DAIT_target_system = {
				NOR = {
					has_star_flag = AIO_HS_system
					has_star_flag = AIO_TUG_system
					any_system_planet = { is_planet_class = pc_ai }
					has_star_flag = synth_queen_capital
					AIO_is_crisis_mega_system = yes
					AIO_safe_remove_system_exclude_API = yes
				}
			}
		}
		event_target:AIO_DAIT_target_system = { AIO_DAIT_make_special_system_effect = yes }
		country_event = { id = AIO_DAIT_events.202 }
	}

	option = {
		name = AIO_DAIT_events.204.degenerate
		trigger = {
			event_target:AIO_DAIT_target_planet = {
				colony? = {
					owner = { is_country_type = default }
					planet = { AIO_safe_remove_planet_exclude_API = no }
				}
			}
		}
		event_target:AIO_DAIT_target_planet = { AIO_DAIT_degenerate_effect = yes }
		country_event = { id = AIO_DAIT_events.202 }
	}

	option = {
		name = AIO_DAIT_events.204.enlighten
		trigger = { event_target:AIO_DAIT_target_planet = { pop_amount > 1 } }
		event_target:AIO_DAIT_target_planet = { AIO_DAIT_enlighted_effect = yes }
		country_event = { id = AIO_DAIT_events.202 }
	}

	option = {
		name = AIO_DAIT_events.204.change_ethic
		trigger = { event_target:AIO_DAIT_target_system = { exists = owner } }
		event_target:AIO_DAIT_target_system = { AIO_DAIT_change_ethic_effect = yes }
		country_event = { id = AIO_DAIT_events.202 }
	}

	option = {
		name = AIO_DAIT_events.back
		country_event = { id = AIO_DAIT_events.202 }
		default_hide_option = yes
	}
}
