namespace = AIO_DAIT_events

country_event = {
    id = AIO_DAIT_events.201
    title = AIO_DAIT_events.201.title
    location = root

	diplomatic = yes
	custom_gui = "AIO_DAIT"

    is_triggered_only = yes
    trigger = {
        OR = {
			has_country_flag = AIO_built_AIO_mega_01
			has_country_flag = AIO_built_AIO_mega_02
			has_country_flag = AIO_built_AIO_mega_03
			has_country_flag = AIO_built_AIO_mega_04
		}
		NOT = { has_active_event = { AIO_DAIT_events.201 } }
	}

    option = {
        name = OK
		default_hide_option = yes
    }

	after = {
		if = {
			limit = { NOT = { has_active_event = { AIO_DAIT_events.202 } } }
			country_event = { id = AIO_DAIT_events.202 }
		}
	}
}

# 每季度更新 DAIT buff
country_event = {
	id = AIO_DAIT_events.202
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		OR = {
			has_country_flag = AIO_built_AIO_mega_01
			has_country_flag = AIO_built_AIO_mega_02
			has_country_flag = AIO_built_AIO_mega_03
			has_country_flag = AIO_built_AIO_mega_04
		}
		NOT = { is_country_type = AIO_empire }
	}

	immediate = {
		AIO_DAIT_add_buffs = yes

		# 给科技
		export_resource_income_to_variable = { resource = AIO_advanced_logic variable = AIO_DAIT_temp_give_tech }
		divide_variable = { which = AIO_DAIT_temp_give_tech value = 10000 }
		floor_variable = AIO_DAIT_temp_give_tech

		while = {
			count = AIO_DAIT_temp_give_tech
			AIO_add_random_tech = yes
		}

		clear_variable = AIO_DAIT_temp_give_tech
	}

	after = { country_event = { id = AIO_DAIT_events.202 months = 3 } }
}


# 摧毁舰船回复装甲/护盾
# This = owner of ship 1 (combatant)
# From = owner of ship 2 (destroyed)
# FromFrom = ship 1
# FromFromFrom = ship 2
country_event = {
	id = AIO_DAIT_events.203
	hide_window = yes

	is_triggered_only = yes
	trigger = {
		OR = {
			check_variable = { which = AIO_DAIT_material_option_6_effect_1_value value > 0 }
			check_variable = { which = AIO_DAIT_energy_option_6_effect_1_value value > 0 }
		}
	}

	immediate = {
		fleet = {
			random_controlled_ship = {
				if = {
					limit = { check_variable = { which = AIO_DAIT_material_option_6_effect_1_value value > 0 } }
					repair_armor_amount = owner.AIO_DAIT_material_option_6_effect_1_value
				}
				else_if = {
					limit = { check_variable = { which = AIO_DAIT_energy_option_6_effect_1_value value > 0 } }
					repair_shield_amount = owner.AIO_DAIT_energy_option_6_effect_1_value
				}
			}
		}
	}
}
