namespace = AIO_interference_events

# Fired when a Reality Interferometer finishes construction. Policy controls its effects.
country_event = {
	id = AIO_interference_events.201
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		create_message = {
			type = MEGASTRUCTURE_BUILT
			localization = AIO_interferometer_completed
			days = 30
			target = from
		}
	}
}

# Reserved entry point for future interference incidents. Incident identity should
# be saved when an incident is scheduled so later policy changes cannot reroll it.
country_event = {
	id = AIO_interference_events.301
	hide_window = yes
	is_triggered_only = yes
	trigger = { exists = event_target:AIO_interference_target }
}
