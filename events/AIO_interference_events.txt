#region effect

situation_event = {
	id = AIO_interference_events.310
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		AIO_interference_accidence_balanced_stage_1 = yes
		AIO_interference_dispatch_notice = { EVENT = AIO_interference_events.410 }
	}

situation_event = {
	id = AIO_interference_events.311
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		if = {
