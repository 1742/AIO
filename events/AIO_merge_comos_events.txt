    title = situation_AIO_merge_cosmos
    desc = AIO_merge_cosmos_events.201.desc
    picture = GFX_evt_situation_the_seal
	show_sound = event_mystic_reveal
	location = FROM

	is_triggered_only = yes
    trigger = { has_country_flag = AIO_built_AIO_mega_01 }

    immediate = {
        hidden_effect = {
            set_variable = { which = AIO_situation_AIT_time value = 0 }
            set_variable = { which = AIO_situation_RAIT_affect_system_num value = 0 }

            switch = {
                trigger = galaxy_size
                tiny = { set_variable = { which = AIO_situation_RAIT_affect_system_num value = 1 } }
                small = { set_variable = { which = AIO_situation_RAIT_affect_system_num value = 3 } }
                medium = { set_variable = { which = AIO_situation_RAIT_affect_system_num value = 5 } }
                large = { set_variable = { which = AIO_situation_RAIT_affect_system_num value = 10 } }
                huge = { set_variable = { which = AIO_situation_RAIT_affect_system_num value = 20 } }
            }

            every_country = {
                limit = {
                    is_country_type = default
                    NOT = { has_ascension_perk = ap_cosmogenesis }
                }

                add_modifier = {
                    modifier = AIO_merge_cosmos_modifier
                    mult = 0.25
                }
            }
        }
    }

    option = {
        name = AIO_merge_cosmos_events.201.option_1
        custom_tooltip = AIO_merge_cosmos_events.201.option_1.tt
    }

	after = {
		hidden_effect = {
			# 开启 DAIT 季度事件，直到完成飞升或灭绝
			AIO_DAIT_init_values = yes
		}
	}
}

# 融合宇宙局势月度触发事件
situation_event = {
    id = AIO_merge_cosmos_events.301
    hide_window = yes

    is_triggered_only = yes
    trigger = { is_situation_type = situation_AIO_merge_cosmos }

	immediate = {
		# 检查资源够不够不然换方案
		owner = { AIO_auto_merge_cosmos_effect = yes }

		# 随机应用无限
		if = {
			limit = {
				situation_monthly_progress > 0
				NOT = { has_situation_flag = AIO_merge_cosmos_RAIT }
				owner = {
					NOR = {
						has_global_flag = AIO_RAIT_all
						has_country_flag = AIO_built_AIO_mega_04
					}
				}
			}
			random_list = {
				2 = {  }
				1 = {
					modifier = {
						factor = 0.5
						owner = { has_country_flag = AIO_built_AIO_mega_02 }
					}
					modifier = {
						factor = 0.5
						owner = { has_country_flag = AIO_built_AIO_mega_03 }
					}
					owner = {
						AIO_RAIT_random_trigger = yes
						change_variable = { which = AIO_situation_AIT_time value = 1 }
					}
					set_timed_situation_flag = { flag = AIO_merge_cosmos_RAIT years = 2 }
				}
			}

			# 河系影响
			owner = { country_event = { id = AIO_merge_cosmos_events.401 } }
		}
	}

    after = {
		owner = { AIO_add_random_tech = yes }
		if = {
			limit = { AIO_RAIT_all = yes }
			owner = {
				AIO_RAIT_claer_flags = yes
				set_global_flag = AIO_RAIT_all
			}
		}
    }
}


#region 河系影响

country_event = {
	id = AIO_merge_cosmos_events.401
	hide_window = yes

	is_triggered_only = yes
	trigger = {
		any_situation = {
			is_situation_type = situation_AIO_merge_cosmos
			NOT = { has_situation_flag = AIO_merge_cosmos_RAIT }
		}
	}

	immediate = {
		# 随机影响星系
		no_scope = {
			while = {
				count = root.AIO_situation_RAIT_affect_system_num
				random_system = {
					limit = {
						any_system_planet = { NOT = { is_planet_class = pc_AIO_empty_space } }
						NOR = {
							is_star_class = sc_AIO_void

							has_star_flag = AIO_HS_system
							has_star_flag = AIO_TUG_system

							has_star_flag = synth_queen_capital
						}
					}
					weights = {
						base = 1
						modifier = {
							factor = 0.01
							is_capital_system = yes
						}
						modifier = {
							factor = 0
							exists = owner
							owner = {
								OR = {
									has_country_flag = AIO_built_AIO_mega_02
									has_country_flag = AIO_built_AIO_mega_03
									has_country_flag = AIO_built_AIO_mega_04
								}
							}
						}
					}

					random_list = {
						1 = { # 随机丰富星系矿藏
							save_event_target_as = AIO_situation_RAIT_target
							AIO_RE_enrich_deposits_effect = { TIME = 2 }
							if = {
								limit = { exists = owner }
								owner = { country_event = { id = AIO_merge_cosmos_events.402 } }
							}
						}
						1 = { # 随机改变天体类型
							random_system_planet = {
								limit = {
                                    AIO_safe_remove_planet_exclude_API = no

									is_artificial = no
									is_star = no
									NOT = { is_planet_class = pc_ark }
								}
								AIO_safe_change_random_pc = yes
								save_event_target_as = AIO_situation_RAIT_target
								if = {
									limit = { exists = controller }
									controller = { country_event = { id = AIO_merge_cosmos_events.403 } }
								}
							}
						}
						1 = { # 随机退化殖民地
							modifier = {
								factor = 0
								any_system_planet = { is_colony = no }
							}
							random_system_planet = {
								limit = { is_colony = yes }
								weights = {
									base = 1
									modifier = {
										factor = 0.01
										is_capital = yes
									}
								}
								AIO_RE_degenerate_effect = yes
								save_event_target_as = AIO_situation_RAIT_target
								if = {
									limit = { exists = owner }
									owner = { country_event = { id = AIO_merge_cosmos_events.404 } }
								}
							}
						}
						1 = { # 随机倒流殖民地
							modifier = {
								factor = 0
								any_system_planet = { is_colony = no }
							}
							random_system_planet = {
								limit = { is_colony = yes }
								weights = {
									base = 1
									modifier = {
										factor = 0.01
										is_capital = yes
									}
								}
								AIO_RE_turn_back_time_effect = yes
								save_event_target_as = AIO_situation_RAIT_target
								if = {
									limit = { exists = owner }
									owner = { country_event = { id = AIO_merge_cosmos_events.405 } }
								}
							}
						}
						1 = { # 随机抹除天体
							random_system_planet = {
								limit = {
                                    AIO_safe_remove_planet_exclude_API = no
                                    NOT = { is_planet_class = pc_AIO_empty_space }
                                }
								save_event_target_as = AIO_situation_RAIT_target
								AIO_safe_remove_planet = yes
								if = {
									limit = { exists = controller }
									controller = { country_event = { id = AIO_merge_cosmos_events.406 } }
								}
							}
						}
						1 = { # 抹除星系
                            modifier = {
                                factor = 0
                                AIO_safe_remove_system_exclude_API = yes
                            }
							save_event_target_as = AIO_situation_RAIT_target
							AIO_RE_erase_system_effect = yes
							if = {
								limit = { exists = owner }
								owner = { country_event = { id = AIO_merge_cosmos_events.407 } }
							}
						}
					}
				}
			}
		}
	}
}
