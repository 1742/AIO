AIO_chips = {
	category = AIO_chips
	purge = purge_AIO_chips
	is_capped_by_modifier = no

	swappable_data = {
		default = {
			building_icon = building_lathe_capital
			icon = AIO_chips
		}
	}

	tags = { research }

	possible_pre_triggers = {
		has_owner = yes
		is_enslaved = no
		is_being_purged = no
		is_being_assimilated = no
		is_sapient = yes
	}

	possible_precalc = can_fill_drone_job

	possible = {
		can_think = yes
		has_virtual_species_trait = no
		colony? = {
			OR = {
				has_district = district_AIO_lathe
				has_district = district_AIO_lathe_frame
			}
		}
	}

	resources = {
		category = planet_AIO_chips
		produces = {
			trigger = {
				owner = {
					NOR = {
						any_situation = {
							is_situation_type = situation_energy_deficit
						}
						any_situation = {
							is_situation_type = situation_nomad_economy
							situation_progress <= 0
						}
					}
				}
			}
			physics_research = 0.375
			society_research = 0.375
			engineering_research = 0.375
			AIO_advanced_logic = 0.025
			mult = value:AIO_chips_produce_factor
		}
		upkeep = {
			trigger = {
				owner = {
					NOR = {
						any_situation = {
							is_situation_type = situation_energy_deficit
						}
						any_situation = {
							is_situation_type = situation_nomad_economy
							situation_progress <= 0
						}
					}
				}
			}
			energy = 4.5
			mult = value:AIO_chips_upkeep_factor
		}
	}

	weight = {
		weight = @maintenance_drone_job
		modifier = {
			factor = 1000
			has_trait = trait_AIO_recursive_chassis
			colony? = { NOT = { free_jobs_of_type = { job = AIO_dormant_chassis } } }
		}
		modifier = {
			factor = @maintenance_drone_job
			colony? = { free_jobs > 0 }
		}
	}

	inline_script = "jobs/automodding_priority_research"
}
