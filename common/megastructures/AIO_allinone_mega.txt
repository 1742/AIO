# 现实中的万门之门
AIO_mega_01_fact = {
    entity = "AIO_mega_1_entity"
	construction_entity = "AIO_mega_1_entity"
	scales_with_planet = no

	portrait = "GFX_megastructure_dyson_sphere_background"

	potential = {
		has_menace_perk = menp_AIO_mega_02
		NOR = {
			has_country_flag = AIO_built_AIO_mega_01
			has_country_flag = AIO_built_AIO_mega_02
			has_country_flag = AIO_built_AIO_mega_03
			has_country_flag = AIO_built_AIO_mega_04
		}
	}

	possible = {
		custom_tooltip = {
			fail_text = AIO_mega_01.fail.nomadic
			from = {
				OR = {
					is_nomadic = no
					any_situation = {
						is_situation_type = situation_nomad_economy
						situation_progress >= @mega_01_energy_cost
					}
				}
			}
		}
	}

	dismantle_potential = { always = no }

	upgrade_from = { AIO_horizon_engine }

	build_time = 1000

	resources = {
		category = megastructures
		cost = {
			energy = @mega_01_energy_cost
		}
		cost = {
			trigger = { country_uses_bio_ships = no }
			alloys = @mega_01_alloys_cost
		}
		cost = {
			trigger = { country_uses_bio_ships = yes }
			alloys = @mega_01_alloys_cost
			mult = 0.5
		}
		cost = {
			trigger = { country_uses_bio_ships = yes }
			food = @mega_01_alloys_cost
			mult = @halved_alloy_to_food_cost_ratio
		}
	}

	# bypass类型
	bypass_type = bypass_AIO_ultimate_gate

	on_build_complete = {
		AIO_create_gate_system_effect = yes

		from = {
			set_country_flag = AIO_built_AIO_mega_01
			AIO_DAIT_ensure_menu = yes
			country_event = { id = AIO_crisis_events.403 }
		}
	}
}

# 无名之雾中的万门之门
AIO_mega_01 = {
    entity = "AIO_mega_1_entity"
	construction_entity = "AIO_mega_1_entity"
	scales_with_planet = no

	portrait = "GFX_megastructure_dyson_sphere_background"

	potential = { always = no }

	dismantle_potential = { always = no }

	build_time = 1

	resources = {
		category = megastructures
		produces = { AIO_advanced_logic = 1000 }

		upkeep = {
			energy = 200
			physics_research = 50
			society_research = 50
			engineering_research = 50
		}
	}

	country_modifier = {
		fleet_mia_time_mult = -0.5
		ship_windup_mult = -0.2
	}

	# bypass类型
	bypass_type = bypass_AIO_ultimate_gate

	on_build_complete = {
		fromfrom = { save_global_event_target_as = AIO_DAIT_source_megastructure }
		# 加特效
		create_ambient_object = {
			type = "AIO_crisis_sphere_effect"
			play_animation_once = no
			location = star
			# 我也不到为啥要上移50才对的准中心，应该是虚空星系原因
			entity_offset_height = 50
			scale = 4

			effect = {
				set_location = {
					target = prev.star
					distance = 0
					angle = random
				}
				save_global_event_target_as = AIO_mist_system_effect
			}
		}
		create_ambient_object = {
			type = "star_explosion"
			play_animation_once = yes
			location = star

			effect = {
				set_location = {
					target = prev
					distance = 0
					angle = random
				}
			}
		}
	}
}
