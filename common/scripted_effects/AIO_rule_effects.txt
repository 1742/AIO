# Country initialization: continuous modifiers only, no rewards or replayed events.
AIO_rules_sync_country_effect = {
    if = {
        limit = {
            OR = {
                is_country_type = default
                is_country_type = fallen_empire
                is_country_type = awakened_fallen_empire
            }
        }
        if = {
            limit = {
                has_global_flag = AIO_rule_energy_active
                NOT = { has_modifier = AIO_rule_energy }
            }
            add_modifier = { modifier = AIO_rule_energy days = -1 }
        }
        if = {
            limit = {
                has_global_flag = AIO_rule_matter_active
                NOT = { has_modifier = AIO_rule_matter }
            }
            add_modifier = { modifier = AIO_rule_matter days = -1 }
        }
        if = {
