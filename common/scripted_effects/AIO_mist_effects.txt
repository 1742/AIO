# Colony. Create a normal arkship in the habitat's system, without changing its carrier.
AIO_mist_build_arkship_effect = {
    owner = { save_event_target_as = AIO_mist_arkship_owner }
    solar_system = { save_event_target_as = AIO_mist_arkship_system }
    create_arkship_effect = {
        TYPE = $TYPE$
        OWNER = event_target:AIO_mist_arkship_owner
        LOCATION = event_target:AIO_mist_arkship_system
    }
    # Match the completion bookkeeping in vanilla megastructures/arkship.
    owner = {
        if = {
            limit = {
                has_origin = origin_synthetic_fertility
                NOT = { has_country_flag = synthetics_traditions_unlocked }
            }
            every_owned_colony = {
                limit = {
                    is_planet_class = pc_ark
                    fleet = { is_same_value = last_created_fleet }
                }
                colony_event = { id = machine_age.3130 }
            }
        }
        set_country_flag = arkship_built_$TYPE$
        country_event = { id = timeline.100 }
    }
}

# Country. The live mist has exactly one shared colony target.
AIO_mist_create_colony_effect = {
    if = {
        limit = {
            exists = event_target:AIO_mist_system
            NOT = { exists = event_target:AIO_mist_colony }
        }
        save_event_target_as = AIO_mist_owner
        if = {
            limit = { is_nomadic = yes }
            create_fleet = {
                name = NAME_AIO_Mist_Anchor
                effect = {
                    set_owner = event_target:AIO_mist_owner
                    create_ship = {
                        name = NAME_AIO_Mist_Anchor
                        design = NAME_AIO_Mist_Anchor
                        graphical_culture = civilian_arkship
                        effect = {
                            save_global_event_target_as = AIO_mist_ship
                            colony = { AIO_mist_initialize_colony_effect = yes }
                        }
                    }
                    set_location = event_target:AIO_mist_system.star
                }
            }
        }
        else = {
            event_target:AIO_mist_system = {
                spawn_planet = {
                    class = pc_AIO_mist
                    size = 12
                    orbit_distance = 0
                    init_effect = {
                        set_name = NAME_AIO_Mist_Colony
                        set_surveyed = {
                            surveyor = event_target:AIO_mist_owner
                            surveyed = yes
                        }
                        create_colony = { owner = event_target:AIO_mist_owner }
                        colony = { AIO_mist_initialize_colony_effect = yes }
                    }
                }
            }
        }
        set_visited = event_target:AIO_mist_system
    }
}

# Colony. No migration, ascension, or reserve gates; virtual filling uses vanilla code.
AIO_mist_initialize_colony_effect = {
    save_global_event_target_as = AIO_mist_colony
    set_carrier_flag = AIO_mist_colony
    set_planet_size = 12
    owner = { AIO_rules_sync_country_effect = yes }
    if = {
        limit = { NOT = { has_deposit = d_AIO_mist_exemption } }
        add_deposit = d_AIO_mist_exemption
    }
}
