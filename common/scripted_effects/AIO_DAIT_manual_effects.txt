# Button THIS = selected object, FROM = player. Snapshot only local event targets.
AIO_DAIT_capture_target = {
    from = { save_event_target_as = AIO_DAIT_maker }
    if = {
        limit = { is_scope_type = galactic_object }
        save_event_target_as = AIO_DAIT_target_system
    }
    else_if = {
        limit = { is_scope_type = planet }
        solar_system = { save_event_target_as = AIO_DAIT_target_system }
    }
}

# country; range 0/1/2, 3 = erase five jumps, 4 = strike witnessed realms.
AIO_DAIT_cycle_range = {
    if = {
        limit = { NOT = { is_variable_set = AIO_DAIT_range } }
        set_variable = { which = AIO_DAIT_range value = 0 }
    }
    change_variable = { which = AIO_DAIT_range value = 1 }
    if = {
        limit = {
            OR = {
                check_variable = { which = AIO_DAIT_range value > 4 }
                AND = {
                    NOT = { is_country_type = AIO_empire }
                    check_variable = { which = AIO_DAIT_range value > 2 }
                }
            }
        }
        set_variable = { which = AIO_DAIT_range value = 0 }
    }
}

# system; notify current occupants, at most once per cast phase.
AIO_DAIT_notify_system = {
    owner? = { AIO_DAIT_notify_country = { TEXT = $TEXT$ } }
    every_system_planet_colony = { owner = { AIO_DAIT_notify_country = { TEXT = $TEXT$ } } }
    every_system_ship_colony = { owner = { AIO_DAIT_notify_country = { TEXT = $TEXT$ } } }
    every_fleet_in_system = { owner? = { AIO_DAIT_notify_country = { TEXT = $TEXT$ } } }
}

AIO_DAIT_notify_country = {
    if = {
        limit = {
            NOT = { has_country_flag = AIO_DAIT_notified }
            NOT = { is_same_value = event_target:AIO_DAIT_maker }
        }
        set_country_flag = AIO_DAIT_notified
        create_message = {
            type = MESSAGE_DYSON_SHOT
            localization = $TEXT$
            days = 30
            target = event_target:AIO_DAIT_target_system
        }
    }
}

AIO_DAIT_clear_notifications = {
    every_country = {
        limit = { has_country_flag = AIO_DAIT_notified }
        remove_country_flag = AIO_DAIT_notified
    }
}

AIO_DAIT_start_enrich = {
    if = {
        limit = {
            AIO_DAIT_enrich_ready = yes
            event_target:AIO_DAIT_target_system = {
                AIO_DAIT_enrich_target_valid = yes
                NOT = { has_star_flag = AIO_DAIT_center_busy }
            }
        }
        save_event_target_as = AIO_DAIT_maker
        set_variable = { which = AIO_DAIT_enrich_paid value = value:AIO_DAIT_enrich_price }
        if = {
            limit = { is_country_type = AIO_empire }
            set_variable = { which = AIO_DAIT_enrich_paid value = 0 }
        }
        add_resource = { AIO_advanced_logic = -1 mult = this.AIO_DAIT_enrich_paid }
        set_country_flag = AIO_DAIT_enrich_pending
        set_variable = { which = AIO_DAIT_enrich_range value = 0 }
        if = {
            limit = { is_variable_set = AIO_DAIT_range }
            set_variable = { which = AIO_DAIT_enrich_range value = AIO_DAIT_range }
        }
        event_target:AIO_DAIT_target_system = {
            set_star_flag = AIO_DAIT_center_busy
            set_star_flag = AIO_DAIT_enrich_center_@event_target:AIO_DAIT_maker
        }
        every_system = {
            limit = {
                OR = {
                    is_same_value = event_target:AIO_DAIT_target_system
                    AND = {
                        event_target:AIO_DAIT_maker = { check_variable = { which = AIO_DAIT_enrich_range value = 1 } }
                        distance = { source = event_target:AIO_DAIT_target_system max_jumps = 1 use_bypasses = no }
                    }
                    AND = {
                        event_target:AIO_DAIT_maker = { check_variable = { which = AIO_DAIT_enrich_range value = 2 } }
                        distance = { source = event_target:AIO_DAIT_target_system max_jumps = 2 use_bypasses = no }
                    }
                    AND = {
                        event_target:AIO_DAIT_maker = { check_variable = { which = AIO_DAIT_enrich_range value = 3 } }
                        distance = { source = event_target:AIO_DAIT_target_system max_jumps = 5 use_bypasses = no }
                    }
                }
                AIO_DAIT_system_valid = yes
            }
            set_star_flag = AIO_DAIT_enrich_candidate_@event_target:AIO_DAIT_maker
        }
        AIO_DAIT_finish_enrich = yes
    }
}

AIO_DAIT_finish_enrich = {
    hidden_effect = {
        save_event_target_as = AIO_DAIT_maker
        set_variable = { which = AIO_DAIT_enrich_affected value = 0 }
    }
    if = {
        limit = {
            AIO_DAIT_enrich_unlocked = yes
            exists = event_target:AIO_DAIT_target_system
            event_target:AIO_DAIT_target_system = {
                AIO_DAIT_system_valid = yes
                has_star_flag = AIO_DAIT_enrich_center_@event_target:AIO_DAIT_maker
            }
        }
        every_system = {
            limit = {
                has_star_flag = AIO_DAIT_enrich_candidate_@event_target:AIO_DAIT_maker
                AIO_DAIT_enrich_target_valid = yes
            }
            hidden_effect = { AIO_DAIT_notify_system = { TEXT = AIO_DAIT_enrich_completed } }
            event_target:AIO_DAIT_maker = { hidden_effect = { change_variable = { which = AIO_DAIT_enrich_affected value = 1 } } }
            AIO_DAIT_enrich_effect = { TIMES = 10 }
        }
    }
    if = {
        limit = { check_variable = { which = AIO_DAIT_enrich_affected value = 0 } }
        add_resource = { AIO_advanced_logic = 0.5 mult = this.AIO_DAIT_enrich_paid }
        hidden_effect = {
            create_message = {
                type = MESSAGE_DYSON_SHOT
                localization = AIO_DAIT_cancelled
                days = 30
            }
        }
    }
    if = {
        limit = { check_variable = { which = AIO_DAIT_enrich_affected value > 0 } }
        hidden_effect = {
            create_message = {
                type = MESSAGE_DYSON_SHOT
                localization = AIO_DAIT_enrich_completed
                days = 30
                target = event_target:AIO_DAIT_target_system
            }
        }
    }
    hidden_effect = {
        AIO_DAIT_cleanup_enrich = yes
        AIO_DAIT_clear_notifications = yes
    }
}

AIO_DAIT_cleanup_enrich = {
    save_event_target_as = AIO_DAIT_maker

AIO_DAIT_start_storm = {
    if = {
        limit = {
            AIO_DAIT_storm_ready = yes
            event_target:AIO_DAIT_target_system = {
                AIO_DAIT_storm_target_valid = yes
                NOT = { has_star_flag = AIO_DAIT_center_busy }
            }
        }
        save_event_target_as = AIO_DAIT_maker
        set_variable = { which = AIO_DAIT_storm_paid value = value:AIO_DAIT_storm_price }
        if = {
            limit = { is_country_type = AIO_empire }
            set_variable = { which = AIO_DAIT_storm_paid value = 0 }
        }
        add_resource = { AIO_advanced_logic = -1 mult = this.AIO_DAIT_storm_paid }
        set_country_flag = AIO_DAIT_storm_pending
        set_variable = { which = AIO_DAIT_storm_range value = 0 }
        if = {
            limit = { is_variable_set = AIO_DAIT_range }
            set_variable = { which = AIO_DAIT_storm_range value = AIO_DAIT_range }
        }
        set_variable = { which = AIO_DAIT_storm_delay value = value:AIO_DAIT_storm_windup }
        set_saved_date = {
            key = AIO_DAIT_storm_due
            days_from_present = AIO_DAIT_storm_delay
        }
        event_target:AIO_DAIT_target_system = {
            set_star_flag = AIO_DAIT_center_busy
            set_star_flag = AIO_DAIT_storm_center_@event_target:AIO_DAIT_maker
        }
        event_target:AIO_DAIT_target_system = {
            create_ambient_object = {
                type = "crisis_sphere_1"
                location = star
                effect = {
                    set_ambient_object_flag = AIO_DAIT_storm_warning_@event_target:AIO_DAIT_maker
                    set_location = { target = prev.star distance = 0 angle = 0 }
                }
            }
        }
        every_system = {
            limit = {
                OR = {
                    is_same_value = event_target:AIO_DAIT_target_system
                    AND = {
                        event_target:AIO_DAIT_maker = { check_variable = { which = AIO_DAIT_storm_range value = 1 } }
                        distance = { source = event_target:AIO_DAIT_target_system max_jumps = 1 use_bypasses = no }
                    }
                    AND = {
                        event_target:AIO_DAIT_maker = { check_variable = { which = AIO_DAIT_storm_range value = 2 } }
                        distance = { source = event_target:AIO_DAIT_target_system max_jumps = 2 use_bypasses = no }
                    }
                    AND = {
                        event_target:AIO_DAIT_maker = { check_variable = { which = AIO_DAIT_storm_range value = 3 } }
                        distance = { source = event_target:AIO_DAIT_target_system max_jumps = 5 use_bypasses = no }
                    }
                }
                AIO_DAIT_system_valid = yes
            }
            set_star_flag = AIO_DAIT_storm_candidate_@event_target:AIO_DAIT_maker
        }
        start_situation = { type = AIO_DAIT_storm_situation target = event_target:AIO_DAIT_target_system }
    }
}

                OR = { is_same_value = event_target:AIO_DAIT_target_system
                    AND = {
                        event_target:AIO_DAIT_maker = { check_variable = { which = AIO_DAIT_erase_range value = 1 } }
                        distance = { source = event_target:AIO_DAIT_target_system max_jumps = 1 use_bypasses = no }
                    }
                    AND = {
                        event_target:AIO_DAIT_maker = { check_variable = { which = AIO_DAIT_erase_range value = 2 } }
                        distance = { source = event_target:AIO_DAIT_target_system max_jumps = 2 use_bypasses = no }
                    }
                    AND = {
                        event_target:AIO_DAIT_maker = { check_variable = { which = AIO_DAIT_erase_range value = 3 } }
                        distance = { source = event_target:AIO_DAIT_target_system max_jumps = 5 use_bypasses = no }
                    }
                }
                AIO_DAIT_system_valid = yes
            }
            set_star_flag = AIO_DAIT_erase_candidate_@event_target:AIO_DAIT_maker
        }
        start_situation = { type = AIO_DAIT_erase_situation target = event_target:AIO_DAIT_target_system }
    }
}

AIO_DAIT_finish_erase = {
    hidden_effect = {
        save_event_target_as = AIO_DAIT_maker
        set_variable = { which = AIO_DAIT_erase_affected value = 0 }
    }
    if = {
        limit = {
            AIO_DAIT_erase_unlocked = yes
            exists = event_target:AIO_DAIT_target_system
            event_target:AIO_DAIT_target_system = {
                AIO_DAIT_system_valid = yes
                has_star_flag = AIO_DAIT_erase_center_@event_target:AIO_DAIT_maker
            }
        }
        every_system = {
            limit = {
                has_star_flag = AIO_DAIT_erase_candidate_@event_target:AIO_DAIT_maker
                AIO_DAIT_erase_target_valid = yes
            }
            hidden_effect = { AIO_DAIT_notify_system = { TEXT = AIO_DAIT_erase_completed } }
            event_target:AIO_DAIT_maker = { hidden_effect = { change_variable = { which = AIO_DAIT_erase_affected value = 1 } } }
            AIO_DAIT_erase_system_effect = yes
            random_system_planet = {
                limit = { is_star = yes }
                create_ambient_object = {
                    type = "star_explosion"
                    play_animation_once = yes
                    location = this
                    effect = {
                        set_location = {
                            target = prev
                            distance = 0
                            angle = random
                        }
                    }
                }
            }
        }
    }
    if = { limit = { check_variable = { which = AIO_DAIT_erase_affected value = 0 } }
        add_resource = { AIO_advanced_logic = 0.5 mult = this.AIO_DAIT_erase_paid }
        hidden_effect = {
            create_message = {
                type = MESSAGE_DYSON_SHOT
                localization = AIO_DAIT_cancelled
                days = 30
            }
        }
    }
    else = {
        hidden_effect = {
            set_timed_country_flag = { flag = AIO_DAIT_erase_cooldown days = 360 }
            set_saved_date = {
                key = AIO_DAIT_erase_ready_date
                days_from_present = 360
                expires = 360
            }
        }
    }
    if = {
        limit = { check_variable = { which = AIO_DAIT_erase_affected value > 0 } }
        hidden_effect = {
            create_message = {
                type = MESSAGE_DYSON_SHOT
                localization = AIO_DAIT_erase_completed
                days = 30
                target = event_target:AIO_DAIT_target_system
            }
        }
    }
    hidden_effect = {
        AIO_DAIT_cleanup_erase = yes
        AIO_DAIT_clear_notifications = yes
    }
}

AIO_DAIT_cleanup_erase = {
    save_event_target_as = AIO_DAIT_maker
    every_system = {
        limit = { has_star_flag = AIO_DAIT_erase_candidate_@event_target:AIO_DAIT_maker }
        remove_star_flag = AIO_DAIT_erase_candidate_@event_target:AIO_DAIT_maker
    }
    every_system = {
        limit = { has_star_flag = AIO_DAIT_erase_center_@event_target:AIO_DAIT_maker }
        remove_star_flag = AIO_DAIT_erase_center_@event_target:AIO_DAIT_maker
        every_system_ambient_object = {
            limit = { has_ambient_object_flag = AIO_DAIT_erase_warning_@event_target:AIO_DAIT_maker }
            destroy_ambient_object = this
        }
        remove_star_flag = AIO_DAIT_center_busy
    }
    remove_country_flag = AIO_DAIT_erase_due
    remove_country_flag = AIO_DAIT_erase_pending
    clear_variable = AIO_DAIT_erase_paid
    clear_variable = AIO_DAIT_erase_range
    clear_variable = AIO_DAIT_erase_delay
    clear_variable = AIO_DAIT_erase_affected
}
