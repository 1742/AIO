# scope: system
AIO_create_gate_system_effect = {
    if = {
        limit = { any_system_megastructure = { is_megastructure_type = AIO_mega_01_fact } }

        # 清除视界引擎特效
        if = {
            limit = { exists = event_target:AIO_horizon_engine_effect }
            event_target:AIO_horizon_engine_effect = { destroy_ambient_object = this }
            clear_global_event_target = AIO_horizon_engine_effect
        }

        # 加特效
        random_system_megastructure = {
            limit = { is_megastructure_type = AIO_mega_01_fact }
            planet = {
                create_ambient_object = {
                    type = "star_explosion"
                    play_animation_once = yes
                    location = this

                    effect = {
                        set_location = {
                            target = prev
                            distance = 0
                            angle = random
                        }
                    }
                }
                create_ambient_object = {
                    type = "crisis_sphere_1"
                    play_animation_once = no
                    location = this

                    effect = {
                        set_location = {
                            target = prev
                            distance = 0
                            angle = random
                        }
                        save_global_event_target_as = AIO_gate_system_effect
                    }
                }
            }
        }

        # 清除星系内天体
        every_system_planet = { inline_script = AIO/AIO_safe_remove_planet }
        inline_script = AIO/AIO_safe_remove_system_megastructures
        every_fleet_in_system = {
            limit = {
                AIO_safe_remove_fleet_exclude_API = no
                NOT = { is_ship_class = shipclass_starbase }
            }
            destroy_fleet = this
        }

        # 改变星系类型
        set_star_class = sc_black_hole
        # 改名
        set_name = AIO_gate
        # 标志位
        set_star_flag = AIO_gate_system
        save_global_event_target_as = AIO_gate_system
    }
}

                        min_orientation_angle = 100
                        max_orientation_angle = 170

                        initializer = AIO_mist_initializer

                        hyperlane = no
                    }
                }
                medium = {
                    spawn_system = {
                        # 设置银心坐标，是极坐标形式
                        # 距离银心距离
                        min_distance = 600
                        max_distance = 650
                        # 角度，逆时针
                        min_orientation_angle = 100
                        max_orientation_angle = 170

                        initializer = AIO_mist_initializer

                        hyperlane = no
                    }
                }
                large = {
                    spawn_system = {
                        # 设置银心坐标，是极坐标形式
                        # 距离银心距离
                        min_distance = 550
                        max_distance = 600
                        # 角度，逆时针
                        min_orientation_angle = 100
                        max_orientation_angle = 170

                        initializer = AIO_mist_initializer

                        hyperlane = no
                    }
                }
                huge = {
                    spawn_system = {
                        # 设置银心坐标，是极坐标形式
                        # 距离银心距离
                        min_distance = 550
                        max_distance = 600
                        # 角度，逆时针
                        min_orientation_angle = 100
                        max_orientation_angle = 170

                        initializer = AIO_mist_initializer

                        hyperlane = no

                }

                    type = AIO_mega_01
                    planet = star
                    orbit_angle = 0
                    orbit_distance = 0
                    owner = root
                }
            }
            set_name = AIO_mist
        }
    }
}

# scope: any
AIO_ascension_galaxy_effect = {
    # 河系影响
    no_scope = {
        every_system = {
            limit = {
                NOR = {
                    is_star_class = sc_AIO_void
                    has_star_flag = AIO_mist_system
                    has_star_flag = AIO_gate_system
                    AIO_is_crisis_mega_system = yes
                    has_star_flag = synth_queen_capital
                }
            }
            while = {
                count = 2
                random_list = {
                    2 = {}
                    3 = {
                        # 随机改变天体类型
                        random_system_planet = {
                            limit = {
                                AIO_safe_remove_planet_exclude_API = no
                                is_artificial = no
                            }
                            AIO_safe_change_random_pc = yes
                        }
                    }
                    2 = {
                        # 随机抹除天体
                        random_system_planet = {
                            limit = {
                                AIO_safe_remove_planet_exclude_API = no
                                NOT = { is_planet_class = pc_AIO_empty_space }
                            }
                            inline_script = AIO/AIO_safe_remove_planet
                        }
                    }
                    1 = {
                        modifier = {
                            factor = 0
                            AIO_safe_remove_system_exclude_API = yes
                        }
                        # 抹除星系
                        AIO_DAIT_erase_system_effect = yes
                    }
                }
            }
        }

        every_country = {
            limit = {
                OR = {
                    is_country_type = default
                    is_country_type = fallen_empire
                    is_country_type = awakened_fallen_empire
                }
            }
            while = {
                count = 2
                random_list = {
                    4 = {}
                    3 = {
                        # 改变政权
                        modifier = {
                            factor = 0
                            NOT = { is_country_type = default }
                        }
                        change_government = {
                            authority = random
                            civics = random
                            remove_invalid_civics = yes
                        }
                    }
                    2 = {
                        # 随机去军事化
                        every_owned_fleet = {
                            limit = { AIO_safe_remove_fleet_exclude_API = no }
                            destroy_fleet = this
                        }
                    }
                    1 = {
                        # 随机抹除国家
                        destroy_country = yes
                    }
                }
            }
        }
    }
}

AIO_ascension_effect = {
    if = {
        limit = {
            exists = event_target:AIO_mist_system
            exists = event_target:AIO_gate_system
        }

        # 转变国家类型
        set_country_type = AIO_empire
        set_ai_personality = AIO_transcendent
        if = {
            limit = { owner_main_species = { is_robotic = yes } }
            change_government = {
                authority = auth_machine_intelligence
                civics = { civic = civic_AIO_dream }
            }
        }
        else = {
            change_government = {
                authority = auth_hive_mind
                civics = { civic = civic_AIO_dream }
            }
        }
        set_name = AIO_empire
        # 杀死领袖
        every_owned_leader = {
            limit = {
                AIO_safe_remove_leader_exclude_API = no
                is_ruler = no
                is_councilor = no
                NOT = { has_leader_flag = gray_leader }
            }
            kill_leader = { show_notification = no fire = yes }
        }
        # 4.5 的首都是殖民地对象：先建立新的空首都，再销毁旧殖民地。
        # 不可在 set_capital 后立即 destroy_colony，否则界面和延迟事件仍引用它。
        event_target:AIO_mist_system = {
            random_system_planet = {
                limit = { is_star = yes }
                set_owner = root
                set_capital = yes
            }
        }
        every_owned_colony = {
            limit = { is_capital = no }
            destroy_colony = yes
        }
        # 移除所有原版巨构
        every_system_within_border = { inline_script = AIO/AIO_safe_remove_system_megastructures }
        # 移除所有舰队及恒星基地
        every_owned_fleet = {
            limit = { AIO_safe_remove_fleet_exclude_API = no }
            destroy_fleet = this
        }
        # 与所有帝国通讯并获得所有情报
        every_playable_country = {
            limit = { NOT = { is_same_value = root } }
            if = {
                limit = { NOT = { has_communications = root } }
                establish_communications_no_message = root
            }
            # 移除扰乱现实好感修正
            if = {
                limit = { has_opinion_modifier = { who = root modifier = opinion_cosmogenesis_thesis } }
                remove_opinion_modifier = { who = root modifier = opinion_cosmogenesis_thesis }
            }
            # 移除己方帝国的宣称好感修正
            if = {
                limit = { has_opinion_modifier = { who = root modifier = opinion_claims_on_us } }
                remove_opinion_modifier = { who = root modifier = opinion_claims_on_us }
            }
            root = { add_intel = { who = prev amount = 100 } }
        }
        # 标志位
        remove_country_flag = AIO_crisis_country
        set_country_flag = AIO_empire_country
        clear_global_event_target = AIO_crisis_country
        save_global_event_target_as = AIO_empire_country
        # 允许进入牢塞星系
        set_country_flag = protected_from_queen_storm
        AIO_DAIT_ensure_menu = yes
    }
}

# scope: any
AIO_destroy_mist_effect = {
    event_target:AIO_mist_system = {
        # 移除特效
        destroy_ambient_object = event_target:AIO_mist_system_effect
        clear_global_event_target = AIO_mist_system_effect
        # 移除巨构
        random_system_megastructure = {
            limit = {
                OR = {
                    is_megastructure_type = AIO_mega_01
                    is_megastructure_type = AIO_mega_02
                    is_megastructure_type = AIO_mega_03
                    is_megastructure_type = AIO_mega_04
                }
            }
            remove_megastructure = this
        }
        inline_script = AIO/AIO_safe_remove_system_megastructures
        # 移除所有舰队和恒星基地
        every_fleet_in_system = {
            limit = { AIO_safe_remove_fleet_exclude_API = no }
            destroy_fleet = this
        }
        set_name = AIO_Abyss
    }
}

# scope: any
AIO_destroy_gate_effect = {
    event_target:AIO_gate_system = {
        random_system_megastructure = {
            limit = { is_megastructure_type = AIO_mega_01_fact }
            remove_megastructure = this
        }
        inline_script = AIO/AIO_safe_remove_system_megastructures
        if = {
            limit = { any_system_ambient_object = { is_same_value = event_target:AIO_gate_system_effect } }
            destroy_ambient_object = event_target:AIO_gate_system_effect
            clear_global_event_target = AIO_gate_system_effect
        }
        random_system_planet = {
            limit = { is_star = yes }
            change_pc = pc_black_hole
        }
        every_fleet_in_system = {
            limit = { AIO_safe_remove_fleet_exclude_API = no }
            destroy_fleet = this
        }
        set_name = AIO_abyss_edge
        remove_star_flag = AIO_gate_system
    }
    clear_global_event_target = AIO_gate_system
}
