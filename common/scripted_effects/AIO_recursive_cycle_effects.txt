# Recursive-cycle situation helpers.

# Country scope.
AIO_recursive_cycle_remove_approach_modifiers_effect = {
	remove_modifier = AIO_recursive_cycle_parallel_modifier
	remove_modifier = AIO_recursive_cycle_overclock_modifier
}

# Situation scope: workforce is snapshotted once per month for next month's
# progress and approach upkeep.
AIO_recursive_cycle_record_month_effect = {
	if = {
		limit = { current_situation_approach = AIO_recursive_cycle_parallel }
		change_variable = { which = AIO_recursive_cycle_parallel_contribution value = 1 }
	}
	else_if = {
		limit = { current_situation_approach = AIO_recursive_cycle_serial }
		change_variable = { which = AIO_recursive_cycle_serial_contribution value = 1 }
	}
	else = {
		change_variable = { which = AIO_recursive_cycle_overclock_contribution value = 1 }
	}
}

# Country scope. Calculates weighted completion rewards, applies stored research,
# and removes only pop groups currently occupying the recursive-node category.
AIO_recursive_cycle_calculate_rewards_effect = {
	AIO_recursive_cycle_remove_approach_modifiers_effect = yes

	set_variable = {
		which = AIO_recursive_cycle_total_contribution
		value = AIO_recursive_cycle_parallel_contribution
	}
	change_variable = {
		which = AIO_recursive_cycle_total_contribution
		value = AIO_recursive_cycle_serial_contribution
	}
	change_variable = {
		which = AIO_recursive_cycle_total_contribution
		value = AIO_recursive_cycle_overclock_contribution
	}

	# Console completion and old saves may not contain contribution bookkeeping.
	if = {
		limit = {
			NOT = {
				check_variable = {
					which = AIO_recursive_cycle_total_contribution
					value > 0
				}
			}
		}
		set_variable = { which = AIO_recursive_cycle_parallel_contribution value = 50000 }
		set_variable = { which = AIO_recursive_cycle_total_contribution value = 50000 }
	}

	# Research months = (parallel*120 + serial*240 + overclock*120) / total.
	set_variable = {
		which = AIO_recursive_cycle_research_months
		value = AIO_recursive_cycle_parallel_contribution
	}
	multiply_variable = { which = AIO_recursive_cycle_research_months value = 120 }

	set_variable = {
		which = AIO_recursive_cycle_temp
		value = AIO_recursive_cycle_serial_contribution
	}
	multiply_variable = { which = AIO_recursive_cycle_temp value = 240 }
	change_variable = {
		which = AIO_recursive_cycle_research_months
		value = AIO_recursive_cycle_temp
	}

	set_variable = {
		which = AIO_recursive_cycle_temp
		value = AIO_recursive_cycle_overclock_contribution
	}
	multiply_variable = { which = AIO_recursive_cycle_temp value = 120 }
	change_variable = {
		which = AIO_recursive_cycle_research_months
		value = AIO_recursive_cycle_temp
	}
	divide_variable = {
		which = AIO_recursive_cycle_research_months
		value = AIO_recursive_cycle_total_contribution
	}

	# Loss ratio = (parallel*5% + serial*10% + overclock*25%) / total.
	set_variable = {
		which = AIO_recursive_cycle_pop_loss
		value = AIO_recursive_cycle_parallel_contribution
	}
	multiply_variable = { which = AIO_recursive_cycle_pop_loss value = 0.05 }

	set_variable = {
		which = AIO_recursive_cycle_temp
		value = AIO_recursive_cycle_serial_contribution
	}
	multiply_variable = { which = AIO_recursive_cycle_temp value = 0.1 }
	change_variable = {
		which = AIO_recursive_cycle_pop_loss
		value = AIO_recursive_cycle_temp
	}

	set_variable = {
		which = AIO_recursive_cycle_temp
		value = AIO_recursive_cycle_overclock_contribution
	}
	multiply_variable = { which = AIO_recursive_cycle_temp value = 0.25 }
	change_variable = {
		which = AIO_recursive_cycle_pop_loss
		value = AIO_recursive_cycle_temp
	}
	divide_variable = {
		which = AIO_recursive_cycle_pop_loss
		value = AIO_recursive_cycle_total_contribution
	}

	if = {
		limit = {
			any_owned_planet = {
				has_active_building = building_AIO_lathe_preserver
			}
		}
		multiply_variable = {
			which = AIO_recursive_cycle_pop_loss
			value = 0.75
		}
	}

	set_variable = {
		which = AIO_recursive_cycle_pop_loss_display
		value = AIO_recursive_cycle_pop_loss
	}
	multiply_variable = {
		which = AIO_recursive_cycle_pop_loss_display
		value = 100
	}

	add_monthly_resource_mult = {
		resource = physics_research
		value = AIO_recursive_cycle_research_months
	}
	add_monthly_resource_mult = {
		resource = society_research
		value = AIO_recursive_cycle_research_months
	}
	add_monthly_resource_mult = {
		resource = engineering_research
		value = AIO_recursive_cycle_research_months
	}

	every_owned_pop_group = {
		limit = {
			is_pop_category = AIO_recursive_node
			is_unemployed = no
		}
		kill_pop_group = {
			pop_group = this
			percentage = root.AIO_recursive_cycle_pop_loss
		}
	}

	clear_variable = AIO_recursive_cycle_temp
}

# Generic country-scope refresh for one ranked permanent modifier.
AIO_recursive_cycle_refresh_ranked_modifier_effect = {
	remove_modifier = $MODIFIER$
	if = {
		limit = {
			check_variable = {
				which = $RANK$
				value > 0
			}
		}
		add_modifier = {
			modifier = $MODIFIER$
			days = -1
			mult = $RANK$
		}
	}
}
