AIO_DAIT_echo_finish_effect = {
    set_timed_star_flag = { flag = AIO_DAIT_echo_cooldown days = 1800 }
    root = { AIO_DAIT_echo_notify_effect = { EVENT = $EVENT$ } }
}


#region notify

# scope: country
AIO_DAIT_echo_mark_country_effect = {
    if = {
        limit = {
            is_country_type = default
            NOT = { has_country_flag = AIO_DAIT_echo_affected_country }
        }
        set_country_flag = AIO_DAIT_echo_affected_country
    }
}

AIO_DAIT_echo_clear_affected_countries_effect = {
    every_country = {
        limit = { has_country_flag = AIO_DAIT_echo_affected_country }
        remove_country_flag = AIO_DAIT_echo_affected_country
    }
}

AIO_DAIT_echo_notify_effect = {
    every_country = {
        limit = { has_country_flag = AIO_DAIT_echo_affected_country }
        country_event = {
            id = AIO_DAIT_events.$EVENT$
            scopes = { from = event_target:AIO_DAIT_echo_system }
        }
        remove_country_flag = AIO_DAIT_echo_affected_country
    }
}

# scope: affected country; from: affected system
AIO_DAIT_echo_create_message_effect = {
    create_message = {
        type = AIO_DAIT_ECHO
        localization = $MESSAGE$
        days = 30
        target = from
        variable = {
            type = name
            localization = SYSTEM
            scope = from
        }
    }
}

# scope: system
AIO_DAIT_echo_mark_system_country_effect = {
    every_fleet_in_system = {
        limit = { NOT = { has_fleet_flag = AIO_Aforgomom_fleet } }
        owner? = { AIO_DAIT_echo_mark_country_effect = yes }
    }
}

AIO_DAIT_echo_mark_erase_owners_effect = {
    every_system = {
        limit = {
            NOR = {
                has_star_flag = AIO_HS_system
                has_star_flag = AIO_TUG_system
                has_star_flag = synth_queen_capital
                AIO_safe_remove_system_exclude_API = yes
            }
            AIO_is_crisis_mega_system = no
            any_system_planet = { NOT = { is_planet_class = pc_AIO_empty_space } }
            OR = {
                is_same_value = prev
                distance = { source = prev max_jumps = $MAX_JUMPS|1$ use_bypasses = no }
            }
        }
        owner? = { AIO_DAIT_echo_mark_country_effect = yes }
        every_system_planet = { owner? = { AIO_DAIT_echo_mark_country_effect = yes } }
        every_fleet_in_system = { owner? = { AIO_DAIT_echo_mark_country_effect = yes } }
    }
}

#endregion
