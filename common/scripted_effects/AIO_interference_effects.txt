#region api

# Scope: system
# OWNER: country that owns the completed Reality Interferometer
AIO_interference_create_source = {
	AIO_interference_prepare_source = { OWNER = $OWNER$ }
	set_star_flag = AIO_interference_source_$TYPE$
	aura_owner = { AIO_interference_rebuild_effect_flags = yes }
}

# Scope: system
AIO_interference_remove_source = {
	remove_star_flag = AIO_interference_source
	remove_star_flag = AIO_interference_source_energy
	remove_star_flag = AIO_interference_source_resource
	remove_star_flag = AIO_interference_source_bio
	remove_star_flag = AIO_interference_source_industry
	remove_star_flag = AIO_interference_source_build
	remove_star_flag = AIO_interference_source_geology
	if = {
		limit = { has_psionic_aura = AIO_reality_interference_aura }
		aura_owner = { AIO_interference_rebuild_effect_flags = yes }
	}
}

# Scope: country
# Called when sources or interference range change. Recalculate effect flags
# and source counts on every system carrying this country's aura.
AIO_interference_rebuild_effect_flags = {
	save_event_target_as = AIO_interference_rebuild_owner
	every_system_with_aura = {
		limit = {
			has_psionic_aura = AIO_reality_interference_aura
			aura_owner = { is_same_value = event_target:AIO_interference_rebuild_owner }
		}
		AIO_interference_update_system_effect_flags = {
			OWNER = event_target:AIO_interference_rebuild_owner
		}
	}
}

# Scope: a system carrying the Reality Interference aura
AIO_interference_prototype_try_spread = {
	if = {
		limit = { has_psionic_aura = AIO_reality_interference_aura }
		aura_owner = { save_event_target_as = AIO_interference_spreading_owner }
		random_neighbor_system = {
			limit = {
				AIO_is_nearby_interference_source = {
					OWNER = event_target:AIO_interference_spreading_owner
				}
				has_psionic_aura = no
			}
			spawn_psionic_aura = {
				owner = event_target:AIO_interference_spreading_owner
				type = AIO_reality_interference_aura
				update_type = no
			}
			set_aura_intensity = 1
			AIO_interference_update_system_effect_flags = {
				OWNER = event_target:AIO_interference_spreading_owner
			}
		}
	}
}

# Scope: system
# OWNER: country that owns the completed Reality Interferometer
# Existing vanilla or foreign auras are allowed to run their on_vanish effect,
# then are replaced by the Reality Interference aura.
AIO_interference_prepare_source = {
	if = {
		limit = {
			NOT = {
				AND = {
					has_psionic_aura = AIO_reality_interference_aura
					aura_owner = { is_same_value = $OWNER$ }
				}
			}
		}
		if = {
			limit = { has_psionic_aura = yes }
			destroy_psionic_aura = yes
		}
		spawn_psionic_aura = {
			owner = $OWNER$
			type = AIO_reality_interference_aura
			update_type = no
		}
		set_aura_intensity = 1
	}
	set_star_flag = AIO_interference_source
}

# Scope: a system carrying an AIO Reality Interference aura
# OWNER: country that owns the aura
# RANGE: effect range(max jumps)
AIO_interference_update_system_effect_flags = {
	if = {
		limit = {
			AIO_is_nearby_specific_interference_source = {
				TYPE = energy
				OWNER = $OWNER$
			}
		}
		set_star_flag = AIO_interference_energy
		remove_star_flag = AIO_interference_energy_residual
	}
	else = {
		if = {
			limit = { has_star_flag = AIO_interference_energy }
			remove_star_flag = AIO_interference_energy
			set_timed_star_flag = {
				flag = AIO_interference_energy_residual
				years = 5
			}
		}
	}

	if = {
		limit = {
			AIO_is_nearby_specific_interference_source = {
				TYPE = resource
				OWNER = $OWNER$
			}
		}
		set_star_flag = AIO_interference_resource
		remove_star_flag = AIO_interference_resource_residual
	}
	else = {
		if = {
			limit = { has_star_flag = AIO_interference_resource }
			remove_star_flag = AIO_interference_resource
			set_timed_star_flag = {
				flag = AIO_interference_resource_residual
				years = 5
			}
		}
	}

	if = {
		limit = {
			AIO_is_nearby_specific_interference_source = {
				TYPE = bio
				OWNER = $OWNER$
			}
		}
		set_star_flag = AIO_interference_bio
		remove_star_flag = AIO_interference_bio_residual
	}
	else = {
		if = {
			limit = { has_star_flag = AIO_interference_bio }
			remove_star_flag = AIO_interference_bio
			set_timed_star_flag = {
				flag = AIO_interference_bio_residual
				years = 5
			}
		}
	}

	if = {
		limit = {
			AIO_is_nearby_specific_interference_source = {
				TYPE = industry
				OWNER = $OWNER$
			}
		}
		set_star_flag = AIO_interference_industry
		remove_star_flag = AIO_interference_industry_residual
	}
	else = {
		if = {
			limit = { has_star_flag = AIO_interference_industry }
			remove_star_flag = AIO_interference_industry
			set_timed_star_flag = {
				flag = AIO_interference_industry_residual
				years = 5
			}
		}
	}

	if = {
		limit = {
			AIO_is_nearby_specific_interference_source = {
				TYPE = build
				OWNER = $OWNER$
			}
		}
		set_star_flag = AIO_interference_build
		remove_star_flag = AIO_interference_build_residual
	}
	else = {
		if = {
			limit = { has_star_flag = AIO_interference_build }
			remove_star_flag = AIO_interference_build
			set_timed_star_flag = {
				flag = AIO_interference_build_residual
				years = 5
			}
		}
	}

	if = {
		limit = {
			AIO_is_nearby_specific_interference_source = {
				TYPE = geology
				OWNER = $OWNER$
			}
		}
		set_star_flag = AIO_interference_geology
		remove_star_flag = AIO_interference_geology_residual
	}
	else = {
		if = {
			limit = { has_star_flag = AIO_interference_geology }
			remove_star_flag = AIO_interference_geology
			set_timed_star_flag = {
				flag = AIO_interference_geology_residual
				years = 5
			}
		}
	}

	AIO_interference_recount_sources = { OWNER = $OWNER$ }
}

# Scope: affected aura system. Count distinct source systems within the aura
# owner's range, including this system. One interferometer is allowed per system.
AIO_interference_recount_sources = {
	set_variable = { which = AIO_interference_energy_sources value = 0 }
	set_variable = { which = AIO_interference_resource_sources value = 0 }
	set_variable = { which = AIO_interference_bio_sources value = 0 }
	set_variable = { which = AIO_interference_industry_sources value = 0 }
	set_variable = { which = AIO_interference_build_sources value = 0 }
	set_variable = { which = AIO_interference_geology_sources value = 0 }
	save_event_target_as = AIO_interference_count_target
	aura_owner = {
		every_system_with_aura = {
			limit = {
				has_psionic_aura = AIO_reality_interference_aura
				has_star_flag = AIO_interference_source
				aura_owner = { is_same_value = $OWNER$ }
				OR = {
					is_same_value = event_target:AIO_interference_count_target
					closest_system = {
						use_bypasses = no
						min_steps = 1
						max_steps = $OWNER$.AIO_interference_range
						valid = {
							has_psionic_aura = AIO_reality_interference_aura
							aura_owner = { is_same_value = $OWNER$ }
						}
						limit = { is_same_value = event_target:AIO_interference_count_target }
					}
				}
			}
			if = {
				limit = { has_star_flag = AIO_interference_source_energy }
				event_target:AIO_interference_count_target = { change_variable = { which = AIO_interference_energy_sources value = 1 } }
			}
			if = {
				limit = { has_star_flag = AIO_interference_source_resource }
				event_target:AIO_interference_count_target = { change_variable = { which = AIO_interference_resource_sources value = 1 } }
			}
			if = {
				limit = { has_star_flag = AIO_interference_source_bio }
				event_target:AIO_interference_count_target = { change_variable = { which = AIO_interference_bio_sources value = 1 } }
			}
			if = {
				limit = { has_star_flag = AIO_interference_source_industry }
				event_target:AIO_interference_count_target = { change_variable = { which = AIO_interference_industry_sources value = 1 } }
			}
			if = {
				limit = { has_star_flag = AIO_interference_source_build }
				event_target:AIO_interference_count_target = { change_variable = { which = AIO_interference_build_sources value = 1 } }
			}
			if = {
				limit = { has_star_flag = AIO_interference_source_geology }
				event_target:AIO_interference_count_target = { change_variable = { which = AIO_interference_geology_sources value = 1 } }
			}
		}
	}
}
