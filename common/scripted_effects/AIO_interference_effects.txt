# Scope: system.
AIO_interference_create_source = {
	AIO_interference_prepare_source = { OWNER = $OWNER$ }
	aura_owner = { AIO_interference_rebuild_effect = yes }
}

# Scope: system.
AIO_interference_remove_source = {
	remove_star_flag = AIO_interference_source
	if = {
		limit = { has_psionic_aura = AIO_reality_interference_aura }
		aura_owner = { AIO_interference_rebuild_effect = yes }
	}
}

# Call when interference source change
# scope: country
AIO_interference_rebuild_effect = {
	save_event_target_as = AIO_interference_rebuild_owner
	every_system_with_aura = {
		limit = {
			has_psionic_aura = AIO_reality_interference_aura
			aura_owner = { is_same_value = event_target:AIO_interference_rebuild_owner }
		}
		AIO_interference_recount_sources = {
			OWNER = event_target:AIO_interference_rebuild_owner
	 	}
	}
}

# scope: system
AIO_interference_prototype_try_spread = {
	if = {
		limit = { has_psionic_aura = AIO_reality_interference_aura }
		aura_owner = { save_event_target_as = AIO_interference_spreading_owner }
		random_neighbor_system = {
			limit = {
				AIO_is_nearby_interference_source = {
					OWNER = event_target:AIO_interference_spreading_owner
				}
				has_psionic_aura = no
			}
			spawn_psionic_aura = {
				owner = event_target:AIO_interference_spreading_owner
				type = AIO_reality_interference_aura
				update_type = no
			}
			set_aura_intensity = 1
		}
	}
}

# Resolve existed aura and spawn interference aura
# scope: system
AIO_interference_prepare_source = {
	if = {
		limit = {
			NOT = {
				AND = {
					has_psionic_aura = AIO_reality_interference_aura
					aura_owner = { is_same_value = $OWNER$ }
				}
			}
		}
		if = {
			limit = { has_psionic_aura = yes }
			destroy_psionic_aura = yes
		}
		spawn_psionic_aura = {
			owner = $OWNER$
			type = AIO_reality_interference_aura
			update_type = no
		}
		set_aura_intensity = 1
	}
	set_star_flag = AIO_interference_source
}

# Count only this owner's source systems within range, including self.
# Scope: one aura system.
AIO_interference_recount_sources = {
	set_variable = { which = AIO_interference_sources value = 0 }
	save_event_target_as = AIO_interference_count_target
	aura_owner = {
		every_system_with_aura = {
			limit = {
				has_psionic_aura = AIO_reality_interference_aura
				has_star_flag = AIO_interference_source
				aura_owner = { is_same_value = $OWNER$ }
				OR = {
					is_same_value = event_target:AIO_interference_count_target
					closest_system = {
						use_bypasses = no
						min_steps = 1
						max_steps = $OWNER$.AIO_interference_range
						valid = {
							has_psionic_aura = AIO_reality_interference_aura
							aura_owner = { is_same_value = $OWNER$ }
						}
						limit = { is_same_value = event_target:AIO_interference_count_target }
					}
				}
			}
			event_target:AIO_interference_count_target = {
				change_variable = { which = AIO_interference_sources value = 1 }
			}
		}
	}
}

# Reset level when change policy
# scope: country
AIO_interference_change_policy = {
	remove_country_flag = AIO_interference_balance_residual
	remove_country_flag = AIO_interference_energy_residual
	remove_country_flag = AIO_interference_ecology_residual
	remove_country_flag = AIO_interference_industry_residual
	remove_country_flag = AIO_interference_metric_residual
	set_timed_country_flag = {
		flag = AIO_interference_change_policy
		years = 5
	}
	set_timed_country_flag = {
		flag = AIO_interference_$OLD_POLICY$_residual
		years = 5
	}
	save_event_target_as = AIO_interference_change_policy_owner
	every_system_with_aura = {
		limit = {
			has_psionic_aura = AIO_reality_interference_aura
			aura_owner = { is_same_value = event_target:AIO_interference_change_policy_owner }
		}
		set_aura_intensity = 1
		# Changing doctrine also lifts the seal left by a reversed accident.
		remove_star_flag = AIO_interference_backlash_sealed
	}
}


#region energy accidence

# Scope: aura system. The star flag serializes this system only, allowing the
# country to run several independent copies of the same situation.
AIO_interference_energy_start = {
	set_star_flag = AIO_energy_backlash_active
	save_event_target_as = AIO_energy_backlash_system
	aura_owner = {
		start_situation = {
			type = AIO_interference_energy_backlash
			target = event_target:AIO_energy_backlash_system
		}
	}
}

# Scope: system. Drops one escalation stage modifier if it is present.
AIO_interference_energy_strip_accidence = {
	if = {
		limit = { has_modifier = AIO_energy_accidence_1 }
		remove_modifier = AIO_energy_accidence_1
	}
	if = {
		limit = { has_modifier = AIO_energy_accidence_2 }
		remove_modifier = AIO_energy_accidence_2
	}
	if = {
		limit = { has_modifier = AIO_energy_accidence_3 }
		remove_modifier = AIO_energy_accidence_3
	}
}

# Scope: system. The stages are system modifiers, so colonies are cleared as
# well in case a stage was ever cached on them.
AIO_interference_energy_clear_accidence = {
	AIO_interference_energy_strip_accidence = yes
	every_system_planet_colony = { AIO_interference_energy_strip_accidence = yes }
	every_system_ship_colony = { AIO_interference_energy_strip_accidence = yes }
}

# Scope: situation. Each stage replaces the previous one, so benefit and cost
# both climb as the accident deepens.
AIO_interference_energy_modifier = {
	if = {
		limit = { exists = target }
		target = {
			AIO_interference_energy_strip_accidence = yes
			add_modifier = { modifier = AIO_energy_accidence_$STAGE$ }
		}
	}
}

# Scope: situation. Second stage discharge: the local grid goes dark. An active
# blockade contains it, halving what the discharge leaves behind.
AIO_interference_energy_surge_one = {
	if = {
		limit = {
			exists = target
			current_situation_approach = AIO_energy_blockade
		}
		target = {
			every_system_planet_colony = {
				add_planet_devastation = 10
				add_modifier = { modifier = AIO_energy_blackout days = 90 }
			}
			every_system_ship_colony = {
				add_planet_devastation = 10
				add_modifier = { modifier = AIO_energy_blackout days = 90 }
			}
		}
	}
	else = {
		limit = { exists = target }
		target = {
			every_system_planet_colony = {
				add_planet_devastation = 20
				add_modifier = { modifier = AIO_energy_blackout days = 180 }
			}
			every_system_ship_colony = {
				add_planet_devastation = 20
				add_modifier = { modifier = AIO_energy_blackout days = 180 }
			}
		}
	}
}

# Scope: situation. Third stage discharge: the cascade drags the national grid in.
AIO_interference_energy_surge_two = {
	if = {
		limit = {
			exists = target
			current_situation_approach = AIO_energy_blockade
		}
		target = {
			every_system_planet_colony = { add_planet_devastation = 10 }
			every_system_ship_colony = { add_planet_devastation = 10 }
		}
		target = { add_modifier = { modifier = AIO_energy_grid_drain days = 180 } }
	}
	else = {
		limit = { exists = target }
		target = {
			every_system_planet_colony = { add_planet_devastation = 20 }
			every_system_ship_colony = { add_planet_devastation = 20 }
		}
		target = { add_modifier = { modifier = AIO_energy_grid_drain days = 360 } }
	}
}

# Scope: situation. Returns this system's field to the start of level 2. The
# first value of level 2 is 1001: level 1 ends at 1000.
AIO_interference_energy_reset_aura = {
	if = {
		limit = { exists = target }
		target = {
			if = {
				limit = {
					has_psionic_aura = AIO_reality_interference_aura
					aura_owner = { is_same_value = root.owner }
				}
				set_aura_intensity = 1001
			}
		}
	}
}

# Scope: situation. If the accident sits inside someone else's borders, tell the
# owner, and let the opinion hit land: it is accumulative, so repeat accidents
# accidents add up, the way vanilla's opinion_cosmogenesis_thesis does.
AIO_interference_accident_notify = {
	if = {
		limit = { exists = target }
		owner = { save_event_target_as = AIO_accident_owner }
		target = {
			random_system_planet = {
				limit = {
					is_colony = yes
					has_owner = yes
					NOT = { owner = { is_same_value = root.owner } }
				}
				owner = {
					add_opinion_modifier = {
						modifier = opinion_AIO_interference_accident
						who = event_target:AIO_accident_owner
					}
					country_event = { id = AIO_interference_events.316 }
				}
			}
		}
	}
}

# Scope: situation. Every ending pays out the data the accident produced, so no
# approach is a pure loss. "value" is months of the owner's own income, clamped so
# a young empire still gets something and a late one cannot bank the whole tree:
# 24 / 500 / 1000000 is the scale vanilla uses for its tier 3 research reward.
AIO_interference_energy_award = {
	owner = {
		add_monthly_resource_mult = {
			resource = physics_research
			value = @tier3researchreward
			min = @tier3researchmin
			max = @tier3researchmax
		}
		complete_crisis_objective = crisobj_AIO_research_thesis
	}
}

# Scope: situation. The reversal lands before the cascade ends, so the field
# keeps the level it reached instead of being reset. The field stays critical,
# so an unlimited star flag is what stops the accident from restarting here; the
# field's on_vanish clears the flag if the field ever disappears.
AIO_interference_energy_tame = {
	if = {
		limit = { exists = target }
		target = {
			remove_star_flag = AIO_energy_backlash_active
			AIO_interference_energy_clear_accidence = yes
			set_star_flag = AIO_interference_backlash_sealed
		}
	}
	AIO_interference_energy_award = yes
	AIO_interference_accident_notify = yes
	destroy_situation = this
}

# Scope: situation. Nobody stopped it: the colonies take the hit and the field
# pays by dropping back to level 2.
AIO_interference_energy_collapse = {
	if = {
		limit = { exists = target }
		target = {
			remove_star_flag = AIO_energy_backlash_active
			AIO_interference_energy_clear_accidence = yes
			every_system_planet_colony = {
				add_planet_devastation = 40
				if = {
					limit = { NOT = { has_deposit = d_AIO_energy_cascade_residue } }
					add_blocker = { type = d_AIO_energy_cascade_residue }
				}
			}
			every_system_ship_colony = {
				add_planet_devastation = 40
				if = {
					limit = { NOT = { has_deposit = d_AIO_energy_cascade_residue } }
					add_blocker = { type = d_AIO_energy_cascade_residue }
				}
			}
		}
	}
	target = { add_modifier = { modifier = AIO_energy_grid_collapse days = 1825 } }
	AIO_interference_energy_award = yes
	AIO_interference_accident_notify = yes
	AIO_interference_energy_reset_aura = yes
	destroy_situation = this
}

# Scope: situation. Blockade or a doctrine change ends the accident early.
AIO_interference_energy_suppress = {
	if = {
		limit = { exists = target }
		target = {
			remove_star_flag = AIO_energy_backlash_active
			AIO_interference_energy_clear_accidence = yes
		}
	}
	AIO_interference_energy_award = yes
	AIO_interference_accident_notify = yes
	AIO_interference_energy_reset_aura = yes
	destroy_situation = this
}
