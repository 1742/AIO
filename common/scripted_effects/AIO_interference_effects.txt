# Scope: system.
AIO_interference_create_source = {
	AIO_interference_prepare_source = { OWNER = $OWNER$ }
	aura_owner = { AIO_interference_rebuild_effect = yes }
}

# Scope: system.
AIO_interference_remove_source = {
	remove_star_flag = AIO_interference_source
	if = {
		limit = { has_psionic_aura = AIO_reality_interference_aura }
		aura_owner = { AIO_interference_rebuild_effect = yes }
	}
}

# Call when interference source change
# scope: country
AIO_interference_rebuild_effect = {
	save_event_target_as = AIO_interference_rebuild_owner
	every_system_with_aura = {
		limit = {
			has_psionic_aura = AIO_reality_interference_aura
			aura_owner = { is_same_value = event_target:AIO_interference_rebuild_owner }
		}
		AIO_interference_recount_sources = {
			OWNER = event_target:AIO_interference_rebuild_owner
	 	}
	}
}

# scope: system
AIO_interference_prototype_try_spread = {
	if = {
		limit = { has_psionic_aura = AIO_reality_interference_aura }
		aura_owner = { save_event_target_as = AIO_interference_spreading_owner }
		random_neighbor_system = {
			limit = {
				AIO_is_nearby_interference_source = {
					OWNER = event_target:AIO_interference_spreading_owner
				}
				has_psionic_aura = no
			}
			spawn_psionic_aura = {
				owner = event_target:AIO_interference_spreading_owner
				type = AIO_reality_interference_aura
				update_type = no
			}
			set_aura_intensity = 1
		}
	}
}

# Resolve existed aura and spawn interference aura
# scope: system
AIO_interference_prepare_source = {
	if = {
		limit = {
			NOT = {
				AND = {
					has_psionic_aura = AIO_reality_interference_aura
					aura_owner = { is_same_value = $OWNER$ }
				}
			}
		}
		if = {
			limit = { has_psionic_aura = yes }
			destroy_psionic_aura = yes
		}
		spawn_psionic_aura = {
			owner = $OWNER$
			type = AIO_reality_interference_aura
			update_type = no
		}
		set_aura_intensity = 1
	}
	set_star_flag = AIO_interference_source
}

# Count only this owner's source systems within range, including self.
# Scope: one aura system.
AIO_interference_recount_sources = {
	set_variable = { which = AIO_interference_sources value = 0 }
	save_event_target_as = AIO_interference_count_target
	aura_owner = {
		every_system_with_aura = {
			limit = {
				has_psionic_aura = AIO_reality_interference_aura
				has_star_flag = AIO_interference_source
				aura_owner = { is_same_value = $OWNER$ }
				OR = {
					is_same_value = event_target:AIO_interference_count_target
					closest_system = {
						use_bypasses = no
						min_steps = 1
						max_steps = $OWNER$.AIO_interference_range
						valid = {
							has_psionic_aura = AIO_reality_interference_aura
							aura_owner = { is_same_value = $OWNER$ }
						}
						limit = { is_same_value = event_target:AIO_interference_count_target }
					}
				}
			}
			event_target:AIO_interference_count_target = {
				change_variable = { which = AIO_interference_sources value = 1 }
			}
		}
	}
}

# Reset level when change policy
# scope: country
AIO_interference_change_policy = {
	remove_country_flag = AIO_interference_balance_residual
	remove_country_flag = AIO_interference_energy_residual
	remove_country_flag = AIO_interference_ecology_residual
	remove_country_flag = AIO_interference_industry_residual
	remove_country_flag = AIO_interference_metric_residual
	set_timed_country_flag = {
		flag = AIO_interference_change_policy
		years = 5
	}
	set_timed_country_flag = {
		flag = AIO_interference_$OLD_POLICY$_residual
		years = 5
	}
	save_event_target_as = AIO_interference_change_policy_owner
	every_system_with_aura = {
		limit = {
			has_psionic_aura = AIO_reality_interference_aura
			aura_owner = { is_same_value = event_target:AIO_interference_change_policy_owner }
		}
		set_aura_intensity = 1
	}
}
