# country；FROM = 锁定星系。确认后收费，720 天后仅在该星系召唤。
AIO_projection_begin = {
    add_resource = { AIO_advanced_logic = @AIO_projection_logic_payment }
    hidden_effect = {
        set_timed_country_flag = { flag = AIO_projection_cooldown days = @AIO_projection_cooldown_days }
        from = {
            save_event_target_as = AIO_projection_Aforgomon_target
            set_star_flag = AIO_projection_pending
            system_event = { id = AIO_projection_events.302 days = @AIO_projection_delay_days }
            AIO_projection_dispatch_notice = {
                MESSAGE_TYPE = MESSAGE_DYSON_SHOT
                MESSAGE = AIO_projection_Aforgomon.located
            }
            random_cosmic_storm = {
                limit = { any_system_within_storm = { is_same_value = event_target:AIO_projection_Aforgomon_target } }
                destroy_cosmic_storm = yes
            }
            create_cosmic_storm = {
                type = nexus_storm
                immediate = yes
                cosmic_storm_start_position = this
            }
            last_created_cosmic_storm = { lock_storm_in_place = yes }
        }
    }
}

# system；通知只定位锁定星系，不依赖施法者继续存在。
AIO_projection_dispatch_notice = {
    save_event_target_as = AIO_projection_Aforgomon_target
    every_playable_country = {
        create_message = {
            type = $MESSAGE_TYPE$
            localization = $MESSAGE$
            days = 30
            target = event_target:AIO_projection_Aforgomon_target
        }
    }
}

# system；只在当前星系生成，保留正式化身的怒气与复生逻辑。
AIO_projection_Aforgomon_effect = {
    AIO_ensure_Aforgomon_country = yes
    set_variable = { which = AIO_Aforgomon_Wrath value = 0 }
    set_star_flag = AIO_Aforgomon_existed
    AIO_create_Aforgomon_fleet = {
        DESIGN = NAME_AIO_Aforgomon_Wrath
        FLAG = AIO_projection_Aforgomon
    }
}
