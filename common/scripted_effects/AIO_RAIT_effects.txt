AIO_RAIT_opinion_modifier = {
    if = {
        limit = {
            NOR = {
                is_same_value = event_target:AIO_crisis_country
                has_ascension_perk = ap_cosmogenesis
            }
            has_communications = event_target:AIO_crisis_country
        }
        add_opinion_modifier = {
            modifier = opinion_cosmogenesis_thesis
            who = event_target:AIO_crisis_country
        }
    }
    if = {
        limit = { is_same_value = event_target:AIO_crisis_country }
        custom_tooltip = cosmogenesis_negative_opinion_tt
    }
}

AIO_mass_shift_effect = {
    every_owned_planet = {
        limit = {
            has_modifier = low_gravity
            is_artificial = no
        }
        random_list = {
            90 = {  }
            10 = {
                modifier = {
                    factor = 0.5
                    planet_size > 16
                }
                modifier = {
                    factor = 0.25
                    planet_size > 19
                }
                modifier = {
                    factor = 1.5
                    planet_size < 12
                }
                modifier = {
                    factor = 2
                    planet_size < 8
                }
                add_modifier = { modifier = leaking_atmosphere days = -1 }
                enable_special_project = { name = COSMOGENESIS_LEAKING_ATMOSPHERE }
                planet = { set_timed_planet_flag = { flag = mass_shift_affected days = 5 } }
            }
        }
    }
    every_owned_planet = {
        limit = {
            NOR = {
                has_modifier = high_gravity
                has_modifier = low_gravity
            }
            is_artificial = no
        }
        random_list = {
            90 = {  }
            10 = {
                modifier = {
                    factor = 0.5
                    planet_size > 19
                }
                modifier = {
                    factor = 0.5
                    planet_size > 22
                }
                modifier = {
                    factor = 2
                    planet_size < 16
                }
                modifier = {
                    factor = 2
                    planet_size < 12
                }
                add_modifier = { modifier = low_gravity days = -1 }
                planet = { set_timed_planet_flag = { flag = mass_shift_affected days = 5 } }
            }
        }
    }
    every_owned_planet = {
        limit = {
            has_modifier = high_gravity
            is_artificial = no
        }
        random_list = {
            40 = {  }
            60 = {
                remove_modifier = high_gravity
                planet = { set_timed_planet_flag = { flag = mass_shift_affected days = 5 } }
            }
        }
    }

    if = {
        limit = { any_owned_planet = { planet = { has_planet_flag = mass_shift_affected } } }
        custom_tooltip = mass_shift_tooltip
        owned_planet_list_tooltip = { limit = { planet = { has_planet_flag = mass_shift_affected } } }
    }
    AIO_RAIT_opinion_modifier = yes
}

# scope: colony
AIO_time_ruin_effect = {
    optimize_memory

    [[STAGE_1]
        if = {
            limit = {
                has_carrier_flag = AIO_RAIT_time_ruin_colony
                NOT = { has_modifier = AIO_RAIT_TR_stage_1_modifier }
            }
            add_modifier = { modifier = AIO_RAIT_TR_stage_1_modifier days = -1 }
            add_deposit = d_AIO_TR_stage_1
        }
        owner = { AIO_RAIT_opinion_modifier = yes }
    ]
    [[STAGE_2_OPT1]
        remove_modifier = AIO_RAIT_TR_stage_1_modifier
        add_modifier = { modifier = AIO_RAIT_TR_stage_2_opt_1_modifier days = -1 }
        owner = { add_resource = { energy = -2000 } }
        owner = {
            add_monthly_resource_mult = {
                resource = unity
                value = @tier3unityreward
                min = @tier3unitymin
                max = @tier3unitymax
            }
        }
    ]
    [[STAGE_2_OPT2]
        remove_modifier = AIO_RAIT_TR_stage_1_modifier
        add_modifier = { modifier = AIO_RAIT_TR_stage_2_opt_2_modifier days = -1 }
    ]
    [[STAGE_3]
        if = {
            limit = { has_modifier = AIO_RAIT_TR_stage_2_opt_1_modifier }
            remove_modifier = AIO_RAIT_TR_stage_2_opt_1_modifier
            add_modifier = { modifier = AIO_RAIT_TR_stage_3_liquidation_modifier days = -1 }
        }
        else_if = {
            limit = { has_modifier = AIO_RAIT_TR_stage_2_opt_2_modifier }
            remove_modifier = AIO_RAIT_TR_stage_2_opt_2_modifier
            add_modifier = { modifier = AIO_RAIT_TR_stage_3_unliquidation_modifier days = -1 }
        }
        add_deposit = d_AIO_TR_stage_3

        owner = {
            if = {
                limit = { AIO_can_AIT = yes }
                add_event_chain_counter = {
                    event_chain = AIO_RAIT_TR_events_chain
                    counter = AIO_AIT_TR_events_chain
                    amount = 1
                }
            }
        }
        enable_special_project = {
            name = AIO_AIT_TIME_RUIN_STAGE_3
            owner = owner
            location = this
        }

        hidden_effect = { carrier_event = { id = AIO_RAIT_events.1105 days = 360 random = 90 } }
    ]
    [[STAGE_4]
        if = {
            limit = {
                owner = { AIO_can_AIT = no }
                OR= {
                    has_modifier = AIO_RAIT_TR_stage_2_opt_1_modifier  # 镇压和清算
                    has_modifier = AIO_RAIT_TR_stage_3_liquidation_modifier  # 镇压、清算和研究
                }
            }
            add_modifier = { modifier = AIO_RAIT_TR_stage_4_liquidation_modifier years = 3 }
            add_deposit = d_AIO_TR_monument
            owner = {
                add_monthly_resource_mult = {
                    resource = unity
                    value = @tier3unityreward
                    min = @tier3unitymin
                    max = @tier3unitymax
                }
            }
        }
        else_if = {
            limit = {
                owner = {
                    is_gestalt = no
                    AIO_can_AIT = no
                }
                OR = {
                    has_modifier = AIO_RAIT_TR_stage_1_modifier  # 啥也没干
                    has_modifier = AIO_RAIT_TR_stage_2_opt_2_modifier  # 镇压、不清算
                    has_modifier = AIO_RAIT_TR_stage_3_unliquidation_modifier  # 镇压、不清算和研究
                }
            }
            add_planet_devastation = 20
            owner = { add_modifier = { modifier = AIO_RAIT_TR_stage_4_unliquidation_modifier years = 3 } }
        }
        else_if = {
            limit = { has_carrier_flag = AIO_finish_TR_special_project_release }
            owner = {
                add_monthly_resource_mult = {
                    resource = physics_research
                    value = @tier4researchreward
                    min = @tier4researchmin
                    max = @tier4researchmax
                }
            }
        }
    ]
    [[CLEAR]
        if = {
            limit = { has_carrier_flag = AIO_RAIT_time_ruin_colony }

            owner = {
                if = {
                    limit = { has_event_chain = AIO_RAIT_TR_events_chain }
                    end_event_chain = AIO_RAIT_TR_events_chain
                }
            }

            if = {
                limit = { has_modifier = AIO_RAIT_TR_stage_1_modifier }
                remove_modifier = AIO_RAIT_TR_stage_1_modifier
            }
            if = {
                limit = { has_modifier = AIO_RAIT_TR_stage_2_opt_1_modifier }
                remove_modifier = AIO_RAIT_TR_stage_2_opt_1_modifier
            }
            if = {
                limit = { has_modifier = AIO_RAIT_TR_stage_2_opt_2_modifier }
                remove_modifier = AIO_RAIT_TR_stage_2_opt_2_modifier
            }
            if = {
                limit = { has_modifier = AIO_RAIT_TR_stage_3_liquidation_modifier }
                remove_modifier = AIO_RAIT_TR_stage_3_liquidation_modifier
            }
            if = {
                limit = { has_modifier = AIO_RAIT_TR_stage_3_unliquidation_modifier }
                remove_modifier = AIO_RAIT_TR_stage_3_unliquidation_modifier
            }
            if = {
                limit = { has_deposit = d_AIO_TR_stage_1 }
                remove_deposit = d_AIO_TR_stage_1
            }
            if = {
                limit = { has_deposit = d_AIO_TR_stage_3 }
                remove_deposit = d_AIO_TR_stage_3
            }

            if = {
                limit = { has_carrier_flag = AIO_finish_TR_special_project_release }
                remove_carrier_flag = AIO_finish_TR_special_project_release
            }

            remove_carrier_flag = AIO_RAIT_time_ruin_colony
        }
    ]
}

        }

        owner = {
            if = {
                limit = { has_event_chain = AIO_RAIT_GS_events_chain }
                end_event_chain = AIO_RAIT_GS_events_chain
            }
        }
    ]
    [[CLEAR]
        star = {
            if = {
                limit = { has_planet_flag = AIO_RAIT_gravity_swell_planet }
                remove_planet_flag = AIO_RAIT_gravity_swell_planet
            }
            if = {
                limit = { has_planet_flag = AIO_RAIT_GS_finish_reverse }
                remove_planet_flag = AIO_RAIT_GS_finish_reverse
            }
            if = {
                limit = { has_deposit = d_AIO_GS_deposit }
                remove_deposit = d_AIO_GS_deposit
            }
        }
        if = {
            limit = {
                exists = owner
                owner = { has_event_chain = AIO_RAIT_GS_events_chain }
            }
            owner = { end_event_chain = AIO_RAIT_GS_events_chain }
        }
    ]
