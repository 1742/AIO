AIO_safe_change_pc = {
    optimize_memory

    if = {
        limit = {
            NOR = {
                is_artificial = yes
                is_planet_class = $NEW_PC$
                is_planet_class = pc_ai
                colony? = { has_carrier_flag = synth_queen_bastille }
            }
        }
        change_pc = $NEW_PC$
        if = {
            limit = {
                is_colony = yes
                is_colonizable = no
            }
            destroy_colony = yes
        }
    }
}

# 星系内行星变为冰封星球
AIO_change_system_planet_in_frozen = {
    every_system_planet = {
        if = {
            limit = {
                OR = {
                    is_planet_class = pc_molten
                    is_planet_class = pc_toxic
                    habitable_planet = yes
                }
            }
            AIO_safe_change_pc = { NEW_PC = pc_frozen }
        }
        else_if = {
            limit = { is_planet_class = pc_barren }
            change_pc = pc_barren_cold
        }
    }
    remove_system_terraforming_candidates = yes
}

# 安全的随机改变天体类型，scope: planet, root: AIO_empire
AIO_safe_change_random_pc = {
    optimize_memory

    if = {
        limit = { AIO_safe_remove_planet_exclude_API = no }

        if = {
            # 恒星
            limit = { is_star = yes }
            AIO_change_random_pc = { STAR = 1 }
            solar_system = {
                AIO_change_star_class = yes
                every_system_planet_colony = {
                    limit = {
                        is_artificial = no
                        AIO_colony_has_ring = no

                        NOR = {
                            is_planet_class = pc_ai
                            has_carrier_flag = synth_queen_bastille
                        }
                    }
                    destroy_colony = yes
                    reset_planet = yes
                }
            }
        }
        else = {
            # 行星
            AIO_change_random_pc = { PLANET = 1 }
        }
    }
}
