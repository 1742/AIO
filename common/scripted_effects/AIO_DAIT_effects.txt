# 为星系内天体添加矿藏
AIO_DAIT_add_deposits_effect = {
    every_system_planet = {
        limit = { NOT = { is_planet_class = pc_AIO_empty_space } }
        if = {
            limit = { has_research_station = yes }
            research_station = { destroy_fleet = this }
            clear_deposits = yes
        }
        while = {
            count = $TIMES|1$
            random_list = {
                1 = { add_deposit = d_energy_3 }
                1 = { add_deposit = d_energy_5 }
                1 = { add_deposit = d_energy_10 }
                1 = { add_deposit = d_minerals_3 }
                1 = { add_deposit = d_minerals_5 }
                1 = { add_deposit = d_minerals_10 }
                1 = { add_deposit = d_food_3 }
                1 = { add_deposit = d_food_5 }
                1 = { add_deposit = d_food_10 }
                1 = { add_deposit = d_alloys_2 }
                1 = { add_deposit = d_alloys_4 }
                1 = { add_deposit = d_alloys_5 }
                1 = { add_deposit = d_volatile_motes_1 }
                1 = { add_deposit = d_volatile_motes_2 }
                1 = { add_deposit = d_volatile_motes_3 }
                1 = { add_deposit = d_exotic_gases_1 }
                1 = { add_deposit = d_exotic_gases_2 }

                1 = { add_deposit = d_rare_crystals_1 }
                1 = { add_deposit = d_rare_crystals_2 }

        }
    }
}

# 按最适合的天体的实际轨道距离赋予气候
AIO_DAIT_reshape_habitable_planet_effect = {
    ordered_system_planet = {
        limit = {
            is_star = no
            is_artificial = no
            is_colony = no
            habitable_planet = no
            planet_size > 5
            AIO_safe_remove_planet_exclude_API = no
        }
        order_by = value:AIO_DAIT_habitable_zone_distance_score
        save_event_target_as = AIO_DAIT_reshape_habitable_planet
    }

    event_target:AIO_DAIT_reshape_habitable_planet? = {
        set_variable = {
            which = AIO_DAIT_distance_to_star
            value = value:distance_to_system_star
        }
        if = {
            limit = { check_variable = { which = AIO_DAIT_distance_to_star value < 75 } }
            random_list = {
                1 = { AIO_safe_change_pc = { NEW_PC = pc_desert } }
                1 = { AIO_safe_change_pc = { NEW_PC = pc_arid } }
                1 = { AIO_safe_change_pc = { NEW_PC = pc_savannah } }
            }
        }
        else_if = {
            limit = { check_variable = { which = AIO_DAIT_distance_to_star value <= 125 } }
            random_list = {
                1 = { AIO_safe_change_pc = { NEW_PC = pc_continental } }
                1 = { AIO_safe_change_pc = { NEW_PC = pc_ocean } }
                1 = { AIO_safe_change_pc = { NEW_PC = pc_tropical } }
            }
        }
        else = {
            random_list = {
                1 = { AIO_safe_change_pc = { NEW_PC = pc_alpine } }
                1 = { AIO_safe_change_pc = { NEW_PC = pc_arctic } }
                1 = { AIO_safe_change_pc = { NEW_PC = pc_tundra } }
            }
        }
        clear_variable = AIO_DAIT_distance_to_star
    }
}

# 在星系内随机宜居行星上创造未开智生命

        }
        create_pop_group = { species = event_target:AIO_DAIT_created_life_species }
        check_planet_employment = yes
    }
}

# 退化星系内所有殖民地的所有智慧物种。
AIO_DAIT_degenerate_effect = {
    every_system_planet_colony = {
        limit = {
            owner = {
                OR = {
                    is_country_type = default
                    is_country_type = fallen_empire
                    is_country_type = awakened_fallen_empire
                }
            }
            NOR = {
                is_planet_class = pc_ai
                has_carrier_flag = synth_queen_bastille
                planet = { AIO_safe_remove_planet_exclude_API = yes }
            }
        }
        if = {
            limit = { is_planet_class = pc_ark }
            fleet? = { destroy_fleet = this }
        }
        else_if = {
            limit = { NOT = { any_owned_pop_group = { is_robot_pop = no } } }
            destroy_colony = yes
        }
        else = {
            every_owned_pop_group = {
                limit = {
                    OR = {
                        has_trait = trait_mechanical
                        has_trait = trait_machine_unit
                    }
                }
                kill_all_pop = yes
            }
            every_planet_army = { remove_army = yes }
            while = {
                limit = { any_owned_pop_group = { is_sapient = yes } }
                weighted_random_owned_pop_group = {
                    limit = { is_sapient = yes }
                    species = { save_event_target_as = species_target }
                    create_devolved_species = yes
                }
                every_owned_pop_group = {
                    limit = {
                        is_same_species = event_target:species_target
                        is_sapient = yes
                    }
                    change_species = event_target:devolved_species_target
                }
            }
            check_planet_employment = yes
        }
    }
}

# 提升星系内未开智物种。
AIO_DAIT_enlighted_effect = {
    every_system_planet = {
        limit = {
            pop_amount > 1
            NOR = {
                is_planet_class = pc_ai
                colony? = { has_carrier_flag = synth_queen_bastille }
            }
        }
        weighted_random_owned_pop_group = {
            if = {
                limit = {
                    is_robot_pop = no
                    NOT = { has_trait = trait_presapient_forcefully_devolved }
                }
                modify_species = {
                    species = this.species
                    sapient = yes
                    add_trait = trait_latent_psionic
                    effect = { save_event_target_as = AIO_DAIT_enlighten_species }
                }
            }
            else_if = {
                limit = { has_trait = trait_presapient_forcefully_devolved }
                modify_species = {
                    species = this.species
                    sapient = yes
                    add_trait = trait_latent_psionic
                    remove_trait = trait_presapient_forcefully_devolved
                    effect = { save_event_target_as = AIO_DAIT_enlighten_species }
                }
            }
            else_if = {
                limit = { is_robot_pop = yes }
                modify_species = {
                    species = this.species
                    sapient = yes
                    add_trait = trait_robot_latent_psionic
                    effect = { save_event_target_as = AIO_DAIT_enlighten_species }
                }
            }
            else = {
                modify_species = {
                    species = this.species
                    add_trait = trait_latent_psionic
                    effect = { save_event_target_as = AIO_DAIT_enlighten_species }
                }
            }
            change_species = event_target:AIO_DAIT_enlighten_species
        }
        check_planet_employment = yes
        if = {
            limit = { NOT = { exists = owner } }
            set_planet_flag = AIO_DAIT_enlighten_no_along
        }
    }
    if = {
        limit = {
            OR = {
                NOT = { exists = owner }
                owner? = { is_nomadic = yes }
            }
        }
        create_country = {
            type = default
            origin = origin_default
            effect = {
                shift_ethic = ethic_spiritualist
                change_government = random
                create_leader = { class = random_ruler }
                create_leader = { class = official }
                create_leader = { class = scientist }
                create_leader = { class = commander }
                prev = {
                    create_starbase = {
                        size = starbase_outpost
                        owner = prev
                        module = shipyard
                        module = shipyard
                    }
                    create_fleet = {
                        effect = {
                            set_owner = prevprev
                            create_ship = {
                                name = random
                                design = NAME_Scholarium_Science
                            }
                            set_location = {
                                target = prev
                                angle = random
                                direction = in_system
                            }
                        }
                    }
                    create_fleet = {
                        effect = {
                            set_owner = prevprev
                            create_ship = {
                                name = random
                                design = NAME_Prospectorium_Constructor
                            }
                            set_location = {
                                target = prev
                                angle = random
                                direction = in_system
                            }
                        }
                    }
                    random_system_planet = {
                        limit = { has_planet_flag = AIO_DAIT_enlighten_no_along }
                        set_owner = prevprev
                        set_capital = yes
                        remove_planet_flag = AIO_DAIT_enlighten_no_along
                    }
                }
                add_resource = {
                    energy = 1000
                    minerals = 1000
                    food = 1000
                    consumer_goods = 500
                    alloys = 500
                    unity = 1000
                }
                add_modifier = { modifier = AIO_the_craving_of_stellaris_modifier years = 10 }
                save_event_target_as = AIO_DAIT_enlighten_country
            }
        }
    }
    else = {
        owner = { save_event_target_as = AIO_DAIT_enlighten_country }
    }
    every_system_planet = {
        limit = { has_planet_flag = AIO_DAIT_enlighten_no_along }
        set_owner = event_target:AIO_DAIT_enlighten_country
        check_planet_employment = yes
        remove_planet_flag = AIO_DAIT_enlighten_no_along
    }
}

# 召唤亚弗戈蒙
AIO_DAIT_Aforgomom_effect = {
    if = {
        limit = { NOT = { exists = event_target:AIO_Aforgomom_country } }
        AIO_create_Aforgomom_effect = yes
    }
    create_fleet = {
        effect = {
            set_name = AIO_Aforgomom_Wrath
            set_owner = event_target:AIO_Aforgomom_country
            create_ship = {
                name = AIO_Aforgomom_Wrath
                design = NAME_AIO_Aforgomom_Wrath
                graphical_culture = event_target:AIO_Aforgomom_country
            }
            set_location = {
                target = prev.star
                distance = 0
                angle = random
            }
            # 加buff
            if = {
                limit = {
                    # 检查数值是否溢出
                    event_target:AIO_Aforgomom_country = {
                        check_variable = { which = AIO_Aforgomom_Wrath value > 0 }
                        check_variable = { which = AIO_Aforgomom_Wrath value < 10 }
                    }
                }
                add_modifier = {
                    modifier = AIO_Aforgomom_Wrath_modifier
                    mult = value:AIO_Aforgomom_Wrath_factor
                    days = -1
                }
            }
            set_fleet_stance = aggressive
            set_aggro_range = 10000
            set_fleet_flag = AIO_Aforgomom_fleet

            fleet_event = { id = AIO_Aforgomom_events.201 days = 100 }
        }
    }
}

# 按天体类型和行星规模结算资源，然后将星系内天体收割为破碎世界。
AIO_DAIT_harvest_system_effect = {
    every_system_planet = {
        limit = {
            NOR = {
                is_planet_class = pc_AIO_empty_space
                is_planet_class = pc_shattered
                is_planet_class = pc_ai
                colony? = { has_carrier_flag = synth_queen_bastille }
                AIO_safe_remove_planet_exclude_API = yes
                has_planet_flag = AIO_DAIT_harvested_body
            }
        }
        if = {
            limit = { is_star = yes }
            root = { add_resource = { energy = 2000 mult = prev.trigger:planet_size } }
            change_pc = pc_t_star
            solar_system = { AIO_change_star_class = yes }
            set_planet_flag = AIO_DAIT_harvested_body
        }
        else_if = {
            limit = { is_artificial = no }
            if = {
                limit = {
                    OR = {
                        is_planet_class = pc_barren
                        is_planet_class = pc_barren_cold
                        is_planet_class = pc_asteroid
                        is_planet_class = pc_crystal_asteroid
                        is_planet_class = pc_broken
                    }
                }
                root = { add_resource = { minerals = 400 mult = prev.trigger:planet_size } }
            }
            else_if = {
                limit = {
                    OR = {
                        is_planet_class = pc_gas_giant
                        is_planet_class = pc_toxic
                    }
                }
                root = {
                    add_resource = { energy = 200 mult = prev.trigger:planet_size }
                    add_resource = { exotic_gases = 100 mult = prev.trigger:planet_size }
                }
            }
            else_if = {
                limit = {
                    OR = {
                        is_planet_class = pc_molten
                        is_planet_class = pc_volcanic
                    }
                }
                root = {
                    add_resource = { energy = 200 mult = prev.trigger:planet_size }
                    add_resource = { volatile_motes = 50 mult = prev.trigger:planet_size }
                }
            }
            else_if = {
                limit = { is_planet_class = pc_frozen }
                root = {
                    add_resource = { minerals = 200 mult = prev.trigger:planet_size }
                    add_resource = { rare_crystals = 50 mult = prev.trigger:planet_size }
                }
            }
            else_if = {
                limit = { is_planet_class = pc_shrouded }
                root = { add_resource = { sr_zro = 200 mult = prev.trigger:planet_size } }
            }
            else_if = {
                limit = { is_planet_class = pc_nanotech }
                root = { add_resource = { nanites = 200 mult = prev.trigger:planet_size } }
            }
            else_if = {
                limit = { is_colonizable = yes }
                root = {
                    add_resource = { minerals = 200 mult = prev.trigger:planet_size }
                    add_resource = { food = 250 mult = prev.trigger:planet_size }
                }
            }
            else = {
                root = { add_resource = { minerals = 200 mult = prev.trigger:planet_size } }
            }
            clear_deposits = yes
            clear_planet_modifiers = yes
            if = {
                limit = { is_colony = yes }
                destroy_colony = yes
            }
            AIO_safe_change_pc = { NEW_PC = pc_shattered }
            set_planet_flag = AIO_DAIT_harvested_body
        }
        else = {
            switch = {
                trigger = is_planet_class
                pc_habitat = {
                    root = { add_resource = { alloys = 1000 } }
                    AIO_safe_remove_habitat = yes
                }
                pc_ringworld_habitable = {
                    root = { add_resource = { alloys = 8000 } }
                    AIO_safe_remove_rw_section = yes
                }
                pc_cosmogenesis_world = {
                    root = { add_resource = { alloys = 10000 } }
                    AIO_safe_remove_cosmogenesis_world = yes
                }
                pc_ark = {
                    root = { add_resource = { alloys = 5000 } }
                    fleet? = { destroy_fleet = this }
                }
            }
        }
    }
}
