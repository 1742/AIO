AIO_DAIT_enrich_effect = {
    every_system_planet = {
        limit = {
            NOT = {
                is_planet_class = pc_shattered
                is_planet_class = pc_AIO_empty_space
                is_planet_class = pc_astral_scar
                is_planet_class = pc_rift_star
                has_planet_flag = AIO_DAIT_harvested_body
            }
        }
        if = {
            limit = { has_research_station = yes }
            research_station = { destroy_fleet = this }
            clear_deposits = yes
        }
        while = {
            count = $TIMES|1$
            random_list = {
                3 = { add_deposit = d_energy_10 }
                3 = { add_deposit = d_minerals_10 }
                3 = { add_deposit = d_food_10 }
                2 = { add_deposit = d_alloys_25 }
                1 = { add_deposit = d_volatile_motes_5 }
                1 = { add_deposit = d_exotic_gases_5 }
                1 = { add_deposit = d_rare_crystals_5 }
            }
        }
    }

    owner? = {
        if = {
            limit = { NOT = { has_modifier = AIO_DAIT_enrich_modifier } }
            add_modifier = { modifier = AIO_DAIT_enrich_modifier }
        }
    }
    every_system_ship_colony = {
        if = {
            limit = {
                owner = {
                    is_country_type = default
                    NOT = { has_modifier = AIO_DAIT_enrich_modifier }
                }
            }
            owner = { add_modifier = { modifier = AIO_DAIT_enrich_modifier } }
        }
    }
}

AIO_DAIT_harvest_effect = {
    every_system_planet = {
        limit = {
            NOR = {
                is_planet_class = pc_shattered
                is_planet_class = pc_black_hole
                is_planet_class = pc_AIO_empty_space
                has_carrier_flag = synth_queen_bastille
                AIO_safe_remove_planet_exclude_API = yes
                has_planet_flag = AIO_DAIT_harvested_body
            }
        }
        if = {
            limit = { is_star = yes }
            event_target:AIO_DAIT_maker = {
                add_resource = { energy = 2000 mult = prev.trigger:planet_size }
            }
            change_pc = pc_t_star
            solar_system = { AIO_change_star_class = yes }
            set_planet_flag = AIO_DAIT_harvested_body
        }
        else_if = {
            limit = { is_artificial = no }
            if = {
                limit = {
                    OR = {
                        is_planet_class = pc_barren
                        is_planet_class = pc_barren_cold
                        is_planet_class = pc_asteroid
                        is_planet_class = pc_crystal_asteroid

                    add_resource = { energy = 200 mult = prev.trigger:planet_size }

                    add_resource = { energy = 200 mult = prev.trigger:planet_size }
                    add_resource = { volatile_motes = 50 mult = prev.trigger:planet_size }
                }

                limit = { is_planet_class = pc_frozen }

            }

                    add_resource = { minerals = 200 mult = prev.trigger:planet_size }

                    add_resource = { minerals = 200 mult = prev.trigger:planet_size }

            clear_planet_modifiers = yes

                    }

                }
            }
        }
    }
    set_star_flag = AIO_DAIT_harvest_system

AIO_DAIT_strike_effect = {
    if = {
        limit = { NOT = { exists = event_target:AIO_Aforgomom_country } }
        AIO_create_Aforgomom_effect = yes
    }
    if = {
        limit = { NOT = { has_star_flag = AIO_Aformom_exist } }
        AIO_create_Aformom_fleet = yes
        set_star_flag = AIO_Aformom_exist
    }
    if = {
        limit = { event_target:AIO_DAIT_maker = { is_country_type = AIO_empire } }
        every_fleet_in_system = {
            limit = {
                exists = owner
                NOR = {
                    has_fleet_flag = AIO_Aforgomom_fleet
                    owner = { is_nomadic = yes }
                }
            }
            owner = {
                every_system_within_border = {
                    limit = {
                        NOR = {
                            has_star_flag = AIO_HS_system
                            has_star_flag = AIO_Aformom_exist
                        }
                    }
                    AIO_create_Aformom_fleet = yes
                    set_star_flag = AIO_Aformom_exist
                }
                every_owned_fleet = {
                    limit = {
                        solar_system = {
                            NOR = {
                                has_star_flag = AIO_HS_system
                                has_star_flag = AIO_Aformom_exist
                            }
                        }
                    }
                    solar_system = {
                        AIO_create_Aformom_fleet = yes
                        set_star_flag = AIO_Aformom_exist
                    }
                }
            }
        }
    }
}

    create_fleet = {
        effect = {
            set_name = AIO_Aforgomom_Wrath

            create_ship = {
                name = AIO_Aforgomom_Wrath
                design = NAME_AIO_Aforgomom_Wrath
                graphical_culture = event_target:AIO_Aforgomom_country
            }
            set_location = {
                target = prev.star
                distance = 0
                angle = random
            }
            if = {
                limit = {
                    owner = {
                        check_variable = { which = AIO_Aforgomom_Wrath value > 0 }
                        check_variable = { which = AIO_Aforgomom_Wrath value < 10 }
                    }
                }
                add_modifier = { modifier = AIO_Aforgomom_Wrath_modifier mult = value:AIO_Aforgomom_Wrath_factor }
            }
            set_fleet_stance = aggressive
            set_aggro_range = 10000
            set_fleet_flag = AIO_Aforgomom_fleet
            fleet_event = { id = AIO_Aforgomom_events.201 days = 100 }
        }
    }
}
