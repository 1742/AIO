AIO_DAIT_enrich_effect = {
    owner? = { AIO_DAIT_mark_affected_country_effect = yes }
    every_system_ship_colony = { owner = { AIO_DAIT_mark_affected_country_effect = yes } }

    every_system_planet = {
        limit = {
            NOT = {
                is_planet_class = pc_shattered
                is_planet_class = pc_AIO_empty_space
                has_planet_flag = AIO_DAIT_harvested_body
            }
        }
        if = {
            limit = { has_research_station = yes }
            research_station = { destroy_fleet = this }
            clear_deposits = yes
        }
        while = {
            count = $TIMES|1$
            random_list = {
                3 = { add_deposit = d_energy_10 }
                3 = { add_deposit = d_minerals_10 }
                3 = { add_deposit = d_food_10 }
                2 = { add_deposit = d_alloys_25 }
                1 = { add_deposit = d_volatile_motes_5 }
                1 = { add_deposit = d_exotic_gases_5 }
                1 = { add_deposit = d_rare_crystals_5 }
            }
        }
    }

    owner? = {
        if = {
            limit = { NOT = { has_modifier = AIO_DAIT_enrich_modifier } }
            add_modifier = { modifier = AIO_DAIT_enrich_modifier }
        }
    }
    every_system_ship_colony = {
        if = {
            limit = { NOT = { has_modifier = AIO_DAIT_enrich_modifier } }
            add_modifier = { modifier = AIO_DAIT_enrich_modifier }
        }
    }

    AIO_DAIT_notify_enrich_effect = yes
}

AIO_DAIT_spawn_effect = {
    owner? = { AIO_DAIT_mark_affected_country_effect = yes }
    every_system_ship_colony = { owner = { AIO_DAIT_mark_affected_country_effect = yes } }

    if = {
        limit = {
            OR = {
                is_star_class = sc_AIO_void
                has_star_flag = AIO_DAIT_harvest_system
            }
        }
        every_system_planet = { AIO_safe_remove_planet = yes }
        AIO_safe_remove_system_megastructures = yes
        every_ambient_object = { destroy_ambient_object = this }
        set_star_class = sc_AIO_void
        random_list = {
            1 = { AIO_create_hexagonal_orbital_array = yes }
            1 = { AIO_create_ringworld_array = yes }
            1 = { AIO_create_triangular_planet_array = yes }
            1 = { AIO_create_black_hole_cluster = yes }
            1 = { AIO_create_concentric_orbital_array = yes }
            1 = { AIO_create_recursive_satellite_tree = yes }
            1 = { AIO_create_black_hole_flower = yes }
        }
        remove_star_flag = AIO_DAIT_harvest_system
    }
    else_if = {
        limit = {
            OR= {
                owner? = {
                    is_country_type = default
                    NOT = { has_modifier = AIO_DAIT_spawn_modifier }
                }
                any_system_ship_colony = {
                    owner = {
                        is_country_type = default
                        NOT = { has_modifier = AIO_DAIT_spawn_modifier }
                    }
                }
            }
        }
        owner? = { add_modifier = { modifier = AIO_DAIT_spawn_modifier } }
        every_system_ship_colony = {
            limit = { owner = { is_country_type = default } }
            owner = {
                if = {
                    limit = { NOT = { has_modifier = AIO_DAIT_spawn_modifier } }
                    add_modifier = { modifier = AIO_DAIT_spawn_modifier }
                }
            }
        }
    }
    else = {
        every_system_planet = {
            limit = {
                NOR = {
                    is_star = yes
                    is_colony = yes
                    is_artificial = yes
                    is_planet_class = pc_city
                    is_planet_class = pc_machine
                    is_planet_class = pc_hive
                    is_planet_class = pc_nanotech
                    AIO_safe_remove_planet_exclude_API = yes
                }
            }
            random_list = {
                2 = { AIO_safe_change_pc = { NEW_PC = pc_desert } }
                2 = { AIO_safe_change_pc = { NEW_PC = pc_arid } }
                2 = { AIO_safe_change_pc = { NEW_PC = pc_savannah } }
                2 = { AIO_safe_change_pc = { NEW_PC = pc_tropical } }
                2 = { AIO_safe_change_pc = { NEW_PC = pc_continental } }
                2 = { AIO_safe_change_pc = { NEW_PC = pc_ocean } }
                2 = { AIO_safe_change_pc = { NEW_PC = pc_tundra } }
                2 = { AIO_safe_change_pc = { NEW_PC = pc_arctic } }
                2 = { AIO_safe_change_pc = { NEW_PC = pc_alpine } }
                2 = { AIO_safe_change_pc = { NEW_PC = pc_volcanic } }
                1 = { AIO_safe_change_pc = { NEW_PC = pc_gaia } }
            }
        }
    }

    AIO_DAIT_notify_spawn_effect = yes
}

AIO_DAIT_harvest_effect = {
    owner? = { AIO_DAIT_mark_affected_country_effect = yes }
    every_system_ship_colony = { owner = { AIO_DAIT_mark_affected_country_effect = yes } }

    every_system_planet = {
        limit = {
            NOR = {
                is_planet_class = pc_shattered
                is_planet_class = pc_black_hole
                is_planet_class = pc_AIO_empty_space
                has_carrier_flag = synth_queen_bastille
                AIO_safe_remove_planet_exclude_API = yes
                has_planet_flag = AIO_DAIT_harvested_body
            }
        }
        if = {
            limit = { is_star = yes }
            root = { add_resource = { energy = 2000 mult = prev.trigger:planet_size } }
            change_pc = pc_t_star
            solar_system = { AIO_change_star_class = yes }
            set_planet_flag = AIO_DAIT_harvested_body
        }
        else_if = {
            limit = { is_artificial = no }
            if = {
                limit = {
                    OR = {
                        is_planet_class = pc_barren
                        is_planet_class = pc_barren_cold
                        is_planet_class = pc_asteroid
                        is_planet_class = pc_crystal_asteroid
                        is_planet_class = pc_broken
                    }
                }
                root = { add_resource = { minerals = 400 mult = prev.trigger:planet_size } }
            }
            else_if = {
                limit = {
                    OR = {
                        is_planet_class = pc_gas_giant
                        is_planet_class = pc_toxic
                    }
                }
                root = {
                    add_resource = { energy = 200 mult = prev.trigger:planet_size }
                    add_resource = { exotic_gases = 100 mult = prev.trigger:planet_size }
                }
            }
            else_if = {
                limit = {
                    OR = {
                        is_planet_class = pc_molten
                        is_planet_class = pc_volcanic
                    }
                }
                root = {
                    add_resource = { energy = 200 mult = prev.trigger:planet_size }
                    add_resource = { volatile_motes = 50 mult = prev.trigger:planet_size }
                }
            }
            else_if = {
                limit = { is_planet_class = pc_frozen }
                root = {
                    add_resource = { minerals = 200 mult = prev.trigger:planet_size }
                    add_resource = { rare_crystals = 50 mult = prev.trigger:planet_size }
                }
            }
            else_if = {
                limit = { is_planet_class = pc_shrouded }
                root = { add_resource = { sr_zro = 200 mult = prev.trigger:planet_size } }
            }
            else_if = {
                limit = { is_planet_class = pc_nanotech }
                root = { add_resource = { nanites = 200 mult = prev.trigger:planet_size } }
            }
            else_if = {
                limit = { is_colonizable = yes }
                root = {
                    add_resource = { minerals = 200 mult = prev.trigger:planet_size }
                    add_resource = { food = 250 mult = prev.trigger:planet_size }
                }
            }
            else = {
                limit = { is_planet_class = pc_ai }
                # TODO
            }
            else = {
                root = { add_resource = { minerals = 200 mult = prev.trigger:planet_size } }
            }
            clear_deposits = yes
            clear_planet_modifiers = yes
            if = {
                limit = { is_colony = yes }
                destroy_colony = yes
            }
            AIO_safe_change_pc = { NEW_PC = pc_shattered }
            set_planet_flag = AIO_DAIT_harvested_body
        }
        else = {
            switch = {
                trigger = is_planet_class
                pc_habitat = {
                    root = { add_resource = { alloys = 1000 } }
                    AIO_safe_remove_habitat = yes
                }
                pc_ringworld_habitable = {
                    root = { add_resource = { alloys = 8000 } }
                    AIO_safe_remove_rw_section = yes
                }
                pc_cosmogenesis_world = {
                    root = { add_resource = { alloys = 10000 } }
                    AIO_safe_remove_cosmogenesis_world = yes
                }
                pc_ark = {
                    root = { add_resource = { alloys = 5000 } }
                    ship = { fleet = { destroy_fleet = this } }
                }
            }
        }
    }
    set_star_flag = AIO_DAIT_harvest_system

    AIO_DAIT_notify_harvest_effect = yes
}

        }
    }
