AIO_DAIT_harvest_effect = {
    inline_script = AIO/AIO_prepare_destroy_system
    every_system_planet = {
        limit = {
            NOR = {
                is_planet_class = pc_shattered
                is_planet_class = pc_black_hole
                is_planet_class = pc_AIO_empty_space
                has_carrier_flag = synth_queen_bastille
                AIO_safe_remove_planet_exclude_API = yes
                has_planet_flag = AIO_DAIT_harvested_body
            }
        }
        if = {
            limit = { is_star = yes }
            event_target:AIO_DAIT_maker = {
                add_resource = { energy = 2000 mult = prev.trigger:planet_size }
            }
            change_pc = pc_t_star
            solar_system = { AIO_change_star_class = yes }
            hidden_effect = { set_planet_flag = AIO_DAIT_harvested_body }
        }
        else_if = {
            limit = { is_artificial = no }

                limit = {
                    OR = {
                        is_planet_class = pc_barren
                        is_planet_class = pc_barren_cold

                        is_planet_class = pc_crystal_asteroid

                        is_planet_class = pc_gas_giant
                        is_planet_class = pc_toxic
                    }
                }

                    add_resource = { energy = 200 mult = prev.trigger:planet_size }
                    add_resource = { exotic_gases = 100 mult = prev.trigger:planet_size }
                }
            }

                limit = {
                    OR = {
                        is_planet_class = pc_molten
                        is_planet_class = pc_volcanic
                    }
                }
                event_target:AIO_DAIT_maker = {
                    add_resource = { energy = 200 mult = prev.trigger:planet_size }
                    add_resource = { volatile_motes = 50 mult = prev.trigger:planet_size }

            }
            else_if = {
                limit = { is_planet_class = pc_frozen }

                    add_resource = { minerals = 200 mult = prev.trigger:planet_size }
                    add_resource = { rare_crystals = 50 mult = prev.trigger:planet_size }

                limit = { is_planet_class = pc_shrouded }

# System: caller has already established that this is an erased system.
# Use the system's position, independent of the just-replaced star entity.
AIO_DAIT_create_void_anchor = {
    if = {
        limit = {
            NOT = { any_system_ambient_object = { has_ambient_object_flag = AIO_DAIT_void_anchor } }
        }
        create_ambient_object = {
            type = AIO_DAIT_void_anchor
            location = this
            effect = {
                set_ambient_object_flag = AIO_DAIT_void_anchor
                # Creation uses a default placement offset; pin the final position.
                set_location = { target = solar_system distance = 0 angle = 0 }
            }
        }
    }
}

# System: discover old empty systems on load; not used by the live erase path.
AIO_DAIT_ensure_void_anchor = {
    if = {
        limit = {
            is_star_class = sc_AIO_void
            AIO_DAIT_system_valid = yes
            NOT = { any_system_planet = { NOT = { is_planet_class = pc_AIO_empty_space } } }
        }
        AIO_DAIT_create_void_anchor = yes
        every_system_ambient_object = {
            limit = { has_ambient_object_flag = AIO_DAIT_void_anchor }
            set_location = { target = solar_system distance = 0 angle = 0 }
        }
    }
}
