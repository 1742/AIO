# Country: restore the launcher once, without disturbing an already open menu.
# Used after host closure, on load, megastructure completion and ascension.
AIO_DAIT_ensure_menu = {
    if = {
        limit = {
            is_ai = no
            AIO_DAIT_enrich_unlocked = yes
            NOT = { has_country_flag = AIO_DAIT_toolbar }
        }
        country_event = { id = AIO_DAIT_events.203 }
    }
}

# Country: retain the selected multiplier across operation/menu changes. Unset means 1.
AIO_DAIT_cycle_enrich_multiplier = {
    if = {
        limit = { check_variable = { which = AIO_DAIT_enrich_multiplier value = 5 } }
        set_variable = { which = AIO_DAIT_enrich_multiplier value = 10 }
    }
    else_if = {
        limit = { check_variable = { which = AIO_DAIT_enrich_multiplier value = 10 } }
        clear_variable = AIO_DAIT_enrich_multiplier
    }
    else = {
        set_variable = { which = AIO_DAIT_enrich_multiplier value = 5 }
    }
}

# Country: choose an operation and reset its range.
AIO_DAIT_select_operation = {
    set_variable = { which = AIO_DAIT_operation value = $OPERATION$ }
    set_variable = { which = AIO_DAIT_range value = 0 }
}
