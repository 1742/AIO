# AIO safe removal effects

# Remove one megastructure without leaving carrier flags, deposits, modifiers, or
# build-limit bookkeeping behind. Scope: megastructure.
AIO_safe_remove_megastructure = {
    optimize_memory

    # The archive owns additional vivarium and specimen state. Its vanilla
    # system event removes every archive in the system and performs that cleanup.
    if = {
        limit = { is_megastructure_type = grand_archive_0 }
        solar_system = { system_event = { id = grand_archive.4501 } }
    }
    else = {
        if = {
            limit = { AIO_is_dyson_swarm = yes }
            planet = {
                remove_modifier = dyson_swarm_1_mod
                remove_modifier = dyson_swarm_2_mod
                remove_modifier = dyson_swarm_3_mod
            }
            if = {
                limit = { exists = owner }
                owner = {
                    if = {
                        limit = { check_variable = { which = dyson_swarm_counter value >= 1 } }
                        change_variable = { which = dyson_swarm_counter value = -1 }
                    }
                }
            }
        }
        else_if = {
            limit = { AIO_is_orbital_arc_furnace = yes }
            solar_system = {
                every_system_planet = {
                    limit = {
                        OR = {
                            has_modifier = orbital_arc_furnace_1_mod
                            has_modifier = orbital_arc_furnace_2_mod
                            has_modifier = orbital_arc_furnace_3_mod
                            has_modifier = orbital_arc_furnace_4_mod
                        }
                    }
                    dismantle_arc_furnace_effect = yes
                }
            }
            planet = {
                set_planet_entity = { entity = "pc_molten" }
                remove_planet_flag = has_arc_furnace
            }
            if = {
                limit = { exists = owner }
                owner = {
                    if = {
                        limit = { check_variable = { which = arc_furnace_counter value >= 1 } }
                        change_variable = { which = arc_furnace_counter value = -1 }
                    }
                }
            }
        }
        else_if = {
            limit = { is_megastructure_type = shroud_seal }
            solar_system = { on_shroud_seal_destroyed = yes }
        }

        # Clear per-country build limits only when this is the country's last
        # megastructure in that family. The current structure is still counted.
        if = {
            limit = {
                exists = owner
                AIO_is_dyson_sphere = yes
            }
            owner = {
                if = {
                    limit = {
                        count_owned_megastructure = {
                            limit = { AIO_is_dyson_sphere = yes }
                            count <= 1
                        }
                    }
                    remove_country_flag = built_dyson_sphere
                    remove_country_flag = built_dyson_sphere_contingency
                    remove_country_flag = built_dyson_sphere_swarm
                    remove_country_flag = built_dyson_sphere_contingency_swarm
                }
            }
        }
        else_if = {
            limit = {
                exists = owner
                AIO_is_matter_decompressor = yes
            }
            owner = {
                if = {
                    limit = {
                        count_owned_megastructure = {
                            limit = { AIO_is_matter_decompressor = yes }
                            count <= 1
                        }
                    }
                    remove_country_flag = built_matter_decompressor_site
                    remove_country_flag = built_matter_decompressor_site_contingency
                }
            }
        }
        else_if = {
            limit = {
                exists = owner
                AIO_is_mega_shipyard = yes
            }
            owner = {
                if = {
                    limit = {
                        count_owned_megastructure = {
                            limit = { AIO_is_mega_shipyard = yes }
                            count <= 1
                        }
                    }
                    remove_country_flag = built_mega_shipyard_site
                    remove_country_flag = built_mega_shipyard_site_contingency
                }
            }
        }
        else_if = {
            limit = {
                exists = owner
                AIO_is_quantum_catapult = yes
            }
            owner = {
                if = {
                    limit = {
                        count_owned_megastructure = {
                            limit = { AIO_is_quantum_catapult = yes }
                            count <= 1
                        }
                    }
                    remove_country_flag = built_quantum_catapult_site
                    remove_country_flag = built_quantum_catapult_site_contingency
                }
            }
        }
        else_if = {
            limit = {
                exists = owner
                AIO_is_spy_orb = yes
            }
            owner = {
                if = {
                    limit = {
                        count_owned_megastructure = {
                            limit = { AIO_is_spy_orb = yes }
                            count <= 1
                        }
                    }
                    remove_country_flag = built_spyorb
                    remove_country_flag = built_spyorb_contingency
                }
            }
        }
        else_if = {
            limit = {
                exists = owner
                AIO_is_think_tank = yes
            }
            owner = {
                if = {
                    limit = {
                        count_owned_megastructure = {
                            limit = { AIO_is_think_tank = yes }
                            count <= 1
                        }
                    }
                    remove_country_flag = built_think_tank
                    remove_country_flag = built_think_tank_contingency
                }
            }
        }
        else_if = {
            limit = {
                exists = owner
                AIO_is_strategic_coordination_center = yes
            }
            owner = {
                if = {
                    limit = {
                        count_owned_megastructure = {
                            limit = { AIO_is_strategic_coordination_center = yes }
                            count <= 1
                        }
                    }
                    remove_country_flag = built_strategic_coordination_center_site
                    remove_country_flag = built_strategic_coordination_center_site_contingency
                }
            }
        }
        else_if = {
            limit = {
                exists = owner
                AIO_is_mega_art_installation = yes
            }
            owner = {
                if = {
                    limit = {
                        count_owned_megastructure = {
                            limit = { AIO_is_mega_art_installation = yes }
                            count <= 1
                        }
                    }
                    remove_country_flag = built_mega_art_installation
                    remove_country_flag = built_mega_art_installation_contingency
                }
            }
        }
        else_if = {
            limit = {
                exists = owner
                AIO_is_interstellar_assembly = yes
            }
            owner = {
                if = {
                    limit = {
                        count_owned_megastructure = {
                            limit = { AIO_is_interstellar_assembly = yes }
                            count <= 1
                        }
                    }
                    remove_country_flag = built_interstellar_assembly_site
                    remove_country_flag = built_interstellar_assembly_site_contingency
                }
            }
        }
        else_if = {
            limit = {
                exists = owner
                OR = {
                    is_dyson_gun = yes
                    is_megastructure_type = dyson_gun_ruined
                }
            }
            owner = {
                if = {
                    limit = {
                        count_owned_megastructure = {
                            limit = {
                                OR = {
                                    is_dyson_gun = yes
                                    is_megastructure_type = dyson_gun_ruined
                                }
                            }
                            count <= 1
                        }
                    }
                    remove_country_flag = built_dyson_gun_site
                    remove_country_flag = built_dyson_gun_site_contingency
                }
            }
            solar_system = { remove_star_flag = dyson_gun_built }
        }
        else_if = {
            limit = {
                exists = owner
                AIO_is_behemoth_mega = yes
            }
            owner = {
                remove_country_flag = behemoth_egg_built
            }
        }

        if = {
            limit = {
                OR = {
                    AIO_is_mega_art_installation = yes
                    AIO_is_dyson_sphere_disco = yes
                }
            }
            solar_system = { remove_star_flag = mega_art_installation_built }
        }

        if = {
            limit = { exists = planet }
            planet = {
                remove_carrier_flag = has_megastructure
                remove_carrier_flag = megastructure
            }
        }

        remove_megastructure = this
    }
}

# Remove every removable megastructure attached to this planet. Scope: planet.
AIO_safe_remove_planet_megastructures = {
    save_event_target_as = AIO_safe_remove_mega_planet

    if = {
        limit = {
            is_colony = yes
            AIO_planet_has_oribital_ring = yes
        }
        solar_system = {
            random_starbase_in_system = {
                limit = {
                    exists = planet
                    planet = { is_same_value = event_target:AIO_safe_remove_mega_planet }
                    OR = {
                        has_starbase_size = orbital_ring_tier_1
                        has_starbase_size = orbital_ring_tier_2
                        has_starbase_size = orbital_ring_tier_3
                    }
                }
                destroy_fleet = this
            }
        }
    }

    solar_system = {
        every_system_megastructure = {
            limit = {
                AIO_is_AIO_mega = no
                AIO_safe_remove_megastructure_exclude_API = no
                exists = planet
                planet = { is_same_value = event_target:AIO_safe_remove_mega_planet }
            }
            AIO_safe_remove_megastructure = yes
        }
    }
}

# Remove every permitted megastructure in this system. Scope: system.
AIO_safe_remove_system_megastructures = {
    every_system_megastructure = {
        limit = {
            AIO_is_AIO_mega = no
            AIO_safe_remove_megastructure_exclude_API = no
        }
        AIO_safe_remove_megastructure = yes
    }
}

# 安全移除轨道居住站，scope: planet
AIO_safe_remove_habitat = {
    solar_system = {
        every_system_megastructure = {
            limit = {
                OR = {
                    is_megastructure_type = habitat_major_orbital
                    is_megastructure_type = habitat_minor_orbital
                }
            }
            remove_megastructure = this
        }
        every_system_planet = {
            limit = { has_planet_flag = has_orbital }
            remove_planet_flag = has_orbital
        }
        if = {
            limit = { has_star_flag = has_habitat }
            remove_star_flag = has_habitat
        }
    }
    if = {
        limit = { is_colony = yes }
        destroy_colony = yes
    }
    remove_planet = yes
}

# 安全移除环形世界区段，scope: planet
AIO_safe_remove_rw_section = {
    optimize_memory

    solar_system = {
        if = {
            limit = {
                NOT = { has_megastructure = ring_world_2 }
                count_system_planet = {
                    limit = {
                        OR = {
                            is_planet_class = pc_ringworld_habitable
                            is_planet_class = pc_shattered_ring_habitable
                        }
                    }
                    count = 1
                }
            }
            star = {
                if = {
                    limit = { has_planet_flag = has_megastructure }
                    remove_planet_flag = has_megastructure
                }
            }
            if = {
                limit = { has_star_flag = ring_world_built }
                remove_star_flag = ring_world_built
            }
        }
    }
    if = {
        limit = { is_colony = yes }
        destroy_colony = yes
    }
    remove_planet = yes
}

# 安全移除突出凝炼机，scope: planet
AIO_safe_remove_cosmogenesis_world = {
    optimize_memory

    if = {
        limit = { is_planet_class = pc_cosmogenesis_world }
        owner = { remove_country_flag = cosmogenesis_world_built }
        destroy_colony = yes
        remove_planet = yes
    }
}

# 移除星球及其上巨构，scope: planet
AIO_safe_remove_planet = {
    optimize_memory

    if = {
        limit = {
            AIO_safe_remove_planet_exclude_API = no
            # 以太相引擎和星系熔炉星系内的天体不受影响
            AIO_is_crisis_mega_system = no
        }

        if = {
            limit = {
                is_star = yes
                NOT = { is_planet_class = pc_AIO_empty_space }
            }
            change_pc = pc_AIO_empty_space
            clear_deposits = yes
            AIO_safe_remove_planet_megastructures = yes
        }
        else_if = {
            limit = {
                is_colony = yes
                is_artificial = no
                NOR = {
                    is_planet_class = pc_ai
                    has_carrier_flag = synth_queen_bastille
                }
            }
            destroy_colony = yes
            AIO_safe_remove_planet_megastructures = yes
            remove_planet = yes
        }
        else_if = {
            limit = { is_planet_class = pc_ark }
            ship = {
                fleet = { destroy_fleet = this }
            }
        }
        else_if = {
            limit = { is_planet_class = pc_habitat }
            AIO_safe_remove_habitat = yes
        }
        else_if = {
            limit = {
                OR = {
                    is_planet_class = pc_ringworld_habitable
                    is_planet_class = pc_shattered_ring_habitable
                }
            }
            AIO_safe_remove_rw_section = yes
        }
        else_if = {
            limit = { is_planet_class = pc_cosmogenesis_world }
            AIO_safe_remove_cosmogenesis_world = yes
        }
        else_if = {
            # 潜在的肃正球
            limit = {
                AIO_is_ai_planet = yes
                NOT = { is_planet_class = pc_ai }
            }
            switch = {
                trigger = has_planet_flag
                machine_world_1 = {
                    remove_planet_flag = machine_world_1
                    remove_planet_flag = crisis_vital_planet
                    solar_system = {
                        remove_star_flag = AI_hub
                        remove_star_flag = AI_system_1
                    }
                    random_galaxy_planet = {
                        limit = {
                            is_star = no
                            is_artificial = no
                            AIO_is_ai_planet = no
                        }
                        set_planet_flag = machine_world_1
                        set_planet_flag = crisis_vital_planet
                        solar_system = {
                            set_star_flag = AI_hub
                            set_star_flag = AI_system_1
                        }
                    }
                }
                machine_world_2 = {
                    remove_planet_flag = machine_world_2
                    remove_planet_flag = crisis_vital_planet
                    solar_system = {
                        remove_star_flag = AI_hub
                        remove_star_flag = AI_system_2
                    }
                    random_galaxy_planet = {
                        limit = {
                            is_star = no
                            is_artificial = no
                            AIO_is_ai_planet = no
                        }
                        set_planet_flag = machine_world_2
                        set_planet_flag = crisis_vital_planet
                        solar_system = {
                            set_star_flag = AI_hub
                            set_star_flag = AI_system_2
                        }
                    }
                }
                machine_world_3 = {
                    remove_planet_flag = machine_world_3
                    remove_planet_flag = crisis_vital_planet
                    solar_system = {
                        remove_star_flag = AI_hub
                        remove_star_flag = AI_system_3
                    }
                    random_galaxy_planet = {
                        limit = {
                            is_star = no
                            is_artificial = no
                            AIO_is_ai_planet = no
                        }
                        set_planet_flag = machine_world_3
                        set_planet_flag = crisis_vital_planet
                        solar_system = {
                            set_star_flag = AI_hub
                            set_star_flag = AI_system_3
                        }
                    }
                }
                machine_world_4 = {
                    remove_planet_flag = machine_world_4
                    remove_planet_flag = crisis_vital_planet
                    solar_system = {
                        remove_star_flag = AI_hub
                        remove_star_flag = AI_system_4
                    }
                    random_galaxy_planet = {
                        limit = {
                            is_star = no
                            is_artificial = no
                            AIO_is_ai_planet = no
                        }
                        set_planet_flag = machine_world_4
                        set_planet_flag = crisis_vital_planet
                        solar_system = {
                            set_star_flag = AI_hub
                            set_star_flag = AI_system_4
                        }
                    }
                }
            }
        }
        else_if = {
            limit = {
                is_planet_class = pc_ai
                NOT = { has_carrier_flag = machine_lair }
            }
            # TODO: 处理肃正球被移除
            # planet_event = { id = AIO_ctrl_events.1007 }
            remove_planet = yes
        }
        else_if = {
            limit = {
                is_planet_class = pc_ai
                has_carrier_flag = machine_lair
            }
            root = { save_event_target_as = final_machine_world_destroyer }
            stop_crisis_sound = yes
            carrier_event = { id = crisis.2046 }
            remove_planet = yes
        }
        else_if = {
            limit = { NOT = { has_carrier_flag = synth_queen_bastille } }
            AIO_safe_remove_planet_megastructures = yes
            remove_planet = yes
        }
    }
}
