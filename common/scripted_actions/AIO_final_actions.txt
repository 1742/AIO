AIO_final_induction_action = {
    user_scope = megastructure
    scope = galactic_object
    possible = {
        custom_tooltip = {
            fail_text = AIO_final_fail.target
            AIO_final_valid_external_target = yes
        }
    }
    button_clickable = {
        owner = {
            custom_tooltip = {
                fail_text = AIO_final_fail.unavailable
                AIO_final_can_select = yes
            }
        }
    }
    button_visible = {
        is_megastructure_type = AIO_mega_04
        owner = {
            has_country_flag = AIO_final_start
            check_variable = { which = AIO_final_research value >= 3 }
            check_variable = { which = AIO_final_research value < 7 }
        }
    }
    icon = GFX_ship_class_conduit
    icon_selected = GFX_ship_class_conduit
    on_started = on_AIO_final_induction
    context_menu_name = AIO_final_induction_action
    activity_key = AIO_final_induction_activity
    tooltip = AIO_final_induction_action_tooltip
}
