AIO_DAIT_enrich_action = {
    user_scope = megastructure
    scope = galactic_object
    possible = {
        AIO_DAIT_enrich_target_valid = yes
        NOT = { has_star_flag = AIO_DAIT_center_busy }
    }
    button_clickable = { owner = { AIO_DAIT_enrich_ready = yes } }
    button_visible = { owner = { AIO_DAIT_enrich_unlocked = yes } }
    icon = GFX_ship_class_conduit
    icon_selected = GFX_ship_class_conduit
    on_started = on_AIO_DAIT_enrich
    context_menu_name = AIO_DAIT_button_enrich
    activity_key = AIO_DAIT_button_enrich
    tooltip = AIO_DAIT_action_enrich_tooltip
}
