# The mist has its own urban district because vanilla cities exclude this district set.
# Districts supply housing; their specializations supply all workforce.
# Keep the vanilla default has_primary_zone = yes for single-slot production layouts.
# The colony class selects its initial district through starting_district.
@AIO_mist_district_alloys_cost = 200
@AIO_mist_district_energy_upkeep = 4
@AIO_mist_district_housing_1 = 800
@AIO_mist_district_housing_2 = 200
@AIO_mist_district_buildtime = 600


district_AIO_mist_city = {
    icon = district_AIO_mist_city
    gridbox = district_city
    district_background = district_AIO_mist_plain
    base_buildtime = @AIO_mist_district_buildtime
    is_uncapped = { always = yes }
    has_primary_zone = yes
    zone_slots = {
        slot_AIO_mist_government
        slot_city_01
        slot_city_02
    }
    show_on_uncolonized = { uses_district_set = AIO_mist }
    potential = { uses_district_set = AIO_mist }
    resources = {
        category = planet_districts_cities
        cost = { alloys = @AIO_mist_district_alloys_cost }
        upkeep = { energy = @AIO_mist_district_energy_upkeep }
    }
    planet_modifier = { planet_housing_add = 1000 }
    ai_weight = { weight = 1 }
}

district_AIO_mist_research = {
    icon = district_AIO_mist_research
    gridbox = district_rw_science
    district_background = district_AIO_mist_plain
    base_buildtime = @AIO_mist_district_buildtime
    is_uncapped = { always = yes }
    zone_slots = { slot_AIO_lathe }
    show_on_uncolonized = { uses_district_set = AIO_mist }
    potential = { uses_district_set = AIO_mist }

    resources = {
        category = planet_districts
        cost = { alloys = @AIO_mist_district_alloys_cost }
        # Construction cost triggers use the purchasing country (from).
        cost = {
            trigger = { from? = { is_virtual_empire = yes } }
            alloys = 800
        }
        upkeep = { energy = @AIO_mist_district_energy_upkeep }
        # Double this district's base upkeep for virtual owners, without affecting other districts.
        upkeep = {
            trigger = { owner = { is_virtual_empire = yes } }
            energy = @AIO_mist_district_energy_upkeep
        }
    }
    planet_modifier = {
        planet_housing_add = @AIO_mist_district_housing_1
    }
    ai_weight = { weight = 1 }
}
