# Urban infrastructure follows vanilla city / hive / nexus districts.
# Resource districts grant their base workforce directly; specializations add native jobs/buildings.
@AIO_mist_district_alloys_cost = 200
@AIO_mist_district_energy_upkeep = 2
@AIO_mist_district_jobs = 800
# Applied to each of the three research job types.
@AIO_mist_district_research_jobs = 200
@AIO_mist_district_buildtime = 360


district_AIO_mist_research = {
    icon = district_hab_science
    base_buildtime = @AIO_mist_district_buildtime
    is_uncapped = { always = yes }
    zone_slots = { slot_AIO_mist_research }
    show_on_uncolonized = { uses_district_set = AIO_mist }
    potential = { uses_district_set = AIO_mist }
    
    resources = {
        category = planet_districts
        cost = { alloys = @AIO_mist_district_alloys_cost }
        upkeep = { energy = @AIO_mist_district_energy_upkeep }
    }
    planet_modifier = { planet_housing_add = @AIO_mist_district_jobs }
    inline_script = { script = jobs/researchers_add AMOUNT = @AIO_mist_district_research_jobs }
    ai_weight = { weight = 1 }
}
