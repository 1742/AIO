AIO_mist_initializer = {
    name = AIO_mist
    class = sc_AIO_void
    usage = misc_system_init
    usage_odds = 0

    planet = {
		count = 1
		class = pc_AIO_empty_space
		orbit_distance = 0
		orbit_angle = 0
		size = 0
		has_ring = no

        init_effect = {
            set_name = AIO_mist
            save_event_target_as = AIO_mist_star
        }
	}

    init_effect = {
        if = {
            limit = { root = { is_nomadic = no } }
            create_starbase = {
                size = starbase_outpost
                owner = root
            }
        }
        else = {
            create_waystation_effect = {
                OWNER = root
                ANGLE = 0
            }
        }

        spawn_megastructure = {
            type = AIO_mega_01
            planet = event_target:AIO_mist_star
            orbit_angle = 0
            orbit_distance = 0
            owner = root
        }

        set_star_flag = AIO_mist_system
        save_global_event_target_as = AIO_mist_system
    }
}
