AIO_DAIT_enrich = {
    potential = { has_active_event = { AIO_DAIT_events.202 } }
    allow = {
        custom_tooltip = {
            fail_text = AIO_DAIT.fail.mega_1
            OR = {
                has_menace_perk = menp_AIO_mega_02
                is_country_type = AIO_empire
            }
        }
        custom_tooltip = {
            fail_text = AIO_DAIT_lock_target.fail.enrich
            event_target:AIO_DAIT_target_system = {
                any_system_planet = {
                    NOT = {
                        is_planet_class = pc_shattered
                        is_planet_class = pc_AIO_empty_space
                        has_planet_flag = AIO_DAIT_harvested_body
                    }
                }
            }
        }
        OR = {
            is_country_type = AIO_empire
            resource_stockpile_compare = { resource = AIO_advanced_logic value >= @AIO_DAIT_cost }
        }
    }
    effect = {
        AIO_DAIT_clear_affected_countries_effect = yes
        AIO_DAIT_pay_cost_effect = yes
        AIO_DAIT_register_construct_effect = yes
        event_target:AIO_DAIT_target_system = { AIO_DAIT_enrich_effect = { TIMES = 10 } }
        AIO_DAIT_notify_enrich_effect = yes
        AIO_DAIT_clear_affected_countries_effect = yes
    }
}
