    potential = {
        from = { AIO_DAIT_menu_visible = yes }

    allow = {
        from = {
            custom_tooltip = {
                fail_text = AIO_DAIT_need_unlock
                AIO_DAIT_enrich_unlocked = yes
            }
        }
    }
    effect = {
        custom_tooltip = AIO_DAIT_button_enrich_tooltip
        hidden_effect = {
            from = {
                if = {
                    limit = { AIO_DAIT_enrich_unlocked = yes }
                    AIO_DAIT_select_operation = { OPERATION = 1 }
                }
            }
        }
    }
}

AIO_DAIT_confirm = {
    potential = {
        from = {
            AIO_DAIT_menu_visible = yes
            AIO_DAIT_menu_operation_selected = yes
        }
    }
    allow = {
        custom_tooltip = {
            fail_text = AIO_DAIT_unavailable
            from = { AIO_DAIT_menu_selected_ready = yes }
        }
        custom_tooltip = {
            fail_text = AIO_DAIT_menu_invalid_target
            AIO_DAIT_menu_selected_target_valid = yes
        }
    }
    effect = {
        hidden_effect = {
            if = {
                limit = {
                    from = { AIO_DAIT_menu_selected_ready = yes }
                    AIO_DAIT_menu_selected_target_valid = yes
                }
                if = {
                    limit = { from = { check_variable = { which = AIO_DAIT_operation value = 1 } } }
                    AIO_DAIT_capture_target = yes
                    event_target:AIO_DAIT_maker = { AIO_DAIT_start_enrich = yes }
                }
                else_if = {
                    limit = { from = { check_variable = { which = AIO_DAIT_operation value = 2 } } }
                    AIO_DAIT_capture_target = yes
                    event_target:AIO_DAIT_maker = { AIO_DAIT_start_spawn = yes }
                }
                else_if = {
                    limit = { from = { check_variable = { which = AIO_DAIT_operation value = 3 } } }
                    AIO_DAIT_capture_target = yes
                    event_target:AIO_DAIT_maker = { AIO_DAIT_start_seed = yes }
                }
                else_if = {
                    limit = { from = { check_variable = { which = AIO_DAIT_operation value = 4 } } }
                    AIO_DAIT_capture_country_target = yes
                    event_target:AIO_DAIT_maker = { AIO_DAIT_start_ethic = yes }
                }
                else_if = {
                    limit = { from = { check_variable = { which = AIO_DAIT_operation value = 5 } } }
                    AIO_DAIT_capture_target = yes
                    event_target:AIO_DAIT_maker = { AIO_DAIT_start_storm = yes }
                }
                else_if = {
                    limit = { from = { check_variable = { which = AIO_DAIT_operation value = 6 } } }
                    AIO_DAIT_capture_target = yes
                    event_target:AIO_DAIT_maker = { AIO_DAIT_start_harvest = yes }
                }
                else_if = {
                    limit = { from = { check_variable = { which = AIO_DAIT_operation value = 7 } } }
                    AIO_DAIT_capture_target = yes
                    event_target:AIO_DAIT_maker = { AIO_DAIT_start_strike = yes }
                }
                else_if = {
                    limit = { from = { check_variable = { which = AIO_DAIT_operation value = 8 } } }
                    AIO_DAIT_capture_target = yes
                    event_target:AIO_DAIT_maker = { AIO_DAIT_start_erase = yes }
                }
            }
        }
    }
}

AIO_DAIT_final_induction = {
    potential = {
        from = {
            AIO_DAIT_menu_visible = yes
            has_country_flag = AIO_final_start
            check_variable = { which = AIO_final_research value >= 2 }
            check_variable = { which = AIO_final_research value < 3 }
        }
    }
    allow = {
        from = {
            AIO_final_can_select = yes

            any_owned_megastructure = { is_megastructure_type = AIO_mega_04 }
        }
        OR = {
            AND = { is_scope_type = galactic_object AIO_final_valid_external_target = yes }
            AND = { is_scope_type = planet solar_system = { AIO_final_valid_external_target = yes } }
        }
    }
    effect = {
        hidden_effect = {
            AIO_DAIT_capture_target = yes
            from = {
                random_owned_megastructure = {
                    limit = { is_megastructure_type = AIO_mega_04 }
                    save_event_target_as = AIO_DAIT_source_mega
                }
            }
            event_target:AIO_DAIT_target_system = {
                fire_on_action = {
                    on_action = on_AIO_final_induction
                    scopes = { from = event_target:AIO_DAIT_source_mega }
                }
            }
        }
    }
}

# The orb toggles the panel; the upper-right button exits the event host.
AIO_DAIT_toggle_menu = {
    potential = {
        from = {
            AIO_DAIT_country_valid = yes
            has_country_flag = AIO_DAIT_toolbar
            NOT = { has_country_flag = AIO_DAIT_dismissed }
        }
    }
    allow = { always = yes }
    effect = {
        custom_tooltip = AIO_DAIT_launcher_tooltip
        hidden_effect = {
            from = {
                if = {
                    limit = { has_country_flag = AIO_DAIT_collapsed }
                    remove_country_flag = AIO_DAIT_collapsed
                }
                else = { set_country_flag = AIO_DAIT_collapsed }
            }
        }
    }
}
