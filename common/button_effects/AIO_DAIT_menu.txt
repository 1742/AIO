# THIS = selected object and may change; FROM = player country.
# Keep potential anchored exclusively to the stable player-country scope.

AIO_DAIT_enrich = {
    potential = { from = { has_country_flag = AIO_DAIT_active } }
    allow = {
        custom_tooltip = {
            fail_text = AIO_DAIT.fail.require_mega_1
            from = {
                OR = {
                    has_country_flag = AIO_built_AIO_mega_01
                    hidden_trigger = { is_country_type = AIO_empire }
                }
            }
        }
        custom_tooltip = {
            fail_text = AIO_DAIT_lock_target.fail.enrich
            event_target:AIO_DAIT_target_system = {
                NOT = { has_star_flag = AIO_DAIT_harvest_system }
                any_system_planet = {
                    NOR = {
                        is_planet_class = pc_shattered
                        is_planet_class = pc_AIO_empty_space
                        has_planet_flag = AIO_DAIT_harvested_body
                    }
                }
            }
        }
        from = {
            OR = {
                hidden_trigger = { is_country_type = AIO_empire }
                resource_stockpile_compare = { resource = AIO_advanced_logic value >= @AIO_DAIT_cost }
            }
        }
    }
    effect = {
        from = { AIO_DAIT_manual_enrich_effect = yes }
    }
}
