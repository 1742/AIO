# Custom menu buttons: FROM is the player, THIS is the selected object.


AIO_DAIT_enrich = {
    potential = {
        from = {
            AIO_DAIT_country_valid = yes
            has_country_flag = AIO_DAIT_toolbar
        }
    }
    allow = {
        custom_tooltip = {
            fail_text = AIO_DAIT_need_target
            AIO_DAIT_ui_target_valid = yes
        }
        from = { AIO_DAIT_enrich_ready = yes }
        custom_tooltip = {
            fail_text = AIO_DAIT_invalid_target
            OR = {
                AND = {
                    is_scope_type = galactic_object
                    AIO_DAIT_enrich_target_valid = yes
                    NOT = { has_star_flag = AIO_DAIT_center_busy }
                }
                AND = {
                    is_scope_type = planet
                    solar_system = {
                        AIO_DAIT_enrich_target_valid = yes
                        NOT = { has_star_flag = AIO_DAIT_center_busy }
                    }
                }
            }
        }
    }
    effect = {
        if = {
            limit = {
                AIO_DAIT_ui_target_valid = yes
                from = { AIO_DAIT_enrich_ready = yes }
            }
            AIO_DAIT_capture_target = yes
            event_target:AIO_DAIT_maker = { AIO_DAIT_start_enrich = yes }
        }
    }
}

AIO_DAIT_final_induction = {
    potential = {
        from = {
            has_country_flag = AIO_DAIT_toolbar
            has_country_flag = AIO_final_start
            check_variable = { which = AIO_final_research value >= 2 }
            check_variable = { which = AIO_final_research value < 3 }
        }
    }
    allow = {
        from = {
            AIO_final_can_select = yes

            any_owned_megastructure = { is_megastructure_type = AIO_mega_04 }
        }
        OR = {
            AND = { is_scope_type = galactic_object AIO_final_valid_external_target = yes }
            AND = { is_scope_type = planet solar_system = { AIO_final_valid_external_target = yes } }
        }
    }
    effect = {
        AIO_DAIT_capture_target = yes
        from = {
            random_owned_megastructure = {
                limit = { is_megastructure_type = AIO_mega_04 }
                save_event_target_as = AIO_DAIT_source_mega
            }
        }
        event_target:AIO_DAIT_target_system = {
            fire_on_action = {
                on_action = on_AIO_final_induction
                scopes = { from = event_target:AIO_DAIT_source_mega }
            }
        }
    }
}
