# Mirrors the owner resolution used by the Ethics button.
defined_text = {
    random = no
    name = GetAIO_DAIT_selected_country
    text = {
        trigger = {
            is_scope_type = country
            NOT = { is_same_value = from }
        }
        localization_key = AIO_DAIT_target_country_name
    }
    text = {
        trigger = { AIO_DAIT_ui_owned_target = yes }
        localization_key = AIO_DAIT_target_owner_name
    }
    default = AIO_DAIT_target_country_none
}
