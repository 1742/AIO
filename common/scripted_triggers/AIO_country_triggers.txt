# 能应用无限的国家
AIO_can_AIT = {
    OR = {
        has_ascension_perk = ap_cosmogenesis
        has_ascension_perk = ap_AIO_allinone
    }
}

# scope: system
AIO_country_has_colony_in_system = {
    OR = {
        any_system_planet_colony = {
            owner = { is_same_value = $CURRENT$ }
        }
        any_system_ship_colony = {
            owner = { is_same_value = $CURRENT$ }
        }
    }
}
