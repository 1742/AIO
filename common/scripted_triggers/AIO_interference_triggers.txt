# Search system within owner's range, excluding self.
# Scope: system.
AIO_is_nearby_interference_source = {
	closest_system = {
		use_bypasses = no
		min_steps = 1
		max_steps = $OWNER$.AIO_interference_range
		valid = {
			has_psionic_aura = AIO_reality_interference_aura
			aura_owner = { is_same_value = $OWNER$ }
		}
		limit = { AIO_system_has_interference_source = { OWNER = $OWNER$ } }
	}
}

# scope: system
AIO_is_backlash_affect_system = {
	has_psionic_aura = AIO_reality_interference_aura
	is_aura_intensity_level = 3
	aura_owner = { is_same_value = $MAKER$ }
}

# scope: country
AIO_has_any_interference_source = {
	OR = {
		any_owned_megastructure = { AIO_is_interference_source = yes }
		any_owned_nonprimary_starbase = { AIO_is_interference_source = yes }
		any_owned_ship = { AIO_is_mobile_interference_source = yes }
	}
}

# Scope: megastructure or starbase. Query the actual installation, never a star flag.
AIO_is_interference_source = {
	if = {
		limit = { is_scope_type = megastructure }
		is_megastructure_type = AIO_reality_interferometer
	}
	else_if = {
		limit = { is_scope_type = starbase }
		OR = {
			has_starbase_module = AIO_arkship_interference_generator
			has_starbase_module = AIO_waystation_interference_generator
		}
	}
	else = { always = no }
}

# Scope: starbase. Compare aura ownership with this installation's owner,
# independently of who owns the system (nomads need not own it).
AIO_starbase_has_foreign_interference = {
	solar_system = {
		has_psionic_aura = AIO_reality_interference_aura
		aura_owner = {
			NOT = { is_same_value = prevprev.owner }
		}
	}
}

# Scope: ship.
AIO_is_mobile_interference_source = {
	is_arkship_ship = yes
	starbase? = { AIO_is_interference_source = yes }
}

# Scope: system. All sources are attributed to their installation's owner.
AIO_system_has_interference_source = {
	OR = {
		any_system_megastructure = {
			AIO_is_interference_source = yes
			owner = { is_same_value = $OWNER$ }
		}
		any_starbase_in_system = {
			is_waystation_starbase = yes
			AIO_is_interference_source = yes
			owner = { is_same_value = $OWNER$ }
		}
		any_ship_in_system = {
			AIO_is_mobile_interference_source = yes
			owner = { is_same_value = $OWNER$ }
		}
	}
}
