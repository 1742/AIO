# Country / system scopes are explicit; no shared global casting target.
AIO_DAIT_country_valid = {
    OR = {
        has_ascension_perk = ap_AIO_allinone
        is_country_type = AIO_empire
    }
}

AIO_DAIT_system_valid = {
    exists = star
    NOR = {
        has_star_flag = AIO_final_reserved
        has_star_flag = AIO_Aforgomon_existed
        has_star_flag = AIO_mist_system
        has_star_flag = AIO_gate_system
        has_star_flag = synth_queen_capital
        AIO_safe_remove_system_exclude_API = yes
        AIO_is_crisis_mega_system = yes
        any_system_planet = { AIO_safe_remove_planet_exclude_API = yes }
    }
}

AIO_DAIT_ui_target_valid = {
    OR = {
        AND = {
            is_scope_type = galactic_object
            AIO_DAIT_system_valid = yes
        }
        AND = {
            is_scope_type = planet
            solar_system = { AIO_DAIT_system_valid = yes }
        }
    }
}

AIO_DAIT_enrich_unlocked = {
    AIO_DAIT_country_valid = yes
    OR = {
        has_country_flag = AIO_built_AIO_mega_01
        is_country_type = AIO_empire
    }
}

AIO_DAIT_enrich_target_valid = {
    AIO_DAIT_system_valid = yes
    NOT = { has_star_flag = AIO_DAIT_harvest_system }
    any_system_planet = {
        NOR = {
            is_planet_class = pc_shattered
            is_planet_class = pc_AIO_empty_space
            has_planet_flag = AIO_DAIT_harvested_body
        }
    }
}

AIO_DAIT_enrich_range_valid = {
    OR = {
        NOT = { is_variable_set = AIO_DAIT_range }
        check_variable = { which = AIO_DAIT_range value <= 2 }
    }
}

AIO_DAIT_enrich_ready = {
    custom_tooltip = {
        fail_text = AIO_DAIT_need_unlock
        AIO_DAIT_enrich_unlocked = yes
    }
    custom_tooltip = {
        fail_text = AIO_DAIT_need_range
        AIO_DAIT_enrich_range_valid = yes
    }
    custom_tooltip = {
        fail_text = AIO_DAIT_wait_pending
        NOT = { has_country_flag = AIO_DAIT_enrich_pending }
    }
    custom_tooltip = {
        fail_text = AIO_DAIT_wait_cooldown
        NOT = { has_country_flag = AIO_DAIT_enrich_cooldown }
    }
    custom_tooltip = {
        fail_text = AIO_DAIT_need_logic
        OR = {
            is_country_type = AIO_empire
            resource_stockpile_compare = {
                resource = AIO_advanced_logic
                value >= value:AIO_DAIT_enrich_price
            }
        }
    }
}
