# Country / system scopes are explicit; no shared global casting target.
# Country: ascended empires ignore even cooldowns inherited before ascension.
AIO_DAIT_cooldown_active = {
    NOT = { is_country_type = AIO_empire }
    has_country_flag = AIO_DAIT_$OPERATION$_cooldown
}

AIO_DAIT_country_valid = {
    OR = {
        has_country_flag = AIO_built_AIO_mega_01
        has_country_flag = AIO_built_AIO_mega_02
        has_country_flag = AIO_built_AIO_mega_03
        has_country_flag = AIO_built_AIO_mega_04
        is_country_type = AIO_empire
    }
}

# Country: the shared selector is used only by operations supporting an area.
AIO_DAIT_range_valid = {
    OR = {
        NOT = { is_variable_set = AIO_DAIT_range }
        AND = {
            check_variable = { which = AIO_DAIT_range value >= 0 }
            check_variable = { which = AIO_DAIT_range value <= 5 }
        }
    }
}

# System: snapshot range on the caster, measured from the selected center.
# The center is included for both 0 jumps and the special Strike realms mode.
AIO_DAIT_system_in_range = {
    OR = {
        is_same_value = event_target:AIO_DAIT_target_system
        AND = {
            event_target:AIO_DAIT_maker = { check_variable = { which = $RANGE$ value = 1 } }
            distance = { source = event_target:AIO_DAIT_target_system max_jumps = 1 use_bypasses = no }
        }
        AND = {
            event_target:AIO_DAIT_maker = { check_variable = { which = $RANGE$ value = 2 } }
            distance = { source = event_target:AIO_DAIT_target_system max_jumps = 2 use_bypasses = no }
        }
        AND = {
            event_target:AIO_DAIT_maker = { check_variable = { which = $RANGE$ value = 3 } }
            distance = { source = event_target:AIO_DAIT_target_system max_jumps = 3 use_bypasses = no }
        }
        AND = {
            event_target:AIO_DAIT_maker = { check_variable = { which = $RANGE$ value = 4 } }
            distance = { source = event_target:AIO_DAIT_target_system max_jumps = 4 use_bypasses = no }
        }
        AND = {
            event_target:AIO_DAIT_maker = { check_variable = { which = $RANGE$ value = 5 } }
            distance = { source = event_target:AIO_DAIT_target_system max_jumps = 5 use_bypasses = no }
        }
    }
}

AIO_DAIT_system_valid = {
    exists = star
    NOR = {
        has_star_flag = AIO_final_reserved
        has_star_flag = AIO_Aforgomon_existed
        has_star_flag = AIO_mist_system
        has_star_flag = AIO_gate_system
        has_star_flag = synth_queen_capital
        AIO_safe_remove_system_exclude_API = yes
        AIO_is_crisis_mega_system = yes
        any_system_planet = { AIO_safe_remove_planet_exclude_API = yes }
    }
}

# Selected objects which expose a direct solar_system scope.
AIO_DAIT_ui_system_object = {
    exists = this
    OR = {
        is_scope_type = planet
        is_scope_type = colony
        is_scope_type = fleet
        is_scope_type = megastructure
        is_scope_type = ambient_object
    }
    exists = solar_system
}

AIO_DAIT_ui_target_valid = {
    OR = {
        AND = {
            is_scope_type = galactic_object
            AIO_DAIT_system_valid = yes
        }
        AND = {
            AIO_DAIT_ui_system_object = yes
            solar_system = { AIO_DAIT_system_valid = yes }
        }
        AND = {
            is_scope_type = ship
            exists = fleet
            fleet = { solar_system = { AIO_DAIT_system_valid = yes } }
        }
    }
}

        }
