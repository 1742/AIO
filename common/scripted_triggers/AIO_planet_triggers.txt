        is_artificial = no
    }
}

# 判断行星是否有星环基地，scope: planet
AIO_colony_has_ring = {
    orbital_defence? = {
        starbase = { is_orbital_ring = yes }
    }
}

        is_artificial = no
        is_colony = yes
        AIO_colony_has_ring = no
    }
