# scope：country
AIO_projection_can_start = {
	custom_tooltip = {
		fail_text = AIO_projection_fail.need_final
		OR = {
			has_country_flag = AIO_final_won
			is_country_type = AIO_empire
		}
	}
	custom_tooltip = {
		fail_text = AIO_projection_fail.need_mega_04
		has_megastructure = AIO_mega_04
	}
	custom_tooltip = {
		fail_text = AIO_projection_fail.cooldown
		NOT = { has_country_flag = AIO_projection_cooldown }
	}
	OR = {
		resource_stockpile_compare = {
			resource = AIO_advanced_logic
			value >= $COST$
		}
		is_country_type = AIO_empire
	}
}

# country event：选择投入时重新检查来源和星系，取消选址不收费。
# from - system fromfrom - mega
AIO_projection_option_available = {
	AIO_projection_can_start = { COST = $COST$ }
	custom_tooltip = {
		fail_text = AIO_projection_fail.target
		exists = from
		from = { AIO_projection_center_available = yes }
	}
	custom_tooltip = {
		fail_text = AIO_projection_fail.need_mega_04
		exists = fromfrom
		fromfrom = {
			is_megastructure_type = AIO_mega_04
			owner = { is_same_value = root }
		}
	}
}

# system；同一地点不能重复预约或叠加化身。
AIO_projection_center_available = {
    NOR = {
		has_star_flag = AIO_projection_pending
		has_star_flag = AIO_Aforgomon_existed
	}
}

# system；共享国家的舰队不算攻击目标。
AIO_Aforgomon_has_targets = {
	any_fleet_in_system = {
		NOT = { is_owned_by = event_target:AIO_Aforgomon_country }
	}
}
