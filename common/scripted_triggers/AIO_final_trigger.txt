# Country scope. Keep all entry points subject to the same preparation requirements.
AIO_final_can_start = {
    has_menace_perk = menp_AIO_break_rules
    exists = event_target:AIO_mist_system
    exists = event_target:AIO_gate_system
    any_owned_megastructure = { is_megastructure_type = AIO_mega_04 }
    NOR = {
        has_country_flag = AIO_final_start
        has_country_flag = AIO_final_won
        has_country_flag = AIO_final_lost
        has_country_flag = AIO_empire_country
    }
}

# scope: system
AIO_final_valid_external_target = {
    exists = star
    NOR = {
        is_star_class = sc_AIO_void
        is_same_value = event_target:AIO_mist_system
        is_same_value = event_target:AIO_gate_system
        has_star_flag = AIO_Aforgomon_existed
        has_star_flag = AIO_DAIT_center_busy
        AIO_safe_remove_system_exclude_API = yes
        AIO_is_crisis_mega_system = yes
        has_star_flag = synth_queen_capital
        any_system_planet = { AIO_safe_remove_planet_exclude_API = yes }
    }
}

# scope: system；无名之雾本身是虚空星系，因此单独放行。
AIO_final_valid_target = {
    OR = {
        AIO_final_valid_external_target = yes
        is_same_value = event_target:AIO_mist_system
    }
}

# scope: country；诱导及约束共用，每轮仅一个项目，显现后禁止启动。
AIO_final_can_select = {
    has_country_flag = AIO_final_start
    resource_stockpile_compare = { resource = AIO_advanced_logic value >= @AIO_final_induction_cost }
    NOR = {
        has_special_project = AIO_FINAL_INDUCTION
        has_special_project = AIO_FINAL_CONSTRAINT
    }
    any_situation = {
        is_situation_type = AIO_final_location_situation
        current_stage = AIO_final_omen_stage
        NOR = {
            has_situation_flag = AIO_final_settled
            has_situation_flag = AIO_final_induced
        }
    }
}

# scope: system event，FROM = 发起动作的巨构。
AIO_final_action_source_valid = {
    AIO_final_valid_external_target = yes
    exists = from
    from = {
        is_megastructure_type = AIO_mega_04
        owner = { AIO_final_can_select = yes }
    }
}
