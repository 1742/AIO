# One pending cast per operation and country; target is persisted by the situation.
AIO_DAIT_spawn_situation = {
    picture = GFX_evt_astral_rift_vortex_nomask
    desc = AIO_DAIT_spawn_situation_desc
    category = neutral
    situation_log_category = urgent_matters
    initial_progress = 0
    progress_direction = monodirectional
    complete_icon = GFX_situation_outcome_positive
    complete_icon_frame = GFX_situation_outcome_frame_green
    on_start = {
        hidden_effect = { set_variable = { which = AIO_DAIT_windup_days value = owner.AIO_DAIT_spawn_delay } }
    }
    total_progress = { base = 0 add = this.AIO_DAIT_windup_days }
    monthly_progress = { base = 30 }
    stages = {
        AIO_DAIT_windup_stage = {
            section_weight = { base = 1 }
            icon = GFX_situation_stage_1
            icon_background = GFX_situation_stage_frame_green
        }
    }
    abort_trigger = {
        OR = {
            NOT = { exists = target }
            owner = {
                NOT = { has_country_flag = AIO_DAIT_spawn_pending }
            }
            owner = { NOT = { AIO_DAIT_spawn_unlocked = yes } }
            target = { NOT = { AIO_DAIT_spawn_target_valid = yes } }
        }
    }
    on_abort = {
        owner = {
            if = {
                limit = { has_country_flag = AIO_DAIT_spawn_pending }
                add_resource = { AIO_advanced_logic = 0.5 mult = this.AIO_DAIT_spawn_paid }
                create_message = {
                    type = MESSAGE_DYSON_SHOT
                    localization = AIO_DAIT_cancelled
                    days = 30
                }
                hidden_effect = { AIO_DAIT_cleanup_spawn = yes }
            }
        }
    }
    on_progress_complete = {
        hidden_effect = {
            target = { save_event_target_as = AIO_DAIT_target_system }
        }
        owner = {
            if = {
                limit = { has_country_flag = AIO_DAIT_spawn_pending }
                AIO_DAIT_finish_spawn = yes
            }
        }
    }
    approach = {
        name = AIO_DAIT_wait_approach
        default = yes
        icon = GFX_situation_approach_suspicious
        icon_background = GFX_situation_approach_bg_yellow
        on_select = { custom_tooltip = AIO_DAIT_wait_approach_desc }
        ai_weight = { base = 1 }
    }
}
