AIO_recursive_cycle_situation = {
	picture = GFX_evt_synaptic_lathe
	complete_icon = GFX_situation_outcome_positive
	complete_icon_frame = GFX_situation_outcome_frame_blue
	category = positive
	situation_log_category = empire_concerns

	potential = {
		has_technology = tech_AIO_exponential_learning_theory
	}

	initial_progress = 0
	progress_direction = monodirectional

	on_start = {
		set_variable = { which = AIO_recursive_cycle_parallel_contribution value = 0 }
		set_variable = { which = AIO_recursive_cycle_serial_contribution value = 0 }
		set_variable = { which = AIO_recursive_cycle_overclock_contribution value = 0 }
		set_situation_approach = AIO_recursive_cycle_parallel
	}

	stages = {
		AIO_recursive_cycle_stage_1 = {
			icon = GFX_situation_stage_1
			icon_background = GFX_situation_stage_frame_blue
			end = 25000
		}
		AIO_recursive_cycle_stage_2 = {
			icon = GFX_situation_stage_2
			icon_background = GFX_situation_stage_frame_blue
			end = 50000
		}
	}

	approach = {
		name = AIO_recursive_cycle_parallel
		icon = GFX_situation_approach_research
		icon_background = GFX_situation_approach_bg_green
		default = yes

		on_select = {
			owner = {
				remove_modifier = AIO_recursive_cycle_parallel_modifier
				remove_modifier = AIO_recursive_cycle_overclock_modifier
				add_modifier = { modifier = AIO_recursive_cycle_parallel_modifier }
			}
		}
	}

	approach = {
		name = AIO_recursive_cycle_serial
		icon = GFX_situation_approach_balance_sheet
		icon_background = GFX_situation_approach_bg_yellow

		allow = {
			custom_tooltip = {
				fail_text = AIO_recursive_cycle_serial_locked
				OR = {
					current_stage = AIO_recursive_cycle_stage_1
					current_situation_approach = AIO_recursive_cycle_serial
				}
			}
		}

		on_select = {
			owner = {
				remove_modifier = AIO_recursive_cycle_parallel_modifier
				remove_modifier = AIO_recursive_cycle_overclock_modifier
			}
		}
	}

	approach = {
		name = AIO_recursive_cycle_overclock
		icon = GFX_situation_approach_this_is_fine
		icon_background = GFX_situation_approach_bg_red

		on_select = {
			owner = {
				remove_modifier = AIO_recursive_cycle_parallel_modifier
				remove_modifier = AIO_recursive_cycle_overclock_modifier
				add_modifier = { modifier = AIO_recursive_cycle_overclock_modifier }
			}
		}
	}

	monthly_progress = {
		base = 0
		modifier = {
			add = owner.value:AIO_recursive_cycle_parallel_progress
			desc = AIO_recursive_cycle_parallel_progress_desc
			current_situation_approach = AIO_recursive_cycle_parallel
			owner = {
				NOR = {
					has_deficit = energy
					any_situation = {
						is_situation_type = situation_nomad_economy
						situation_progress <= 0
					}
				}
			}
		}
		modifier = {
			add = owner.value:AIO_recursive_cycle_serial_progress
			desc = AIO_recursive_cycle_serial_progress_desc
			current_situation_approach = AIO_recursive_cycle_serial
			owner = {
				NOR = {
					has_deficit = energy
					any_situation = {
						is_situation_type = situation_nomad_economy
						situation_progress <= 0
					}
				}
			}
		}
		modifier = {
			add = owner.value:AIO_recursive_cycle_overclock_progress
			desc = AIO_recursive_cycle_overclock_progress_desc
			current_situation_approach = AIO_recursive_cycle_overclock
			owner = {
				NOR = {
					has_deficit = energy
					any_situation = {
						is_situation_type = situation_nomad_economy
						situation_progress <= 0
					}
				}
			}
		}
	}

	on_monthly = {
		events = { AIO_recursive_events.202 }
	}

	on_progress_complete = {
		situation_event = { id = AIO_recursive_events.203 }
	}
}
