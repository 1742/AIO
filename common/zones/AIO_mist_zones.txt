# Jobs are supplied only by specializations, once per district in that specialization.

zone_AIO_mist_government = {
    icon = GFX_diplomatic_weight_large
    base_buildtime = @zone_buildtime
    potential = { # planet scope
    }

    unlock = { # planet scope
        NOT = {
            has_designation = col_resort
        }
    }

    resources = {
        category = planet_zones
        inline_script = {
            script = buildings/nomadic_cost_switcher
            REGULAR_RESOURCE = minerals
            NOMADIC_RESOURCE = alloys
            COST = @zone_cost
            NOMADIC_COST_MULT = 0.5
        }
    }

    max_buildings = 6

    zone_sets = {
        AIO_mist_government
    }

    include = {
        all
    }

    excluded_building_sets = {
        generator
        mining
        farming
        knights
        betharian
        fallen_empire_generator
        fallen_empire_farming
        fallen_empire_mining
        urban_automation
        urban_automation_arcology
        generator_automation
        mining_automation
        farming_automation
    }

    ai_priority = 0

    triggered_district_planet_modifier = {
        potential = {
            exists = owner
            owner = {
                is_gestalt = no
                is_tankbound_empire = no
            }
        }
        job_clerk_add = 100
    }

    triggered_district_planet_modifier = {
        potential = {
            exists = owner
            owner = { is_tankbound_empire = yes }
        }
        job_production_overseer_add = 100
    }

    triggered_district_planet_modifier = {
        potential = {
            exists = owner
            owner = { is_gestalt = yes }
        }
        job_logistics_drone_add = 100
    }
}
