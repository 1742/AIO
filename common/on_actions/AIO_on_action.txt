# AIO on_action触发事件

# 定向应用无限入口
# THIS = 目标行星
# FROM = 发起动作的巨构
on_AIO_DAIT_lock_target_started = {
	events = {
		AIO_DAIT_events.201
	}
}

on_game_start_country = {
    events = {
        AIO_mod_events.201 # 更新提醒
        AIO_crisis_events.808  # 万物归一起源
    }
}

on_yearly_pulse = {
    events = {
        AIO_crisis_events.701  # 天灾目标，征用神经芯片
        AIO_crisis_events.703  # 天灾目标，控有科学枢纽
    }
}

on_tech_increased = {
    events = {
        AIO_crisis_events.702  # 天灾目标，研究科技
        AIO_crisis_events.807 # 完成科技升维假设，提示建造视相引擎
		AIO_fe_events.201 # 授予 AIO 堕落帝国建筑科技对应的原版科技
    }
}

on_country_destroyed = {
    events = {
        AIO_crisis_events.601 # AIO天灾帝国被摧毁
    }
}

on_entering_system_fleet = {
    events = {
        AIO_crisis_events.805  # 驱逐无名之雾的其他舰船
    }
}

on_ship_built = {
    events = {
        AIO_crisis_events.801  # 禁止AIO帝国造船
    }
}

    }
