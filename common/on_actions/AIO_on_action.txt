# AIO on_action触发事件

# 定向应用无限入口
# THIS = 目标行星
# FROM = 发起动作的巨构


on_game_start_country = {
    events = {
        AIO_DAIT_events.201
        AIO_mod_events.201 # 更新提醒
        AIO_crisis_events.201  # 万物归一起源
    }
}

# No scope. Add our load handler alongside other mods' handlers.

    }
}

on_yearly_pulse_country = {
    events = {
        AIO_crisis_events.902 # 给予递归载体物种模板
    }
}

on_tech_increased = {
    events = {
        AIO_crisis_events.401 # 完成科技升维假设，提示建造视相引擎
        AIO_crisis_events.801  # 天灾目标，研究科技
        AIO_crisis_events.902 # 虚拟帝国完成指数级学习理论
		AIO_fe_events.201 # 授予 AIO 堕落帝国建筑科技对应的原版科技
    }
}

}

    }
}
