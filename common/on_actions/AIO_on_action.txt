# AIO on_action触发事件

# 定向应用无限入口
# THIS = 目标行星
# FROM = 发起动作的巨构
on_AIO_DAIT_lock_target_started = {
	events = {
		AIO_DAIT_events.201
	}
}

on_game_start_country = {
    events = {
        AIO_mod_events.201 # 更新提醒
        AIO_crisis_events.201  # 万物归一起源
    }
}

on_yearly_pulse = {
    events = {
        AIO_crisis_events.804  # 天灾目标，建造现实干涉仪
    }
}

on_tech_increased = {
    events = {
        AIO_crisis_events.401 # 完成科技升维假设，提示建造视相引擎
        AIO_crisis_events.801  # 天灾目标，研究科技
		AIO_compatible_events.201 # 授予原版堕落帝国建筑科技
        AIO_compatible_events.202 # 授予原版灵能光环科技
    }
}

on_country_destroyed = {
    events = {
        AIO_crisis_events.701 # AIO天灾帝国被摧毁
    }
}

on_entering_system_fleet = {
    events = {
        AIO_crisis_events.901  # 驱逐无名之雾的其他舰船
    }
}

# This = owner of fleet 1 (combatant)
# From = owner of fleet 2 (destroyed)
# FromFrom = fleet 1
# FromFromFrom = fleet 2
on_fleet_destroyed_perp = {
    events = {
        AIO_Aforgomom_events.202 # 亚弗戈蒙之怒被摧毁
    }
}

# Executed when a megastructure has a new owner
# this = new owner
# from = megastructure
# fromfrom = old owner (if existing)
on_megastructure_change_owner = {
    events = {
        AIO_crisis_events.902 # 失去现实干涉仪
    }
}

# Fired when a ship finishes researching an anomaly.
# this = ship
# from = planet
anomaly_success = {
    events = {
        AIO_crisis_events.8021 # 天灾目标，完成特殊项目
    }
}

# Fired when a ship completes a special project.
# this = country
# fromfromfrom = ship
special_project_success = {
    events = {
        AIO_crisis_events.8022 # 天灾目标，完成特殊项目
    }
}

# Executed when the exploration of an astral rift ends
# THIS = Country: Explorer country
# FROM = AstralRift: Explored astral rift
# FROMFROM = Exploring Fleet
on_astral_rift_exploration_complete = {
    events = {
        AIO_crisis_events.803 # 天灾目标，探索星界裂隙
    }
}
