# Reality Interference prototype.
# One native Psionic Aura carries several effects selected by star flags.
AIO_reality_interference_aura = {
	intensity_level = {
		end = 1000
		custom_tooltip = AIO_reality_interference_aura_level_1_tt
		inline_script = {
			script = AIO/reality_interference_aura
		}
	}
	intensity_level = {
		end = 2000
		custom_tooltip = AIO_reality_interference_aura_level_2_tt
		inline_script = {
			script = AIO/reality_interference_aura
		}
	}
	intensity_level = {
		end = 2500
		custom_tooltip = AIO_reality_interference_aura_level_3_tt
		inline_script = {
			script = AIO/reality_interference_aura
		}
	}

	# Propagation is independent of intensity levels. Every existing field node
	# attempts to advance the frontier once per month, while the scripted
	# trigger on the target system keeps it inside the owner's source range.
	on_monthly = {
		if = {
			limit = {
				is_aura_intensity_level = 3
				NOT = {
					has_star_flag = AIO_energy_backlash_active
					has_star_flag = AIO_balanced_backlash_active
					has_star_flag = AIO_ecology_backlash_active
					has_star_flag = AIO_interference_backlash_sealed
				}
				OR = {
					has_star_flag = AIO_interference_source
					AIO_is_nearby_interference_source = { OWNER = aura_owner }
				}
			}
			# Each doctrine rolls its own accident; the star flags above keep them from
			# stacking on the same system.
			if = {
				limit = { aura_owner = { has_policy_flag = AIO_interference_mode_energy } }
				AIO_interference_energy_start = yes
			}
			else_if = {
				limit = { aura_owner = { has_policy_flag = AIO_interference_mode_balanced } }
				AIO_interference_balanced_start = yes
			}
			else_if = {
				limit = { aura_owner = { has_policy_flag = AIO_interference_mode_ecology } }
				AIO_interference_ecology_start = yes
			}
		}
		if = {
			limit = { NOT = { has_star_flag = AIO_interference_spread_block } }
			AIO_interference_prototype_try_spread = yes
			set_timed_star_flag = {
				flag = AIO_interference_spread_block
				months = 3
			}
		}
	}

	on_vanish = {
		if = {
			limit = { has_star_flag = AIO_interference_source }
			remove_star_flag = AIO_interference_source
			aura_owner = { AIO_interference_rebuild_effect = yes }
		}
		# A sealed system only stays sealed while its field exists.
		remove_star_flag = AIO_interference_backlash_sealed
		clear_variable = AIO_interference_sources
	}

	increase_trigger = {
		OR = {
			has_star_flag = AIO_interference_source
			AIO_is_nearby_interference_source = {
				OWNER = aura_owner
			}
		}
	}
	monthly_increase = {
		base = 1
		modifier = {
			add = value:AIO_reality_interference_nearby_source_growth|OWNER|root.aura_owner|
			OR = {
				has_star_flag = AIO_interference_source
				AIO_is_nearby_interference_source = {
					OWNER = aura_owner
				}
			}
			desc = AIO_interference_source_intensity_desc
		}
		modifier = {
			add = value:AIO_reality_interference_proof_growth|OWNER|root.aura_owner|
			OR = {
				any_system_planet_colony = {
					is_owned_by = root.aura_owner
					total_workforce_with_job_tag = {
						tags = { AIO_proof }
						value > 0
					}
				}
				any_system_ship_colony = {
					is_owned_by = root.aura_owner
					total_workforce_with_job_tag = {
						tags = { AIO_proof }
						value > 0
					}
				}
			}
			desc = AIO_interference_proof_intensity_desc
		}
		modifier = {
			mult = 2
			aura_owner = { has_country_flag = AIO_interference_change_policy }
		}
	}

	monthly_decrease = {
		base = 0
		modifier = {
			subtract = 5
			NOR = {
				has_star_flag = AIO_interference_source
				AIO_is_nearby_interference_source = {
					OWNER = root.aura_owner
				}
			}
			desc = AIO_interference_disconnected_desc
		}
	}

	# Reuse vanilla textures for the prototype. Dedicated AIO textures can be
	# substituted after the mechanics have been validated in game.
	aura_galaxy_texture = "gfx/map/psionic_auras/desire_aura.dds"
	aura_hyperlane_texture = "gfx/map/psionic_auras/desire_aura_hyperlane.dds"
}
