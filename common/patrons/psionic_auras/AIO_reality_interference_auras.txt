# Reality Interference prototype.
# One native Psionic Aura carries the effects selected by the owner's doctrine.
AIO_reality_interference_aura = {
	intensity_level = {
		end = 1000
		inline_script = { script = AIO/reality_interference_aura }
	}
	intensity_level = {
		end = 2000
		inline_script = { script = AIO/reality_interference_aura }
	}
	intensity_level = {
		end = 2500
		inline_script = { script = AIO/reality_interference_aura }
	}

	on_start = {
		AIO_interference_recount_sources = { OWNER = root.aura_owner }
		aura_owner = { AIO_interference_update_district_capacity = yes }
	}

	on_gain_level = {
		aura_owner = { AIO_interference_update_district_capacity = yes }
	}

	# Propagation is independent of intensity levels. Every existing field node
	# attempts to advance the frontier once per month, while the scripted
	# trigger on the target system keeps it inside the owner's source range.
	on_monthly = {
		if = {
			limit = {
				is_aura_intensity_level = 3
				AIO_system_has_interference_source = { OWNER = root.aura_owner }
				aura_owner = {
					NOR = {
						has_country_flag = AIO_backlash_active
						has_country_flag = AIO_backlash_seal
					}
				}
			}
			aura_owner = {
				switch = {
					trigger = has_policy_flag
					AIO_interference_mode_balanced = {
						AIO_interference_backlash_start = { TYPE = balanced }
					}
					AIO_interference_mode_energy = {
						AIO_interference_backlash_start = { TYPE = energy }
					}
					AIO_interference_mode_ecology = {
						AIO_interference_backlash_start = { TYPE = ecology }
					}
					AIO_interference_mode_industry = {
						AIO_interference_backlash_start = { TYPE = industry }
					}
					AIO_interference_mode_metric = {
						AIO_interference_backlash_start = { TYPE = metric }
					}
				}
			}
		}
		if = {
			limit = { NOT = { has_star_flag = AIO_interference_spread_block } }
			AIO_interference_prototype_try_spread = yes
			set_timed_star_flag = {
				flag = AIO_interference_spread_block
				months = 3
			}
		}
	}

	on_vanish = {
		clear_variable = AIO_interference_sources
	}

	increase_trigger = {
		OR = {
			AIO_system_has_interference_source = { OWNER = root.aura_owner }
			AIO_is_nearby_interference_source = {
				OWNER = root.aura_owner
			}
		}
	}
	monthly_increase = {
		base = 2
		modifier = {
			add = value:AIO_reality_interference_source_growth
			OR = {
				AIO_system_has_interference_source = { OWNER = root.aura_owner }
				AIO_is_nearby_interference_source = {
					OWNER = root.aura_owner
				}
			}
			desc = AIO_interference_source_intensity_desc
		}
		modifier = {
			add = value:AIO_interference_aura_proof_growth|OWNER|root.aura_owner|
			OR = {
				any_system_planet_colony = {
					is_owned_by = root.aura_owner
					total_workforce_with_job_tag = {
						tags = { AIO_proof }
						value > 0
					}
				}
				any_system_ship_colony = {
					is_owned_by = root.aura_owner
					total_workforce_with_job_tag = {
						tags = { AIO_proof }
						value > 0
					}
				}
			}
			desc = AIO_interference_proof_intensity_desc
		}
		modifier = {
			mult = 2
			aura_owner = { has_country_flag = AIO_interference_change_policy }
		}
	}

	monthly_decrease = {
		base = 0
		modifier = {
			subtract = 5
			NOR = {
				AIO_system_has_interference_source = { OWNER = root.aura_owner }
				AIO_is_nearby_interference_source = {
					OWNER = root.aura_owner
				}
			}
			desc = AIO_interference_disconnected_desc
		}
	}
	daily_decrease = {  }

	# Reuse vanilla textures for the prototype. Dedicated AIO textures can be
	# substituted after the mechanics have been validated in game.
	aura_galaxy_texture = "gfx/map/psionic_auras/desire_aura.dds"
	aura_hyperlane_texture = "gfx/map/psionic_auras/desire_aura_hyperlane.dds"
}
