# Reality Interference prototype.
# One native Psionic Aura carries several effects selected by star flags.
AIO_reality_interference_aura = {
	triggered_system_modifier = {
		potential = { always = yes }
		planet_jobs_physics_research_produces_mult = 0.05
		mult = trigger:is_aura_intensity_level
	}
	triggered_system_modifier = {
		potential = {
			solar_system = {
				OR = {
					has_star_flag = AIO_interference_energy
					has_star_flag = AIO_interference_energy_residual
				}
			}
		}
		modifier = {
			planet_jobs_energy_produces_mult = 0.1
			ship_weapon_damage = 0.2
        	ship_shield_mult = -0.05
		}
		mult = trigger:is_aura_intensity_level
	}
	triggered_system_modifier = {
		potential = {
			OR = {
				has_star_flag = AIO_interference_resource
				has_star_flag = AIO_interference_resource_residual
			}
		}
		station_gatherers_produces_mult = 0.33
		planet_jobs_minerals_produces_mult = 0.1
        planet_jobs_worker_volatile_motes_produces_mult = 0.05
        planet_jobs_worker_exotic_gases_produces_mult = 0.05
        planet_jobs_worker_rare_crystals_produces_mult = 0.05
		mult = trigger:is_aura_intensity_level
	}
	triggered_system_modifier = {
		potential = {
			solar_system = {
				OR = {
					has_star_flag = AIO_interference_bio
					has_star_flag = AIO_interference_bio_residual
				}
			}
		}
		modifier = {
			planet_jobs_food_produces_mult = 0.1
			planet_jobs_society_research_produces_mult = 0.15
			planet_jobs_biomass_produces_mult = 0.1
		}
		mult = trigger:is_aura_intensity_level
	}
	triggered_system_modifier = {
		potential = {
			solar_system = {
				OR = {
					has_star_flag = AIO_interference_industry
					has_star_flag = AIO_interference_industry_residual
				}
			}
		}
		modifier = {
			planet_jobs_consumer_goods_produces_mult = 0.05
			planet_jobs_alloys_produces_mult = 0.05
			planet_jobs_engineering_research_produces_mult = 0.15
			planet_jobs_minerals_upkeep_mult = 0.1
		}
		mult = trigger:is_aura_intensity_level
	}
	triggered_system_modifier = {
		potential = {
			OR = {
				has_star_flag = AIO_interference_build
				has_star_flag = AIO_interference_build_residual
			}
		}
		planet_building_build_speed_mult = 0.1
		starbase_building_build_speed_mult = 0.1
		starbase_upgrade_speed_mult = 0.1
		starbase_shipyard_build_speed_mult = 0.1
		megastructure_build_speed_mult = 0.33
		mult = trigger:is_aura_intensity_level
	}
	triggered_system_modifier = {
		potential = {
			OR = {
				has_star_flag = AIO_interference_cost
				has_star_flag = AIO_interference_cost_residual
			}
		}
		planet_jobs_alloys_upkeep_mult = -0.1
		planet_buildings_alloys_cost_mult = -0.1
		starbase_shipyard_build_alloys_cost_mult = -0.1
		mult = trigger:is_aura_intensity_level
	}

	intensity_level = {
		end = 1000
		custom_tooltip = AIO_reality_interference_aura_level_1_tt
	}
	intensity_level = {
		end = 2000
		custom_tooltip = AIO_reality_interference_aura_level_2_tt
	}
	intensity_level = {
		end = 2500
		custom_tooltip = AIO_reality_interference_aura_level_3_tt
	}

	# Propagation is independent of intensity levels. Every existing field node
	# attempts to advance the frontier once per month, while the scripted
	# trigger on the target system keeps it inside the owner's source range.
	on_monthly = {
		if = {
			limit = { NOT = { has_star_flag = AIO_interference_spread_cooldown } }
			AIO_interference_prototype_try_spread = yes
			set_timed_star_flag = {
				flag = AIO_interference_spread_cooldown
				months = 3
			}
		}
	}

	on_progress_complete = {
		hidden_effect = {
			set_aura_intensity = 2000
		}
	}

	on_vanish = {
		aura_owner = { save_event_target_as = AIO_interference_vanishing_owner }
		AIO_clear_interference_flag = yes
		event_target:AIO_interference_vanishing_owner = {
			AIO_interference_rebuild_effect_flags = yes
		}
	}

	increase_trigger = {
		OR = {
			has_star_flag = AIO_interference_source
			AIO_is_nearby_interference_source = {
				OWNER = root.aura_owner
			}
		}
	}
	monthly_increase = {
		base = 1
		modifier = {
			add = value:AIO_reality_interference_nearby_source_growth
			OR = {
				has_star_flag = AIO_interference_source
				AIO_is_nearby_interference_source = {
					OWNER = root.aura_owner
				}
			}
			desc = AIO_interference_source_intensity_desc
		}
	}

	monthly_decrease = {
		base = 0
		modifier = {
			subtract = 5
			NOR = {
				has_star_flag = AIO_interference_source
				AIO_is_nearby_interference_source = {
					OWNER = root.aura_owner
				}
			}
			desc = AIO_interference_disconnected_desc
		}
	}

	# Reuse vanilla textures for the prototype. Dedicated AIO textures can be
	# substituted after the mechanics have been validated in game.
	aura_galaxy_texture = "gfx/map/psionic_auras/desire_aura.dds"
	aura_hyperlane_texture = "gfx/map/psionic_auras/desire_aura_hyperlane.dds"
}
