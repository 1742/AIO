# Shared national backlash situation body. TYPE selects policy and approaches;
# EVENT_TENS maps to the 3x0-3x5 situation events.
picture = $PICTURE$
category = negative
situation_log_category = developments
complete_icon_frame = GFX_situation_outcome_frame_red
complete_icon = GFX_situation_outcome_negative
fail_icon_frame = GFX_situation_outcome_frame_green
fail_icon = GFX_situation_outcome_positive
initial_progress = 15
progress_direction = monodirectional

on_start = {
	situation_event = { id = AIO_interference_events.$EVENT_TENS$0 }
}
on_progress_complete = {
	if = {
		limit = { current_situation_approach = AIO_$TYPE$_reverse }
		situation_event = { id = AIO_interference_events.$EVENT_TENS$3 }
	}
	else = {
		situation_event = { id = AIO_interference_events.$EVENT_TENS$4 }
	}
}
on_fail = {
	situation_event = { id = AIO_interference_events.$EVENT_TENS$5 }
}

stages = {
	stage_1 = {
		icon = GFX_situation_stage_1
		icon_background = GFX_situation_stage_frame_red
		end = 35
	}
	stage_2 = {
		icon = GFX_situation_stage_2
		icon_background = GFX_situation_stage_frame_red
		end = 75
		on_first_enter = {
			situation_event = { id = AIO_interference_events.$EVENT_TENS$1 }
		}
	}
	stage_3 = {
		icon = GFX_situation_stage_3
		icon_background = GFX_situation_stage_frame_red
		end = 100
		on_first_enter = {
			set_situation_flag = AIO_approach_locked
			situation_event = { id = AIO_interference_events.$EVENT_TENS$2 }
		}
	}
}

inline_script = {
	script = AIO/interference_backlash_situation_progress
	TYPE = $TYPE$
}

approach = {
	name = AIO_$TYPE$_maintain
	icon = GFX_situation_approach_research
	icon_background = GFX_situation_approach_bg_red
	default = yes
	allow = {
		OR = {
			current_situation_approach = AIO_$TYPE$_maintain
			AND = {
				owner = { has_policy_flag = AIO_interference_mode_$TYPE$ }
				NOT = { has_situation_flag = AIO_approach_locked }
			}
		}
	}
	resources = {
		category = situations
		upkeep = {
			$MAINTAIN_UPKEEP$
		}
	}
	ai_weight = { base = 10 }
}
approach = {
	name = AIO_$TYPE$_blockade
	icon = GFX_situation_approach_scales
	icon_background = GFX_situation_approach_bg_yellow
	allow = {
		OR = {
			current_situation_approach = AIO_$TYPE$_blockade
			AND = {
				owner = { has_policy_flag = AIO_interference_mode_$TYPE$ }
				NOT = { has_situation_flag = AIO_approach_locked }
			}
		}
	}
	resources = {
		category = situations
		upkeep = {
			$BLOCKADE_UPKEEP$
		}
	}
	ai_weight = { base = 5 }
}
approach = {
	name = AIO_$TYPE$_reverse
	icon = GFX_situation_approach_dark_matter
	icon_background = GFX_situation_approach_bg_green
	potential = { owner = { has_menace_perk = menp_AIO_mega_03 } }
	allow = {
		OR = {
			current_situation_approach = AIO_$TYPE$_reverse
			AND = {
				owner = { has_policy_flag = AIO_interference_mode_$TYPE$ }
				NOT = { has_situation_flag = AIO_approach_locked }
			}
		}
	}
	resources = {
		category = situations
		upkeep = {
			energy = 200
			AIO_advanced_logic = 5
		}
	}
	ai_weight = { base = 20 }
