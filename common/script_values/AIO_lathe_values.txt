# The mist is a mixed colony: count filled node jobs, not all residents.
AIO_chips_num_per_100 = {
    base = 0
    complex_trigger_modifier = {
        trigger = num_assigned_jobs
        trigger_scope = colony
        parameters = { job = AIO_chips }
        mode = add
        mult = 0.01
    }
}

AIO_chips_produce_factor = {
    base = 0
    add = value:AIO_chips_num_per_100
    divide = 20
}

AIO_chips_num_per_100_double = {
    base = 0
    add = value:AIO_chips_num_per_100
    mult = 2
}

AIO_chips_upkeep_factor = {
    base = 0
    add = value:AIO_chips_num_per_100
    divide = 40
}

# Vanilla num_chip_slave_purge, distributed only across node population groups.
AIO_chips_purge_factor = {
    base = 0
    add = value:AIO_chips_num_per_100
    mult = 4
    divide = value:AIO_chips_pop_groups
    min = 1
}

AIO_chips_pop_groups = {
    base = 0
    complex_trigger_modifier = {
        trigger = count_owned_pop_group
        trigger_scope = colony
        parameters = { limit = { is_pop_category = AIO_chips } }
        mode = add
    }
    min = 1
}
