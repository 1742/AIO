	mult = 2
}

# scope: system
AIO_interference_aura_proof_growth = {
	complex_trigger_modifier = {
		trigger = total_system_workforce_with_job_tag
		parameters = {
			tags = { AIO_proof }
			limit = {
				colony.owner = { is_same_value = $OWNER$ }
			}
		}
		mode = add
	}
	divide = 100
	mult = 0.5
	max = 5
}

# scope: country
AIO_interferometer_num = {
	base = 0
	complex_trigger_modifier = {
		trigger = count_owned_megastructure
		parameters = {
			limit = { AIO_is_interference_source = yes }
		}
		mode = add
	}
	complex_trigger_modifier = {
		trigger = count_owned_nonprimary_starbase
		parameters = { limit = { AIO_is_interference_source = yes } }
		mode = add
	}
	complex_trigger_modifier = {
		trigger = count_owned_ship
		parameters = { limit = { AIO_is_mobile_interference_source = yes } }
		mode = add
	}
	min = 0
}

# Scope: system. Count installations, including multiple arkships in one system.
AIO_interference_local_source_count = {
	base = 0
	complex_trigger_modifier = {
		trigger = count_system_megastructure
		parameters = {
			limit = {
				AIO_is_interference_source = yes
				owner = { is_same_value = $OWNER$ }
			}
		}
		mode = add
	}
	complex_trigger_modifier = {
		trigger = count_starbase_in_system
		parameters = {
			limit = {
				is_waystation_starbase = yes
				AIO_is_interference_source = yes
				owner = { is_same_value = $OWNER$ }
			}
		}
		mode = add
	}
	complex_trigger_modifier = {
		trigger = count_ship_in_system
		parameters = {
			limit = {
				AIO_is_mobile_interference_source = yes
				owner = { is_same_value = $OWNER$ }
			}
		}
		mode = add
	}
}

# Scope: country. Negative reservations for the other generator type.
# Count at each installation so mobile starbases are explicitly included.
# One generator per installation; both completed and queued modules reserve a slot.
AIO_interference_arkship_reservation = {
	base = 0
	complex_trigger_modifier = {
		trigger = count_owned_ship
		parameters = {
			limit = {
				is_arkship_ship = yes
				starbase? = {
					count_starbase_modules = {
						type = AIO_arkship_interference_generator
						include_being_constructed = yes
						count > 0
					}
				}
			}
		}
		mode = add
	}
	mult = -1
	min = -5
	max = 0
}

AIO_interference_waystation_reservation = {
	base = 0
	complex_trigger_modifier = {
		trigger = count_owned_nonprimary_starbase
		parameters = {
			limit = {
				is_waystation_starbase = yes
				count_starbase_modules = {
					type = AIO_waystation_interference_generator
					include_being_constructed = yes
					count > 0
				}
			}
		}
		mode = add
	}
	mult = -1
	min = -5
	max = 0
}
