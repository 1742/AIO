# Scope: a system carrying an AIO Reality Interference aura.
# Each source in this system or an adjacent system contributes 2 monthly intensity.
AIO_reality_interference_nearby_source_growth = {
	base = 0

	modifier = {
		add = 1
		has_star_flag = AIO_interference_source
		has_psionic_aura = AIO_reality_interference_aura
	}

	complex_trigger_modifier = {
		trigger = count_neighbor_system
		parameters = {
			limit = {
				has_star_flag = AIO_interference_source
				has_psionic_aura = AIO_reality_interference_aura
				aura_owner = { is_same_value = root.aura_owner }
			}
		}
		mode = add
	}

	mult = 2
}

# Scope: system carrying the AIO aura. One hundred employed proof workers add
# 0.5 monthly intensity; only the aura owner's workforce counts.
AIO_reality_interference_proof_growth = {
	complex_trigger_modifier = {
		trigger = total_system_workforce_with_job_tag
		parameters = {
			tags = { AIO_proof }
			limit = {
				colony.owner = { is_same_value = root.aura_owner }
			}
		}
		mode = add
	}
	divide = 100
	mult = 0.5
	max = 5
}

# Planet or fleet scope. Active sources of one type stack up to three; a fading
# residual keeps the strength of one source until it expires.
AIO_interference_type_source_count = {
	base = 0
	add = solar_system.AIO_interference_$TYPE$_sources
	modifier = {
		add = 1
		solar_system = { has_star_flag = AIO_interference_$TYPE$_residual }
	}
	max = 3
}

AIO_interference_type_level_multiplier = {
	base = 0
	add = value:AIO_interference_type_source_count|TYPE|$TYPE$|
	mult = trigger:is_aura_intensity_level
}
