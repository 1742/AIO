# Scope: a system carrying the AIO Reality Interference aura.
AIO_reality_interference_nearby_source_growth = {
	base = 0
	modifier = {
		add = 1
		has_star_flag = AIO_interference_source
		has_psionic_aura = AIO_reality_interference_aura
	}
	complex_trigger_modifier = {
		trigger = count_neighbor_system
		parameters = {
			limit = {
				has_star_flag = AIO_interference_source
				has_psionic_aura = AIO_reality_interference_aura
				aura_owner = { is_same_value = $OWNER$ }
			}
		}
		mode = add
	}
	mult = 2
}

# scope: system
AIO_reality_interference_proof_growth = {
	complex_trigger_modifier = {
		trigger = total_system_workforce_with_job_tag
		parameters = {
			tags = { AIO_proof }
			limit = {
				colony.owner = { is_same_value = $OWNER$ }
			}
		}
		mode = add
	}
	divide = 100
	mult = 0.5
	max = 5
}

# Scope: a planet or fleet affected by the aura.
# Aura intensity belongs to the system, not to the affected planet or fleet.
AIO_interference_intensity_multiplier = {
	base = 0
	add = solar_system.AIO_interference_sources
	mult = solar_system.trigger:is_aura_intensity_level
}

# scope: situation with a country target
AIO_interference_empire_backlash_situation_growth = {
	base = 0
	add = owner.value:AIO_interferometer_num
	add = owner.value:AIO_empire_proof_workforce
}

# scope: situation with a system target
AIO_interference_backlash_situation_growth = {
	base = 0
	add = target.value:AIO_interferometer_num
	add = target.value:AIO_proof_workforce
}

# scope: country
AIO_interferometer_num = {
	base = 0
	complex_trigger_modifier = {
		trigger = count_owned_megastructure
		parameters = {
			limit = { is_megastructure_type = AIO_reality_interferometer }
		}
		mode = add
	}
	min = 0
}

# scope: country
AIO_empire_proof_workforce = {
	complex_trigger_modifier = {
		trigger = total_country_workforce_with_job_tag
		parameters = { tags = { AIO_proof } }
		mode = add
	}
	divide = 100
	min = 0
}

# scope: system
AIO_proof_workforce = {
	complex_trigger_modifier = {
		trigger = total_country_workforce_with_job_tag
		trigger_scope = owner
		parameters = { tags = { AIO_proof } }
		mode = add
	}
	divide = 100
	min = 0
}
