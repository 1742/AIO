# Scope: an aura system. Growth and buffs use the same covering-source count.
AIO_reality_interference_source_growth = {
	base = 0
	add = AIO_interference_sources
	mult = 2
}

# scope: system
AIO_interference_aura_proof_growth = {
	complex_trigger_modifier = {
		trigger = total_system_workforce_with_job_tag
		parameters = {
			tags = { AIO_proof }
			limit = {
				colony.owner = { is_same_value = $OWNER$ }
			}
		}
		mode = add
	}
	divide = 100
	mult = 0.5
	max = 5
}

# Scope: a planet or fleet affected by the aura.

AIO_interference_intensity_multiplier = {
	base = 0
	add = solar_system.AIO_interference_sources
	mult = solar_system.trigger:is_aura_intensity_level
}

# scope: situation
AIO_interference_backlash_situation_growth = {
	base = 0
	add = owner.value:AIO_interferometer_num
	add = owner.value:AIO_proof_workforce
}

# scope: country
AIO_interferometer_num = {
	base = 0
	complex_trigger_modifier = {
		trigger = count_owned_megastructure
		parameters = {
			limit = { AIO_is_interference_source = yes }
		}
		mode = add
	}
	complex_trigger_modifier = {
		trigger = count_owned_nonprimary_starbase
		parameters = { limit = { AIO_is_interference_source = yes } }
		mode = add
	}
	complex_trigger_modifier = {
		trigger = count_owned_ship
		parameters = { limit = { AIO_is_mobile_interference_source = yes } }
		mode = add
	}
	min = 0
}

# Scope: country. Both module types share a cap, including construction queues.
AIO_interference_generator_capacity_used = {
	base = 0
	complex_trigger_modifier = {
		trigger = count_starbase_modules
		parameters = {
			type = AIO_arkship_interference_generator
			include_being_constructed = yes
		}
		mode = add
	}
	complex_trigger_modifier = {
		trigger = count_starbase_modules
		parameters = {
			type = AIO_waystation_interference_generator
			include_being_constructed = yes
		}
		mode = add
	}
}

# Scope: system. Count installations, including multiple arkships in one system.
AIO_interference_local_source_count = {
	base = 0
	complex_trigger_modifier = {
		trigger = count_system_megastructure
		parameters = {
			limit = {
				AIO_is_interference_source = yes
				owner = { is_same_value = $OWNER$ }
			}
		}
		mode = add
	}
	complex_trigger_modifier = {
		trigger = count_starbase_in_system
		parameters = {
			limit = {
				is_waystation_starbase = yes
				AIO_is_interference_source = yes
				owner = { is_same_value = $OWNER$ }
			}
		}
		mode = add
	}
	complex_trigger_modifier = {
		trigger = count_ship_in_system
		parameters = {
			limit = {
				AIO_is_mobile_interference_source = yes
				owner = { is_same_value = $OWNER$ }
			}
		}
		mode = add
	}
}
