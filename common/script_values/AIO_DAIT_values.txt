    modifier = { add = 1 check_variable = { which = AIO_DAIT_range value = 1 } }
    modifier = { add = 2 check_variable = { which = AIO_DAIT_range value = 2 } }
    modifier = { add = 3 check_variable = { which = AIO_DAIT_range value = 3 } }
    modifier = { add = 4 check_variable = { which = AIO_DAIT_range value = 4 } }
